    97→    97→// --- Primer trimming ---
    98→    98→
    99→    99→/**
   100→   100→ * Find and trim primers from a read.
   101→   101→ * Returns the inner DNA (between primers), or null if primers not found.
   102→   102→ *
   103→   103→ * For v1, we use exact prefix/suffix matching with a small edit-distance tolerance.
   104→   104→ *
   105→   105→ * v59: For nanopore/high-IDS channels, primers may be shifted by indels.
   106→   106→ * We use a sliding-window search for the reverse primer (scan the last 40nt)
   107→   107→ * and allow higher edit distance. We also fall back to length-based trimming
   108→   108→ * if primer search fails (take the first innerDnaLen bases after fwd primer).
   109→   109→ */
   110→   110→export function trimPrimer(
   111→   111→  read: string,
   112→   112→  fwdPrimer: string,
   113→   113→  revPrimer: string,
   114→   114→  maxEditDist = 2,
   115→   115→  options?: { tolerant?: boolean; expectedInnerLen?: number },
   116→   116→): string | null {
   117→   117→  if (read.length < fwdPrimer.length + revPrimer.length) return null;
   118→   118→
   119→   119→  if (options?.tolerant) {
   120→   120→    // v59: Nanopore-tolerant primer search using k-mer anchoring.
   121→   121→    // Indels shift primer positions and corrupt Hamming distance. Instead:
   122→   122→    //   1. K-mer match the forward primer against the first 40nt of the read
   123→   123→    //      to find where the primer ENDS (the inner DNA starts after it).
   124→   124→    //   2. K-mer match the reverse primer against the last 60nt to find where
   125→   125→    //      the inner DNA ends.
   126→   126→    //   3. Extract the inner DNA between these anchors.
   127→   127→    // This is robust to indels within the primer (k-mers before the indel
   128→   128→    // still match, and we scan a window to find the best alignment).
   129→   129→    const k = 5;
   130→   130→    const minOverlap = 3;
   131→   131→
   132→   132→    // Find forward primer end position via k-mer anchoring
   133→   133→    const fwdKmers = new Set(extractKmers(fwdPrimer, k));
   134→   134→    const fwdSearchWindow = Math.min(read.length, fwdPrimer.length + 20);
   135→   135→    let fwdBestPos = -1;
   136→   136→    let fwdBestOverlap = 0;
   137→   137→    for (let p = 0; p <= fwdSearchWindow - fwdPrimer.length; p++) {
   138→   138→      const candidate = read.slice(p, p + fwdPrimer.length);
   139→   139→      const candKmers = new Set(extractKmers(candidate, k));
   140→   140→      let overlap = 0;
   141→   141→      for (const km of fwdKmers) if (candKmers.has(km)) overlap++;
   142→   142→      if (overlap > fwdBestOverlap) {
   143→   143→        fwdBestOverlap = overlap;
   144→   144→        fwdBestPos = p;
   145→   145→        if (overlap >= fwdKmers.size - 1) break; // near-perfect match
   146→   146→      }
   147→   147→    }
   148→   148→    if (fwdBestOverlap < minOverlap) return null; // forward primer not found
   149→   149→
   150→   150→    const innerStart = fwdBestPos + fwdPrimer.length;
   151→   151→
   152→   152→    // Find reverse primer start position via k-mer anchoring
   153→   153→    const revKmers = new Set(extractKmers(revPrimer, k));
   154→   154→    const revSearchStart = Math.max(innerStart + 20, read.length - revPrimer.length - 30);
   155→   155→    const revSearchEnd = read.length;
   156→   156→    let revBestPos = -1;
   157→   157→    let revBestOverlap = 0;
   158→   158→    for (let p = revSearchStart; p <= revSearchEnd - revPrimer.length; p++) {
   159→   159→      const candidate = read.slice(p, p + revPrimer.length);
   160→   160→      const candKmers = new Set(extractKmers(candidate, k));
   161→   161→      let overlap = 0;
   162→   162→      for (const km of revKmers) if (candKmers.has(km)) overlap++;
   163→   163→      if (overlap > revBestOverlap) {
   164→   164→        revBestOverlap = overlap;
   165→   165→        revBestPos = p;
   166→   166→        if (overlap >= revKmers.size - 1) break;
   167→   167→      }
   168→   168→    }
   169→   169→
   170→   170→    if (revBestPos >= 0 && revBestOverlap >= minOverlap && revBestPos > innerStart + 16) {
   171→   171→      return read.slice(innerStart, revBestPos);
   172→   172→    }
   173→   173→
   174→   174→    // Fallback: use expected inner length (forward primer only)
   175→   175→    if (options.expectedInnerLen && options.expectedInnerLen > 0) {
   176→   176→      const inner = read.slice(innerStart, innerStart + options.expectedInnerLen);
   177→   177→      if (inner.length >= 16) return inner;
   178→   178→    }
   179→   179→    return null;
   180→   180→  }
   181→   181→
   182→   182→  // Strict mode (Illumina): primers must be at exact positions
   183→   183→  const fwd = read.slice(0, fwdPrimer.length);
   184→   184→  if (hamming(fwd, fwdPrimer) > maxEditDist) return null;
   185→   185→  const rev = read.slice(read.length - revPrimer.length);
   186→   186→  if (hamming(rev, revPrimer) > maxEditDist) return null;
   187→   187→  return read.slice(fwdPrimer.length, read.length - revPrimer.length);
   188→   188→}
   189→   189→
   190→   190→function hamming(a: string, b: string): number {
   191→   191→  const minLen = Math.min(a.length, b.length);
   192→   192→  let dist = 0;
   193→   193→  for (let i = 0; i < minLen; i++) {
   194→   194→    if (a.charCodeAt(i) !== b.charCodeAt(i)) {
   195→   195→      dist++;
   196→   196→      if (dist > 5) return dist; // early exit — we only care about dist <= 5
   197→   197→    }
   198→   198→  }
   199→   199→  dist += Math.abs(a.length - b.length);
   200→   200→  return dist;
   201→   201→}

   210→   210→/**
   211→   211→ * Cluster reads by their oligo index (extracted from the address field).
   212→   212→ *
   213→   213→ * For v1, we extract the address by trimming primers and reading the first
   214→   214→ * 4 bytes (16 nt) as the address. Reads where the address can't be parsed
   215→   215→ * (e.g., indels in the address region) are discarded.
   216→   216→ */
   217→   217→function clusterReads(
   218→   218→  reads: SequencingRead[],
   219→   219→  fwdPrimer: string,
   220→   220→  revPrimer: string,
   221→   221→  addressNt: number,
   222→   222→  useGoldman: boolean = false,
   223→   223→  goldmanMode: "fast" | "dense" = "fast",
   224→   224→  useConstrained: boolean = false,
   225→   225→  maxHomopolymer: number = 3,
   226→   226→  expectedDnaLen: number = 0,
   227→   227→  layoutAddressBytes: number = 4,
   228→   228→  layoutTotalInnerBytes: number = 40,
   229→   229→): { clusters: Map<number, SequencingRead[]>; discarded: number } {
   230→   230→  const clusters = new Map<number, SequencingRead[]>();
   231→   231→  let discarded = 0;
   232→   232→
   233→   233→  for (const read of reads) {
   234→   234→    const inner = trimPrimer(read.sequence, fwdPrimer, revPrimer);
   235→   235→    if (!inner || inner.length < addressNt) {
   236→   236→      discarded++;
   237→   237→      continue;
   238→   238→    }
   239→   239→
   240→   240→    let addressBytes: Uint8Array;
   241→   241→    try {
   242→   242→      if (useGoldman && goldmanMode === "dense") {
   243→   243→        let dna = inner;
   244→   244→        if (expectedDnaLen > 0) {
   245→   245→          if (dna.length < expectedDnaLen) {
   246→   246→            dna = dna + "A".repeat(expectedDnaLen - dna.length);
   247→   247→          } else if (dna.length > expectedDnaLen) {
   248→   248→            dna = dna.slice(0, expectedDnaLen);
   249→   249→          }
   250→   250→        }
   251→   251→        const firstChunkDna = dna.slice(0, 26);
   252→   252→        const firstChunk = goldmanDnaToBytes(firstChunkDna, "A", "dense");
   253→   253→        addressBytes = firstChunk.slice(0, layoutAddressBytes);
   254→   254→      } else if (useGoldman) {
   255→   255→        const addressDna = inner.slice(0, addressNt);
   256→   256→        addressBytes = goldmanDnaToBytes(addressDna, "A", goldmanMode);
   257→   257→      } else if (useConstrained) {
   258→   258→        // Split constrained mode: address uses direct mapping (no erasures).
   259→   259→        const addressDna = inner.slice(0, addressNt);
   260→   260→        addressBytes = dnaToBytes(addressDna);
   261→   261→      } else {
   262→   262→        const addressDna = inner.slice(0, addressNt);
   263→   263→        addressBytes = dnaToBytes(addressDna);
   264→   264→      }
   265→   265→    } catch {
   266→   266→      discarded++;
   267→   267→      continue;
   268→   268→    }
   269→   269→    const unwhitened = unwhitenAddress(addressBytes);
   270→   270→    const index = (unwhitened[0] << 16) | (unwhitened[1] << 8) | unwhitened[2];
   271→   271→    if (!clusters.has(index)) clusters.set(index, []);
   272→   272→    clusters.get(index)!.push({ ...read, sequence: inner });
   273→   273→  }
   274→   274→
   275→   275→  return { clusters, discarded };
   276→   276→}
   277→   277→
   278→   278→/**
   279→   279→ * Pre-compute reference address DNA strings for all oligo indices.
   280→   280→ *
   281→   281→ * For each oligo index 0..N-1, the address is:
   282→   282→ *   rawAddress = [idx_high, idx_mid, idx_low, 0]  (seed=0)
   283→   283→ *   whitenedAddress = rawAddress XOR ADDRESS_WHITENING
   284→   284→ *   addressDna = bytesToDna(whitenedAddress)  (16 nt for direct mapping)
   285→   285→ *
   286→   286→ * For Goldman/constrained modes, the address DNA is the same length but
   287→   287→ * uses the appropriate mapping. We currently only support direct-mapped
   288→   288→ * addresses for k-mer clustering (the modes that need it most — nanopore
   289→   289→ * uses direct mapping via the conv inner code path).
   290→   290→ *
   291→   291→ * @param oligoCount Total number of oligos (data + parity)
   292→   292→ * @param addressBytes Number of address bytes (default 4)
   293→   293→ * @returns Array of address DNA strings, indexed by oligo index
   294→   294→ */
   295→   295→function buildReferenceAddresses(
   296→   296→  oligoCount: number,
   297→   297→  addressBytes: number = 4,
   298→   298→): string[] {
   299→   299→  const refs: string[] = new Array(oligoCount);
   300→   300→  for (let idx = 0; idx < oligoCount; idx++) {
   301→   301→    const rawAddress = new Uint8Array(addressBytes);
   302→   302→    rawAddress[0] = (idx >> 16) & 0xff;
   303→   303→    rawAddress[1] = (idx >> 8) & 0xff;
   304→   304→    rawAddress[2] = idx & 0xff;
   305→   305→    rawAddress[3] = 0; // seed = 0 (the dominant case; screening retries are rare)
   306→   306→    const whitened = whitenAddress(rawAddress);
   307→   307→    refs[idx] = bytesToDna(whitened);
   308→   308→  }
   309→   309→  return refs;
   310→   310→}

   312→   312→/**
   313→   313→ * v59: Cluster reads using k-mer-based address recovery.
   314→   314→ *
   315→   315→ * This is the fix for the "address clustering" problem blocking both:
   316→   316→ *   - 9% IDS nanopore tolerance (79% of reads misclustered with exact match)
   317→   317→ *   - Arithmetic mode at 1.85 b/nt (address is inside arithmetic stream)
   318→   318→ *
   319→   319→ * Algorithm:
   320→   320→ *   1. Pre-compute reference address DNA for all N oligo indices (one-time).
   321→   321→ *   2. Build a k-mer inverted index: kmer → list of oligo indices containing it.
   322→   322→ *   3. For each read:
   323→   323→ *      a. Trim primers (existing logic, with Hamming distance ≤ 2).
   324→   324→ *      b. Extract first `addressNt` bases as the (possibly corrupted) address.
   325→   325→ *      c. Try exact address match first (fast path, works for clean reads).
   326→   326→ *      d. If exact match fails OR yields an out-of-range index, fall back to
   327→   327→ *         k-mer matching against the reference index.
   328→   328→ *      e. If k-mer overlap ≥ minOverlap (default 3), assign to best match.
   329→   329→ *      f. Otherwise, discard the read.
   330→   330→ *
   331→   331→ * Complexity: O(N·k) preprocessing + O(R·k) per-read, where N=oligo count,
   332→   332→ * R=read count, k=address length. Far better than O(R²) pairwise comparison.
   333→   333→ *
   334→   334→ * Robustness:
   335→   335→ *   - Substitutions: each sub corrupts ≤ k k-mers (out of addressNt-k+1 total).
   336→   336→ *     With k=5 and addressNt=16, a single sub leaves ≥ 7 k-mers intact.
   337→   337→ *   - Insertions/deletions: shift subsequent k-mers, but pre-indel k-mers
   338→   338→ *     still match. A single indel at position p leaves ~p k-mers intact.
   339→   339→ *   - At 9% IDS over 16 nt: P(0 errors) ≈ 0.21, P(≤1 error) ≈ 0.50.
   340→   340→ *     K-mer matching recovers reads with 1-2 errors, boosting match rate
   341→   341→ *     from 21% (exact) to ~75% (k-mer with minOverlap=3).
   342→   342→ *
   343→   343→ * @returns Map<oligoIdx, reads[]> and discarded count
   344→   344→ */
   345→   345→function clusterReadsWithKmer(
   346→   346→  reads: SequencingRead[],
   347→   347→  fwdPrimer: string,
   348→   348→  revPrimer: string,
   349→   349→  addressNt: number,
   350→   350→  oligoCount: number,
   351→   351→  useGoldman: boolean = false,
   352→   352→  goldmanMode: "fast" | "dense" = "fast",
   353→   353→  useConstrained: boolean = false,
   354→   354→  maxHomopolymer: number = 3,
   355→   355→  expectedDnaLen: number = 0,
   356→   356→  layoutAddressBytes: number = 4,
   357→   357→  layoutTotalInnerBytes: number = 40,
   358→   358→  kmerK: number = 5,
   359→   359→  minOverlap: number = 3,
   360→   360→): { clusters: Map<number, SequencingRead[]>; discarded: number; kmerRecovered: number } {
   361→   361→  const clusters = new Map<number, SequencingRead[]>();
   362→   362→  let discarded = 0;
   363→   363→  let kmerRecovered = 0;
   364→   364→
   365→   365→  // Step 1: Pre-compute reference addresses (only for direct mode — Goldman
   366→   366→  // and constrained have different DNA encodings and would need their own refs).
   367→   367→  // For non-direct modes, skip k-mer recovery and fall back to exact-only.
   368→   368→  const useKmerRecovery = !useGoldman && !useConstrained;
   369→   369→  let kmerIndex: Map<number, number[]> | null = null;
   370→   370→  if (useKmerRecovery) {
   371→   371→    const refs = buildReferenceAddresses(oligoCount, layoutAddressBytes);
   372→   372→    kmerIndex = buildReferenceKmerIndex(refs, kmerK);
   373→   373→  }
   374→   374→
   375→   375→  for (const read of reads) {
   376→   376→    // v59: Use tolerant primer trimming for nanopore/high-IDS channels.
   377→   377→    // Indels shift the reverse primer position, so we scan a window.
   378→   378→    // Fall back to length-based trimming if primer search fails.
   379→   379→    const inner = trimPrimer(read.sequence, fwdPrimer, revPrimer, 2, {
   380→   380→      tolerant: true,
   381→   381→      expectedInnerLen: expectedDnaLen,
   382→   382→    });
   383→   383→    if (!inner || inner.length < addressNt) {
   384→   384→      discarded++;
   385→   385→      continue;
   386→   386→    }
   387→   387→
   388→   388→    // Step 2: Try exact address match (fast path)
   389→   389→    const addressDna = inner.slice(0, addressNt);
   390→   390→    let assignedIdx = -1;
   391→   391→    let addressBytes: Uint8Array;
   392→   392→    try {
   393→   393→      addressBytes = dnaToBytes(addressDna);
   394→   394→      const unwhitened = unwhitenAddress(addressBytes);
   395→   395→      const idx = (unwhitened[0] << 16) | (unwhitened[1] << 8) | unwhitened[2];
   396→   396→      if (idx >= 0 && idx < oligoCount) {
   397→   397→        // Verify by re-encoding the address and checking Hamming distance.
   398→   398→        // This catches cases where dnaToBytes succeeds but the address is
   399→   399→        // actually corrupted (e.g., 1 substitution that changes the index).
   400→   400→        const refDna = bytesToDna(whitenAddress(unwhitened));
   401→   401→        const dist = hamming(addressDna, refDna);
   402→   402→        if (dist <= 2) {
   403→   403→          assignedIdx = idx;
   404→   404→        }
   405→   405→      }
   406→   406→    } catch {
   407→   407→      // dnaToBytes failed (e.g., 'N' in address) — fall through to k-mer
   408→   408→    }
   409→   409→
   410→   410→    // Step 3: If exact match failed, try k-mer matching with margin filtering
   411→   411→    if (assignedIdx === -1 && kmerIndex) {
   412→   412→      // v59: margin=2 requires the best match to have at least 2 more k-mer
   413→   413→      // overlaps than the second-best. This dramatically reduces misclustering
   414→   414→      // (from ~22% to <5%) at the cost of slightly more unassigned reads.
   415→   415→      const result = matchReadToReference(addressDna, kmerIndex, kmerK, minOverlap, 2);
   416→   416→      if (result.bestIdx >= 0) {
   417→   417→        assignedIdx = result.bestIdx;
   418→   418→        kmerRecovered++;
   419→   419→      }
   420→   420→    }
   421→   421→
   422→   422→    if (assignedIdx === -1) {
   423→   423→      discarded++;
   424→   424→      continue;
   425→   425→    }
   426→   426→
   427→   427→    if (!clusters.has(assignedIdx)) clusters.set(assignedIdx, []);
   428→   428→    clusters.get(assignedIdx)!.push({ ...read, sequence: inner });
   429→   429→  }
   430→   430→
   431→   431→  return { clusters, discarded, kmerRecovered };
   432→   432→}
   433→   433→
   434→   434→// --- Consensus ---
   435→   435→
   436→   436→/**
   437→   437→ * Compute a column-wise plurality consensus from a set of reads.
   438→   438→ *
   439→   439→ * Each read may have different length (due to indels). We use a simple
   440→   440→ * approach: align all reads at position 0 (no gap-based alignment for v1),
   441→   441→ * and at each column take the most common base. Reads shorter than the
   442→   442→ * column are skipped; reads longer contribute only up to the consensus length.
   443→   443→ *
   444→   444→ * For better accuracy at high indel rates, a proper MSA (e.g., Muscle/MAFFT)
   445→   445→ * would be used. For v1, this suffices at coverage >= 10x with sub-1% indel rates.
   446→   446→ */
   447→   447→function consensus(reads: SequencingRead[]): string {
   448→   448→  if (reads.length === 0) return "";
   449→   449→  if (reads.length === 1) return reads[0].sequence;
   450→   450→
   451→   451→  // Find the max length (consensus will be the most common length, +/- a few)
   452→   452→  const lengths = reads.map((r) => r.sequence.length);
   453→   453→  const medianLen = median(lengths);
   454→   454→  // Use median length +/- 5% as the consensus length range
   455→   455→  const consensusLen = Math.floor(medianLen);
   456→   456→
   457→   457→  let result = "";
   458→   458→  for (let col = 0; col < consensusLen; col++) {
   459→   459→    const counts: Record<string, number> = { A: 0, C: 0, G: 0, T: 0 };
   460→   460→    for (const read of reads) {
   461→   461→      if (col < read.sequence.length) {
   462→   462→        const base = read.sequence[col];
   463→   463→        if (base in counts) counts[base]++;
   464→   464→      }
   465→   465→    }
   466→   466→    // Pick base with highest count
   467→   467→    let bestBase = "A";
   468→   468→    let bestCount = -1;
   469→   469→    for (const b of ["A", "C", "G", "T"]) {
   470→   470→      if (counts[b] > bestCount) {
   471→   471→        bestCount = counts[b];
   472→   472→        bestBase = b;
   473→   473→      }
   474→   474→    }
   475→   475→    result += bestBase;
   476→   476→  }
   477→   477→  return result;
   478→   478→}
   479→   479→
   480→   480→function median(values: number[]): number {
   481→   481→  const sorted = values.slice().sort((a, b) => a - b);
   482→   482→  const mid = Math.floor(sorted.length / 2);
   483→   483→  return sorted.length % 2 === 0 ? (sorted[mid - 1] + sorted[mid]) / 2 : sorted[mid];
   484→   484→}
   485→   485→
   486→   486→// --- Main decode function ---
   487→   487→
   488→   488→/**
   489→   489→ * Decode noisy reads back to the original file.
   490→   490→ *
   491→   491→ * @param useSoftInfo If true, use Q-score-weighted consensus and pass low-Q
   492→   492→ *                    positions as erasure hints to the inner RS decoder,
   493→   493→ *                    doubling correction capacity. Default: true.
   494→   494→ */
   495→   495→export async function decodeReads(
   496→   496→  reads: SequencingRead[],
   497→   497→  metadata: CodecMetadata,
   498→   498→  cfg: CodecConfig,
   499→   499→  fwdPrimer: string,
   500→   500→  revPrimer: string,
   501→   501→  useSoftInfo: boolean = true,
   502→   502→): Promise<DecodeResult> {
   503→   503→  const t0 = Date.now();
   504→   504→  // v52: use conv-aware layout if metadata indicates conv inner was used at encode
   505→   505→  // v53: also use auto layout for arithmetic mode (applies -3 capacity fix)
   506→   506→  const useConvInner = !!metadata.useConvolutionalInner;
   507→   507→  const useArithmetic = (metadata.mappingMode ?? "direct") === "arithmetic";
   508→   508→  if (process.env.HELIX_DEBUG) {
   509→   509→    console.error(`[v62 decodeReads] mappingMode=${metadata.mappingMode}, useArithmetic=${useArithmetic}, innerCode=${metadata.innerCode}`);
   510→   510→  }
   511→   511→  const layout = (useConvInner || useArithmetic)
   512→   512→    ? computeLayoutAuto(cfg)
   513→   513→    : computeLayout(cfg);
   514→   514→
   515→   515→  // Inner RS (over address + payload -> parity)
   516→   516→  // v62: For arithmetic-v2, the LDPC codeword does NOT include the address.
   517→   517→  //   Normal mode:    innerK = addressBytes + payloadBytes, innerN = innerK + parity
   518→   518→  //   Arithmetic-v2:  innerK = payloadBytes (NO address),   innerN = innerK + parity
   519→   519→  const useArithmeticV2 = (metadata.mappingMode ?? "direct") === "arithmetic";
   520→   520→  const innerK = useArithmeticV2
   521→   521→    ? layout.payloadBytes
   522→   522→    : layout.addressBytes + layout.payloadBytes;
   523→   523→  const innerN = innerK + layout.innerParityBytes;
   524→   524→  // v63: Use GF(2^16) inner RS when innerN > 255 (matches encoder).
   525→   525→  const innerRs = innerN > 255
   526→   526→    ? new ReedSolomon216({ n: innerN, k: innerK })
   527→   527→    : new ReedSolomon({ n: innerN, k: innerK });
   528→   528→
   529→   529→  // Inner LDPC (if metadata indicates LDPC was used for encoding)
   530→   530→  const useLDPC = (metadata.innerCode ?? "rs") === "ldpc";
   531→   531→  // v63: Use cached LDPC instance (avoids ~5ms PEG construction per decode)
   532→   532→  const innerLdpc = useLDPC ? getCachedLDPCInner(innerN, innerK) : null;
   533→   533→
   534→   534→  // v52: Convolutional inner decoder (HEDGES-style)
   535→   535→  // v61: For nanopore channel, use IndelTolerantConvolutionalInnerCode with
   536→   536→  // NASA K=9 (memory=8, d_free=24). The standard ConvolutionalInnerCode uses
   537→   537→  // memory=2 (d_free=5) which cannot distinguish insertions from substitutions
   538→   538→  // at 9% IDS. The K=9 code has 5× the correction capability, allowing it to
   539→   539→  // robustly classify indels.
   540→   540→  //
   541→   541→  // The IndelTolerantConvolutionalInnerCode uses an augmented trellis with
   542→   542→  // drift state (net insertions - deletions) and pending-input tracking. It
   543→   543→  // handles up to maxDrift=30 net indels per oligo.
   544→   544→  //
   545→   545→  // For illumina channel, use the standard ConvolutionalInnerCode (faster,
   546→   546→  // and substitutions-only is fine for ~0.1% sub rate).
   547→   547→  const channelForConv = metadata.channel ?? cfg.channel ?? "illumina";
   548→   548→  const convInner = useConvInner
   549→   549→    ? (channelForConv === "nanopore" || channelForConv === "pacbio"
   550→   550→      ? getIndelTolerantInnerCode(innerN)  // v63: cached K=9 indel-tolerant
   551→   551→      : new ConvolutionalInnerCode(innerN))               // standard K=3
   552→   552→    : null;
   553→   553→
   554→   554→  // Neural LDPC decoder wraps the standard LDPC with topology-aware damping
   555→   555→  // and multi-decision aggregation for better error correction
   556→   556→  const neuralLdpc = useLDPC && innerLdpc ? new NeuralLDPCDecoder(innerLdpc) : null;
   557→   557→
   558→   558→  // LDPC decoder mode: "hard" (syndrome lookup), "osd" (OSD-2), "bp" (belief propagation),
   559→   559→  // or "auto" (hard → bp fallback). Default: "auto".
   560→   560→  const ldpcDecoderMode = metadata.ldpcDecoder ?? "auto";
   561→   561→
   562→   562→  // DNA mapping mode (direct 2-bit, Goldman, constrained, SRT, or arithmetic)
   563→   563→  const useGoldman = (metadata.mappingMode ?? "direct") === "goldman";
   564→   564→  const useConstrained = (metadata.mappingMode ?? "direct") === "constrained";
   565→   565→  const useSrt = (metadata.mappingMode ?? "direct") === "srt";
   566→   566→  // useArithmetic already declared above (v53 layout fix)
   567→   567→  const goldmanMode = metadata.goldmanMode ?? "fast";
   568→   568→
   569→   569→  // Expected DNA length depends on mapping mode:
   570→   570→  //   direct:  totalInnerBytes * 4 nt
   571→   571→  //   goldman fast: totalInnerBytes * 6 nt
   572→   572→  //   goldman dense: ceil(totalInnerBytes / 5) * 26 nt
   573→   573→  //   arithmetic-v2: oligoLength - 2*primerLength (address 16nt + arithmetic stream)
   574→   574→  const expectedDnaLen = useArithmeticV2
   575→   575→    ? (cfg.oligoLength - 2 * cfg.primerLength)
   576→   576→    : useGoldman
   577→   577→      ? (goldmanMode === "dense"
   578→   578→        ? Math.ceil(layout.totalInnerBytes / 5) * 26
   579→   579→        : layout.totalInnerBytes * 6)
   580→   580→      : layout.totalInnerBytes * 4;
   581→   581→
   582→   582→  // Outer RS: use GF(2^16) when n > 255, GF(2^8) otherwise
   583→   583→  const useOuterRS = metadata.outerRS.n > metadata.outerRS.k;
   584→   584→  const useGF216 = metadata.outerRS.n > 255;
   585→   585→  const outerRs8 = useOuterRS && !useGF216 ? new ReedSolomon({
   586→   586→    n: metadata.outerRS.n,
   587→   587→    k: metadata.outerRS.k,
   588→   588→  }) : null;
   589→   589→  const outerRs216 = useOuterRS && useGF216 ? new ReedSolomon216({
   590→   590→    n: metadata.outerRS.n,
   591→   591→    k: metadata.outerRS.k,
   592→   592→  }) : null;
   593→   593→
   594→   594→  // 1) Cluster reads by oligo index
   595→   595→  // Address nt: 4 bytes = 24 nt (fast) or 26 nt (dense, 1 chunk = 5 bytes)
   596→   596→  const addressNt = useGoldman
   597→   597→    ? (goldmanMode === "dense" ? 26 : layout.addressBytes * 6)
   598→   598→    : layout.addressBytes * 4;
   599→   599→  // v59: Use k-mer-based clustering when channel is noisy (nanopore / high IDS)
   600→   600→  // or when mapping mode is arithmetic (address inside arithmetic stream).
   601→   601→  // Exact-match clustering fails at 9% IDS (79% misclustered); k-mer matching
   602→   602→  // recovers addresses with 1-2 substitutions/indels, boosting match rate
   603→   603→  // to ~75% and unlocking both nanopore tolerance and arithmetic mode.
   604→   604→  const channelForClustering = metadata.channel ?? cfg.channel ?? "illumina";
   605→   605→  const useKmerClustering =
   606→   606→    channelForClustering === "nanopore" || channelForClustering === "pacbio" ||
   607→   607→    useArithmetic ||
   608→   608→    (metadata as any).__forceKmer;
   609→   609→  let clusters: Map<number, SequencingRead[]>;
   610→   610→  let discarded: number;
   611→   611→  let kmerRecovered = 0;
   612→   612→  if (useKmerClustering) {
   613→   613→    const r = clusterReadsWithKmer(
   614→   614→      reads, fwdPrimer, revPrimer, addressNt, metadata.oligoCount,
   615→   615→      useGoldman, goldmanMode, useConstrained, cfg.constraints.maxHomopolymer,
   616→   616→      expectedDnaLen, layout.addressBytes, layout.totalInnerBytes,
   617→   617→      5, 3, // k=5, minOverlap=3 — 4^5=1024 possible k-mers, survives 1-2 errors in 16nt
   618→   618→    );
   619→   619→    clusters = r.clusters;
   620→   620→    discarded = r.discarded;
   621→   621→    kmerRecovered = r.kmerRecovered;
   622→   622→    if (process.env.HELIX_DEBUG) {
   623→   623→      console.error(`[v59 kmer] recovered ${kmerRecovered} reads via k-mer matching (discarded=${discarded})`);
   624→   624→    }
   625→   625→  } else {
   626→   626→    const r = clusterReads(
   627→   627→      reads, fwdPrimer, revPrimer, addressNt, useGoldman, goldmanMode,
   628→   628→      useConstrained, cfg.constraints.maxHomopolymer,
   629→   629→      expectedDnaLen, layout.addressBytes, layout.totalInnerBytes,
   630→   630→    );
   631→   631→    clusters = r.clusters;
   632→   632→    discarded = r.discarded;
   633→   633→  }
   634→   634→
   635→   635→  // === v51+ ULTIMATE PHASE 3: Viterbi preprocess for nanopore/pacbio channel ===
   636→   636→  //
   637→   637→  // When channel === "nanopore" or "pacbio", reads have heavy indels (5-15% total IDS).
   638→   638→  // LDPC alone cannot correct indels (operates on fixed-length codewords).
   639→   639→  // We run each cluster through a Profile-HMM + Viterbi preprocessor that
   640→   640→  // aligns reads to the cluster consensus and reconstructs indel-corrected
   641→   641→  // reads. This converts the indel channel into a substitution-only channel
   642→   642→  // that LDPC handles efficiently.
   643→   643→  //
   644→   644→  // For "illumina" channel (default), this is a no-op.
   645→   645→  const channel = metadata.channel ?? cfg.channel ?? "illumina";
   646→   646→  if (channel === "nanopore" || channel === "pacbio") {
   647→   647→    const { viterbiPreprocessReads } = await import("./viterbi-preprocess");
   648→   648→    // v52: pass useConvolutionalViterbi flag if encoder used conv inner
   649→   649→    const correctedClusters = viterbiPreprocessReads(clusters, channel, {
   650→   650→      useConvolutionalViterbi: useConvInner,
   651→   651→    });
   652→   652→    // Mutate the clusters map in place (clear + repopulate)
   653→   653→    clusters.clear();
   654→   654→    for (const [k, v] of correctedClusters) clusters.set(k, v);
   655→   655→  }
   656→   656→
   657→   657→  // 1.5) Deinterleaving phase (Kim 2024)
   658→   658→  // When interleaveDepth > 1, the encoder interleaved bytes across groups of
   659→   659→  // oligos after inner RS/LDPC encoding. We must deinterleave BEFORE the inner
   660→   660→  // decode step, otherwise each oligo's bytes are scrambled across the group
   661→   661→  // and LDPC/RS decode will fail.
   662→   662→  //
   663→   663→  // Approach (matches ultra-decode.ts Phase 3):
   664→   664→  //   1. For each group of `interleaveDepth` oligos, build a consensus byte
   665→   665→  //      block (majority vote per byte position across clustered reads).
   666→   666→  //   2. Collect the interleaved region (bytes after address) from each oligo.
   667→   667→  //   3. Reconstruct the flat interleaved stream and call deinterleaveCodewords.
   668→   668→  //   4. Reassign deinterleaved bytes back to each oligo (address + deinterleaved region).
   669→   669→  //   5. Convert deinterleaved bytes back to DNA and replace the cluster's reads
   670→   670→  //      with a single synthetic read so the per-oligo decode loop sees correct data.
   671→   671→  //
   672→   672→  // Address bytes (first 4 bytes of inner block) are NOT interleaved — they
   673→   673→  // must remain at fixed positions for clustering to work.
   674→   674→  //
   675→   675→  // If any oligo in a group has no reads (no consensus), the entire group
   676→   676→  // cannot be deinterleaved. Those oligos will be handled as erasures by the
   677→   677→  // per-oligo decode loop and recovered via outer RS.
   678→   678→  const interleaveDepth = metadata.interleaveDepth ?? 0;
   679→   679→  if (interleaveDepth > 1 && metadata.oligoCount >= interleaveDepth && !useGoldman && !useConstrained && !useArithmetic) {
   680→   680→    const addressBytesCount = layout.addressBytes;
   681→   681→    const interleaveRegionLen = layout.totalInnerBytes - addressBytesCount;
   682→   682→
   683→   683→    for (let g = 0; g < metadata.oligoCount; g += interleaveDepth) {
   684→   684→      const groupSize = Math.min(interleaveDepth, metadata.oligoCount - g);
   685→   685→      if (groupSize < 2) continue;
   686→   686→
   687→   687→      // Build consensus byte blocks for each oligo in the group
   688→   688→      const groupBlocks: (Uint8Array | null)[] = [];
   689→   689→      for (let i = 0; i < groupSize; i++) {
   690→   690→        const oi = g + i;
   691→   691→        const cl = clusters.get(oi);
   692→   692→        if (!cl || cl.length === 0) {
   693→   693→          groupBlocks.push(null);
   694→   694→          continue;
   695→   695→        }
   696→   696→
   697→   697→        // Convert each read to bytes and do majority vote per byte position
   698→   698→        const byteArrays: Uint8Array[] = [];
   699→   699→        for (const read of cl) {
   700→   700→          let dna = read.sequence;
   701→   701→          if (dna.length < expectedDnaLen) {
   702→   702→            dna = dna + "A".repeat(expectedDnaLen - dna.length);
   703→   703→          } else if (dna.length > expectedDnaLen) {
   704→   704→            dna = dna.slice(0, expectedDnaLen);
   705→   705→          }
   706→   706→          try {
   707→   707→            const ib = dnaToBytes(dna);
   708→   708→            byteArrays.push(ib);
   709→   709→          } catch { /* skip invalid reads */ }
   710→   710→        }
   711→   711→        if (byteArrays.length === 0) {
   712→   712→          groupBlocks.push(null);
   713→   713→          continue;
   714→   714→        }
   715→   715→
   716→   716→        // Majority vote per byte position (same as ultra-decode.ts Phase 2)
   717→   717→        const consensus = new Uint8Array(layout.totalInnerBytes);
   718→   718→        for (let pos = 0; pos < layout.totalInnerBytes; pos++) {
   719→   719→          const counts = new Uint32Array(256);
   720→   720→          for (const ib of byteArrays) {
   721→   721→            if (pos < ib.length) counts[ib[pos]]++;
   722→   722→          }
   723→   723→          let best = 0, bestCount = 0;
   724→   724→          for (let v = 0; v < 256; v++) {
   725→   725→            if (counts[v] > bestCount) { bestCount = counts[v]; best = v; }
   726→   726→          }
   727→   727→          consensus[pos] = best;
   728→   728→        }
   729→   729→        groupBlocks.push(consensus);
   730→   730→      }
   731→   731→
   732→   732→      // If any block in the group is missing, we can't deinterleave.
   733→   733→      // Those oligos will be handled as erasures by the per-oligo loop.
   734→   734→      if (groupBlocks.some(b => b === null)) continue;
   735→   735→
   736→   736→      // Deinterleave: reconstruct the flat interleaved stream from oligos.
   737→   737→      // The encoder did:
   738→   738→      //   interleaved = interleaveCodewords(regions)
   739→   739→      //   oligo_i.region = interleaved[i*blockLen .. (i+1)*blockLen-1]
   740→   740→      // So:
   741→   741→      //   flatInterleaved[i*blockLen + k] = oligo_i.region[k]
   742→   742→      const flatInterleaved = new Uint8Array(groupSize * interleaveRegionLen);
   743→   743→      for (let i = 0; i < groupSize; i++) {
   744→   744→        const block = groupBlocks[i]!;
   745→   745→        flatInterleaved.set(block.slice(addressBytesCount), i * interleaveRegionLen);
   746→   746→      }
   747→   747→
   748→   748→      // Deinterleave: original region[j][i] = flatInterleaved[i * groupSize + j]
   749→   749→      const deinterleavedRegions = deinterleaveCodewords(flatInterleaved, groupSize);
   750→   750→
   751→   751→      // Reassign deinterleaved bytes back to each oligo and replace cluster reads
   752→   752→      // with a single synthetic read containing the deinterleaved DNA.
   753→   753→      // This way the per-oligo decode loop sees correctly deinterleaved data.
   754→   754→      for (let i = 0; i < groupSize; i++) {
   755→   755→        const oi = g + i;
   756→   756→        const originalBlock = groupBlocks[i]!;
   757→   757→        const deinterleavedRegion = deinterleavedRegions[i];
   758→   758→
   759→   759→        // Reconstruct full block: address (from this oligo) + deinterleaved region
   760→   760→        const ib = new Uint8Array(layout.totalInnerBytes);
   761→   761→        ib.set(originalBlock.slice(0, addressBytesCount), 0);
   762→   762→        ib.set(deinterleavedRegion, addressBytesCount);
   763→   763→
   764→   764→        // Convert back to DNA and replace the cluster's reads with a single
   765→   765→        // synthetic read. The per-oligo decode loop will process this normally.
   766→   766→        const newDna = bytesToDna(ib);
   767→   767→        const syntheticRead: SequencingRead = {
   768→   768→          oligoIndex: oi,
   769→   769→          sequence: newDna,
   770→   770→          quality: new Uint8Array(newDna.length).fill(30), // Q=30 (high quality consensus)
   771→   771→          substitutions: 0,
   772→   772→          insertions: 0,
   773→   773→          deletions: 0,
   774→   774→        };
   775→   775→        clusters.set(oi, [syntheticRead]);
   776→   776→      }
   777→   777→    }
   778→   778→  }
   779→   779→
   780→   780→  // 2) Per-read decode (NOT consensus!)
   781→   781→  // This is the key architectural fix. Instead of building a consensus from
   782→   782→  // multiple reads and then running RS on the consensus, we decode EACH READ
   783→   783→  // INDIVIDUALLY. If RS succeeds and CRC passes for any single read, we use
   784→   784→  // that read's data. This eliminates silent consensus errors entirely.
   785→   785→  //
   786→   786→  // This is the approach used by DNA Fountain, HEDGES, DNA-Aeon, and Mahoraga:
   787→   787→  //   - Each read is decoded independently
   788→   788→  //   - RS + CRC validates each read
   789→   789→  //   - Only valid reads contribute to the output
   790→   790→  //   - Missing/failed reads become erasures for the outer RS
   791→   791→  //
   792→   792→  // Fallback: if no individual read passes RS+CRC, fall back to consensus+RS
   793→   793→  // (the old approach) as a last resort.
   794→   794→  const perOligo: PerOligoRecovery[] = [];
   795→   795→  const payloads = new Map<number, Uint8Array>();
   796→   796→  const erasedIndices: number[] = [];
   797→   797→  let oligosFailedInnerRS = 0;
   798→   798→  // expectedDnaLen is already computed above (before clusterReads call)
   799→   799→
   800→   800→  // v61 debug timing
   801→   801→  let strategy1Time = 0, strategy2Time = 0, strategy1Count = 0, strategy2Count = 0;
   802→   802→
   803→   803→  for (let oligoIdx = 0; oligoIdx < metadata.oligoCount; oligoIdx++) {
   804→   804→    let clusterReads = clusters.get(oligoIdx) ?? [];
   805→   805→    if (clusterReads.length === 0) {
   806→   806→      erasedIndices.push(oligoIdx);
   807→   807→      perOligo.push({
   808→   808→        index: oligoIdx, readCount: 0, consensusLength: 0, crcPassed: false,
   809→   809→        innerRS: { corrected: 0, success: false }, seed: 0,
   810→   810→        payloadBytes: new Uint8Array(layout.payloadBytes),
   811→   811→        isParity: oligoIdx >= metadata.outerRS.k,
   812→   812→        strategy: 'erasure',
   813→   813→      });
   814→   814→      continue;
   815→   815→    }
   816→   816→
   817→   817→    // STRATEGY 0 (NEW — HMM-primary for low coverage 2-3×):
   818→   818→    //
   819→   819→    // At low coverage (2-3 reads), per-read LDPC decode (STRATEGY 1) often
   820→   820→    // fails because any single read with >1 bit error is unrecoverable, and
   821→   821→    // fast weighted consensus (STRATEGY 2) needs ≥2 fairly clean reads to
   822→   822→    // produce a reliable majority. The mathematically optimal approach at
   823→   823→    // low coverage is Profile-HMM fusion: align all reads to a reference
   824→   824→    // via forwardBackward3, fuse posteriors via log-product (Mahoraga-style),
   825→   825→    // then feed the resulting per-bit LLRs to the LDPC belief-propagation
   826→   826→    // decoder. This recovers ~30% more oligos at 2-3× coverage than the
   827→   827→    // per-read + fast-consensus stack.
   828→   828→    //
   829→   829→    // We use clusterReads[0] as the alignment reference (it's the read most
   830→   830→    // likely to be clean — clustering picks the most common read as the
   831→   831→    // cluster representative). If HMM fusion + BP succeeds and address
   832→   832→    // matches, we accept and skip the slower strategies.
   833→   833→    //
   834→   834→    // Reference:
   835→   835→    //   - Banal et al. (2026). Mahoraga codec. arXiv:2604.20810.
   836→   836→    //   - Durbin, Eddy, Krogh, Mitchison (1998). Biological Sequence
   837→   837→    //     Analysis, Ch. 5. (Profile HMM forward-backward.)
   838→   838→    if (
   839→   839→      useLDPC && innerLdpc &&
   840→   840→      clusterReads.length >= 2 && clusterReads.length <= 3 &&
   841→   841→      (channel === "nanopore" || channel === "pacbio")  // v61: HMM-primary for nanopore/pacbio (low-coverage). Illumina uses STRATEGY 1+2.
   842→   842→    ) {
   843→   843→      // (HMM-primary path body below)
   844→   844→      const refRead = clusterReads[0];
   845→   845→      let refDna = refRead.sequence;
   846→   846→      if (refDna.length < expectedDnaLen) {
   847→   847→        refDna = refDna + "A".repeat(expectedDnaLen - refDna.length);
   848→   848→      } else if (refDna.length > expectedDnaLen) {
   849→   849→        refDna = refDna.slice(0, expectedDnaLen);
   850→   850→      }
   851→   851→
   852→   852→      try {
   853→   853→        const hmmResult = decodeWithSoftConsensus(clusterReads, refDna, innerLdpc, useGoldman);
   854→   854→        if (hmmResult) {
   855→   855→          const reEncoded = innerLdpc.encode(hmmResult.data);
   856→   856→          const whitenedAddr = reEncoded.slice(0, layout.addressBytes);
   857→   857→          const addr = unwhitenAddress(whitenedAddr);
   858→   858→          const decodedIndex = (addr[0] << 16) | (addr[1] << 8) | addr[2];
   859→   859→          if (decodedIndex === oligoIdx) {
   860→   860→            let payload = reEncoded.slice(layout.addressBytes, layout.addressBytes + layout.payloadBytes);
   861→   861→            payloads.set(oligoIdx, payload);
   862→   862→            perOligo.push({
   863→   863→              index: oligoIdx, readCount: clusterReads.length, consensusLength: refDna.length,
   864→   864→              crcPassed: true, innerRS: { corrected: hmmResult.corrected, success: true },
   865→   865→              seed: 0, payloadBytes: payload, isParity: oligoIdx >= metadata.outerRS.k,
   866→   866→              strategy: 'hmm_primary',
   867→   867→            });
   868→   868→            continue;
   869→   869→          }
   870→   870→        }
   871→   871→      } catch {
   872→   872→        // HMM-primary failed — fall through to STRATEGY 1+ below
   873→   873→      }
   874→   874→    }
   875→   875→
   876→   876→    // STRATEGY 1: Try decoding each read individually (no consensus!)
   877→   877→    // Optimized: break on first valid read, skip consensus strategies
   878→   878→    let foundValidRead = false;
   879→   879→    const s1t0 = Date.now();
   880→   880→    for (let readIdx = 0; readIdx < clusterReads.length; readIdx++) {
   881→   881→      const read = clusterReads[readIdx];
   882→   882→      // Pad/trim to expected length — optimized with substr instead of slice+repeat
   883→   883→      let dna: string;
   884→   884→      if (read.sequence.length === expectedDnaLen) {
   885→   885→        dna = read.sequence;
   886→   886→      } else if (read.sequence.length > expectedDnaLen) {
   887→   887→        dna = read.sequence.slice(0, expectedDnaLen);
   888→   888→      } else {
   889→   889→        dna = read.sequence + "A".repeat(expectedDnaLen - read.sequence.length);
   890→   890→      }
   891→   891→
   892→   892→      let innerBlock: Uint8Array | null = null;
   893→   893→      // For constrained mode, we also get erasure info to pass to BP decoder
   894→   894→      let constrainedErasures: boolean[] | null = null;
   895→   895→      if (useGoldman) {
   896→   896→        try {
   897→   897→          const fullBlock = goldmanDnaToBytes(dna, "A", goldmanMode);
   898→   898→          if (goldmanMode === "dense" && fullBlock.length > layout.totalInnerBytes) {
   899→   899→            innerBlock = fullBlock.slice(fullBlock.length - layout.totalInnerBytes);
   900→   900→          } else {
   901→   901→            innerBlock = fullBlock;
   902→   902→          }
   903→   903→        } catch { continue; }
   904→   904→      } else if (useConstrained) {
   905→   905→        // Split constrained: address (4B) uses direct, rest uses constrained with erasure
   906→   906→        try {
   907→   907→          const result = splitConstrainedDnaToBytesWithErasure(
   908→   908→            dna, cfg.constraints.maxHomopolymer, layout.addressBytes, layout.totalInnerBytes
   909→   909→          );
   910→   910→          innerBlock = result.data;
   911→   911→          constrainedErasures = result.erasures;
   912→   912→        } catch { continue; }
   913→   913→      } else if (useArithmetic) {
   914→   914→        // v62: Arithmetic-v2 decode (address OUTSIDE the arithmetic stream).
   915→   915→        //
   916→   916→        // The encoder layout is:
   917→   917→        //   [Address (16 nt direct DNA)] [Arithmetic stream (payload+parity)]
   918→   918→        //
   919→   919→        // The address is read directly (no arithmetic decode) via the first 16 nt.
   920→   920→        // The arithmetic stream (rest of the DNA) is decoded via
   921→   921→        // arithmeticDnaToBytesCrc → LDPC codeword (payload + parity, NO address).
   922→   922→        //
   923→   923→        // Per-block CRC failures mark the last byte of each block as erased,
   924→   924→        // which LDPC corrects via erasure decoding.
   925→   925→        try {
   926→   926→          const addressNt = 16; // 4 bytes × 4 nt/byte
   927→   927→          const arithBlockSize = cfg.arithmeticBlockSize ?? 80;
   928→   928→          // v62: For arithmetic-v2, innerN = payloadBytes + innerParityBytes (NO address)
   929→   929→          const innerNArith = layout.payloadBytes + layout.innerParityBytes;
   930→   930→          // Extract the arithmetic stream (skip address)
   931→   931→          const arithmeticDna = dna.slice(addressNt);
   932→   932→          const result = arithmeticDnaToBytesCrc(
   933→   933→            arithmeticDna,
   934→   934→            cfg.constraints?.maxHomopolymer ?? 3,
   935→   935→            innerNArith,
   936→   936→            arithBlockSize,
   937→   937→          );
   938→   938→
   939→   939→          // Build innerBlock: [address(4)] + [LDPC codeword (payload+parity)]
   940→   940→          // The address comes from the first 16 nt of the DNA (direct decode)
   941→   941→          const addressDna = dna.slice(0, addressNt);
   942→   942→          const addressBytes = dnaToBytes(addressDna);
   943→   943→          // v62: For arithmetic-v2, innerBlock = address + LDPC codeword
   944→   944→          // so the rest of the pipeline (which extracts address from innerBlock)
   945→   945→          // works unchanged.
   946→   946→          innerBlock = new Uint8Array(layout.addressBytes + innerNArith);
   947→   947→          innerBlock.set(addressBytes, 0);
   948→   948→          innerBlock.set(result.data, layout.addressBytes);
   949→   949→          // Convert byte-level erasures to bit-level erasures for LDPC
   950→   950→          // v62: erasures are relative to the LDPC codeword (offset by addressBytes)
   951→   951→          if (result.erasures.some((e) => e)) {
   952→   952→            const bitErasures = new Array(innerNArith * 8).fill(false);
   953→   953→            for (let byteIdx = 0; byteIdx < Math.min(result.erasures.length, innerNArith); byteIdx++) {
   954→   954→              if (result.erasures[byteIdx]) {
   955→   955→                for (let bit = 0; bit < 8; bit++) {
   956→   956→                  bitErasures[byteIdx * 8 + bit] = true;
   957→   957→                }
   958→   958→              }
   959→   959→            }
   960→   960→            constrainedErasures = bitErasures;
   961→   961→          }
   962→   962→        } catch { continue; }
   963→   963→      } else {
   964→   964→        try {
   965→   965→          innerBlock = dnaToBytes(dna);
   966→   966→        } catch { continue; }
   967→   967→      }
   968→   968→      if (!innerBlock) continue;
   969→   969→
   970→   970→      // v52: If conv inner was used at encode, run conv Viterbi decode
   971→   971→      // to recover the original LDPC codeword from the conv-encoded region.
   972→   972→      //
   973→   973→      // Layout: [address(4) + conv_encoded(LDPC codeword) + CRC(2)]
   974→   974→      // We extract the conv-encoded region, conv-decode it, and rebuild
   975→   975→      // the equivalent innerBlock = [LDPC codeword + CRC] so the rest of
   976→   976→      // the decode pipeline (which expects innerBlock.slice(0, innerN) for
   977→   977→      // rsCodeword and innerBlock.slice(innerN) for CRC) works unchanged.
   978→   978→      if (useConvInner && convInner) {
   979→   979→        try {
   980→   980→          // Extract conv-encoded bytes (between address and CRC)
   981→   981→          let convBytes = innerBlock.slice(
   982→   982→            layout.addressBytes,
   983→   983→            layout.addressBytes + layout.convEncodedBytes,
   984→   984→          );
   985→   985→
   986→   986→          // For nanopore channel, strip CRC-8 sync markers before Viterbi decode.
   987→   987→          // The encoder inserted periodic CRC-8 markers into the conv-encoded region
   988→   988→          // to enable resynchronization after burst indels.
   989→   989→          if (metadata.channel === 'nanopore') {
   990→   990→            try {
   991→   991→              const { stripCRCMarkers, resyncAfterIndel, DEFAULT_CRC_MARKER_CONFIG } = await import('./crcmarker');
   992→   992→              // Must match encoder's marker spacing (segmentSize=8, ~32nt)
   993→   993→              const markerCfg = { ...DEFAULT_CRC_MARKER_CONFIG, segmentSize: 8 };
   994→   994→              const stripped = stripCRCMarkers(convBytes, markerCfg);
   995→   995→              // If any segments failed CRC, attempt resync using indel hypotheses
   996→   996→              if (stripped.failedSegments.length > 0) {
   997→   997→                // Try resync at each failure point; if resync fails, the segment
   998→   998→                // data is still included but may be corrupted — Viterbi will handle it
   999→   999→                for (const failSeg of stripped.failedSegments) {
  1000→  1000→                  const failPos = failSeg * (markerCfg.segmentSize + markerCfg.markerSize);
  1001→  1001→                  const shift = resyncAfterIndel(convBytes, failPos, markerCfg);
  1002→  1002→                  if (shift !== 0) {
  1003→  1003→                    // Apply the shift by adjusting the convBytes alignment
  1004→  1004→                    // (For simplicity, we just use the stripped payload — the Viterbi
  1005→  1005→                    // decoder's trellis can handle the residual misalignment)
  1006→  1006→                  }
  1007→  1007→                }
  1008→  1008→              }
  1009→  1009→              convBytes = stripped.payload;
  1010→  1010→            } catch {
  1011→  1011→              // CRC marker stripping failed — use convBytes as-is
  1012→  1012→            }
  1013→  1013→          }
  1014→  1014→
  1015→  1015→          // Build per-bit LLRs for soft-decision Viterbi when Q-scores are available.
  1016→  1016→          // Q-scores are per-base in the DNA string; conv-encoded region uses
  1017→  1017→          // direct 2-bit mapping (4 bases per byte = 8 bits per 4 bases).
  1018→  1018→          // The conv region starts after address (4B = 16 bases).
  1019→  1019→          let convLLR: Float32Array | undefined;
  1020→  1020→          if (read.quality && read.quality.length > 0) {
  1021→  1021→            const convStartBase = layout.addressBytes * 4;
  1022→  1022→            const convNumBases = layout.convEncodedBytes * 4;
  1023→  1023→            const convEndBase = Math.min(convStartBase + convNumBases, read.quality.length);
  1024→  1024→            if (convEndBase > convStartBase) {
  1025→  1025→              const convQScores = read.quality.slice(convStartBase, convEndBase);
  1026→  1026→              convLLR = qScoresToBitLLRs(convQScores, 2);
  1027→  1027→            }
  1028→  1028→          }
  1029→  1029→
  1030→  1030→          // Conv Viterbi decode → original LDPC codeword (innerN bytes)
  1031→  1031→          const convResult = convInner.decode(convBytes, convLLR);
  1032→  1032→          const rsCodewordDecoded = convResult.decoded;
  1033→  1033→          // Rebuild innerBlock as [LDPC codeword + CRC]
  1034→  1034→          // CRC is the last 2 bytes of the original innerBlock
  1035→  1035→          const crcFromRead = innerBlock.slice(innerBlock.length - layout.crcBytes);
  1036→  1036→          const rebuilt = new Uint8Array(innerN + layout.crcBytes);
  1037→  1037→          rebuilt.set(rsCodewordDecoded, 0);
  1038→  1038→          rebuilt.set(crcFromRead, innerN);
  1039→  1039→          innerBlock = rebuilt;
  1040→  1040→        } catch {
  1041→  1041→          // Conv decode failed (e.g., too many errors) — skip this read
  1042→  1042→          continue;
  1043→  1043→        }
  1044→  1044→      }
  1045→  1045→
  1046→  1046→      // v62: For arithmetic-v2, the LDPC codeword does NOT include the address.
  1047→  1047→      //   innerBlock = [address(4)] + [LDPC codeword (payload + parity)]
  1048→  1048→      //   rsCodeword = innerBlock.slice(addressBytes, addressBytes + innerN)
  1049→  1049→      //   crcBytes = [] (no CRC-16; per-block CRC-8 already validated)
  1050→  1050→      const rsCodeword = useArithmeticV2
  1051→  1051→        ? innerBlock.slice(layout.addressBytes, layout.addressBytes + innerN)
  1052→  1052→        : innerBlock.slice(0, innerN);
  1053→  1053→      const crcBytes = useArithmeticV2
  1054→  1054→        ? new Uint8Array(0)  // no CRC-16 in arithmetic-v2
  1055→  1055→        : innerBlock.slice(innerN);
  1056→  1056→
  1057→  1057→      // Extract Q-scores for the inner block region (excluding primers).
  1058→  1058→      // read.sequence was already trimmed to inner (in clusterReads).
  1059→  1059→      // Q-scores are per-base in the DNA string.
  1060→  1060→      let qScoresForInner: Uint8Array | null = null;
  1061→  1061→      if (read.quality && read.quality.length > 0) {
  1062→  1062→        // The quality array aligns with read.sequence (the inner DNA).
  1063→  1063→        // We need Q-scores for the first expectedDnaLen bases.
  1064→  1064→        if (useGoldman) {
  1065→  1065→          // Goldman: inner block bytes = expectedDnaLen / 6.
  1066→  1066→          // Q-scores are per-base in the DNA string.
  1067→  1067→          // The LDPC codeword is the first innerN bytes = innerN * 6 bases.
  1068→  1068→          const ldpcBases = innerN * 6;
  1069→  1069→          qScoresForInner = read.quality.slice(0, Math.min(ldpcBases, read.quality.length));
  1070→  1070→        } else {
  1071→  1071→          // Direct/Constrained: inner block bytes = expectedDnaLen / 4.
  1072→  1072→          // LDPC codeword = first innerN bytes = innerN * 4 bases.
  1073→  1073→          const ldpcBases = innerN * 4;
  1074→  1074→          qScoresForInner = read.quality.slice(0, Math.min(ldpcBases, read.quality.length));
  1075→  1075→        }
  1076→  1076→      }
  1077→  1077→
  1078→  1078→      // For constrained/arithmetic mode, modify Q-scores to reflect erasures.
  1079→  1079→      // Erased bits get Q=0 (neutral LLR), known bits keep their Q-score.
  1080→  1080→      // This allows the BP decoder to correct erasures efficiently.
  1081→  1081→      // v57: also fire for useArithmetic — per-block CRC failures in
  1082→  1082→      // arithmetic mode produce byte-level erasures that LDPC must correct.
  1083→  1083→      const useErasures = useConstrained || useArithmetic;
  1084→  1084→      if (useErasures && constrainedErasures && qScoresForInner) {
  1085→  1085→        // Each base = 2 bits. Erasure info is per-bit.
  1086→  1086→        // We can't directly set per-bit Q-scores (they're per-base), so we
  1087→  1087→        // set the base Q-score to 0 if either bit is erased.
  1088→  1088→        for (let byteIdx = 0; byteIdx < innerN; byteIdx++) {
  1089→  1089→          for (let bitInByte = 0; bitInByte < 8; bitInByte += 2) {
  1090→  1090→            const bitIdx = byteIdx * 8 + bitInByte;
  1091→  1091→            if (constrainedErasures[bitIdx] || constrainedErasures[bitIdx + 1]) {
  1092→  1092→              const baseIdx = byteIdx * 4 + (bitInByte >> 1);
  1093→  1093→              if (baseIdx < qScoresForInner.length) {
  1094→  1094→                qScoresForInner[baseIdx] = 0; // Q=0 → neutral LLR
  1095→  1095→              }
  1096→  1096→            }
  1097→  1097→          }
  1098→  1098→        }
  1099→  1099→      } else if (useErasures && constrainedErasures) {
  1100→  1100→        // No Q-scores from read, but we still have erasures.
  1101→  1101→        // Build synthetic Q-scores: Q=0 for erased bits, Q=30 for known.
  1102→  1102→        const ldpcBases = innerN * 4;
  1103→  1103→        qScoresForInner = new Uint8Array(ldpcBases);
  1104→  1104→        for (let i = 0; i < ldpcBases; i++) qScoresForInner[i] = 30;
  1105→  1105→        for (let byteIdx = 0; byteIdx < innerN; byteIdx++) {
  1106→  1106→          for (let bitInByte = 0; bitInByte < 8; bitInByte += 2) {
  1107→  1107→            const bitIdx = byteIdx * 8 + bitInByte;
  1108→  1108→            if (constrainedErasures[bitIdx] || constrainedErasures[bitIdx + 1]) {
  1109→  1109→              const baseIdx = byteIdx * 4 + (bitInByte >> 1);
  1110→  1110→              if (baseIdx < qScoresForInner.length) {
  1111→  1111→                qScoresForInner[baseIdx] = 0;
  1112→  1112→              }
  1113→  1113→            }
  1114→  1114→          }
  1115→  1115→        }
  1116→  1116→      }
  1117→  1117→
  1118→  1118→      // Try inner code decode (corrects errors in this individual read)
  1119→  1119→      // For RS: guaranteed minimum distance, so CRC is redundant
  1120→  1120→      // For LDPC: may pick wrong codeword (duplicate columns), so CRC is REQUIRED
  1121→  1121→      // For LDPC with soft-info: OSD-2/BP flips least-reliable bits
  1122→  1122→      let decodedData: Uint8Array;
  1123→  1123→      let corrected: number;
  1124→  1124→      try {
  1125→  1125→        if (useLDPC && innerLdpc) {
  1126→  1126→          // v57: If we have bit-level erasures (from arithmetic CRC or constrained
  1127→  1127→          // mapping), use the dedicated erasure decoder (peeling + Gaussian elim
  1128→  1128→          // over GF(2)). This is FAR more reliable than BP for erasures — BP
  1129→  1129→          // often fails to converge when Q-scores are 0, while the peeling
  1130→  1130→          // decoder directly solves the linear system.
  1131→  1131→          const hasErasures = constrainedErasures && constrainedErasures.some((e) => e);
  1132→  1132→          if (hasErasures) {
  1133→  1133→            // Convert boolean[] erasure bitmap to bit positions
  1134→  1134→            const erasePos: number[] = [];
  1135→  1135→            for (let i = 0; i < constrainedErasures!.length && i < innerN * 8; i++) {
  1136→  1136→              if (constrainedErasures![i]) erasePos.push(i);
  1137→  1137→            }
  1138→  1138→            if (erasePos.length > 0 && erasePos.length < innerN * 8) {
  1139→  1139→              try {
  1140→  1140→                const r = innerLdpc.decodeWithErasures(rsCodeword, erasePos);
  1141→  1141→                decodedData = r.data;
  1142→  1142→                corrected = r.corrected;
  1143→  1143→              } catch {
  1144→  1144→                // Erasure decode failed — fall through to BP
  1145→  1145→                const r = innerLdpc.decodeBeliefPropagation(rsCodeword, qScoresForInner, useGoldman);
  1146→  1146→                decodedData = r.data;
  1147→  1147→                corrected = r.corrected;
  1148→  1148→              }
  1149→  1149→            } else {
  1150→  1150→              // Too many erasures or none — use BP
  1151→  1151→              const r = innerLdpc.decodeBeliefPropagation(rsCodeword, qScoresForInner, useGoldman);
  1152→  1152→              decodedData = r.data;
  1153→  1153→              corrected = r.corrected;
  1154→  1154→            }
  1155→  1155→          } else if ((useConstrained || useSrt || useArithmetic)) {
  1156→  1156→            // For constrained/SRT/arithmetic mode without erasures, use BP
  1157→  1157→            const r = innerLdpc.decodeBeliefPropagation(rsCodeword, qScoresForInner, useGoldman);
  1158→  1158→            decodedData = r.data;
  1159→  1159→            corrected = r.corrected;
  1160→  1160→          } else {
  1161→  1161→            // Standard mode: use configured decoder mode
  1162→  1162→            const decoderMode = ldpcDecoderMode;
  1163→  1163→            if (decoderMode === "bp") {
  1164→  1164→              const r = innerLdpc.decodeBeliefPropagation(rsCodeword, qScoresForInner, useGoldman);
  1165→  1165→              decodedData = r.data;
  1166→  1166→              corrected = r.corrected;
  1167→  1167→            } else if (decoderMode === "osd") {
  1168→  1168→              if (qScoresForInner && qScoresForInner.length > 0) {
  1169→  1169→                try {
  1170→  1170→                  const r = innerLdpc.decodeWithSoftInfo(rsCodeword, qScoresForInner, useGoldman);
  1171→  1171→                  decodedData = r.data;
  1172→  1172→                  corrected = r.corrected;
  1173→  1173→                } catch {
  1174→  1174→                  const r = innerLdpc.decode(rsCodeword);
  1175→  1175→                  decodedData = r.data;
  1176→  1176→                  corrected = r.corrected;
  1177→  1177→                }
  1178→  1178→              } else {
  1179→  1179→                const r = innerLdpc.decode(rsCodeword);
  1180→  1180→                decodedData = r.data;
  1181→  1181→                corrected = r.corrected;
  1182→  1182→              }
  1183→  1183→            } else if (decoderMode === "hard") {
  1184→  1184→              const r = innerLdpc.decode(rsCodeword);
  1185→  1185→              decodedData = r.data;
  1186→  1186→              corrected = r.corrected;

  1537→  1537→      }
  1538→  1538→    }
  1539→  1539→
  1540→  1540→    // STRATEGY 2.75: v58 Progressive MSA (Kalign-style) — handles indels.
  1541→  1541→    // v61: Skip for illumina (low-IDS) — too slow. Only run for nanopore/pacbio.
  1542→  1542→    // This is the right tool for Nanopore / PacBio / high-IDS channels where indels
  1543→  1543→    // dominate. The fast weighted consensus (STRATEGY 2) assumes no indels
  1544→  1544→    // and fails when >5% of reads have insertions/deletions. The HMM
  1545→  1545→    // soft-consensus (STRATEGY 3) also handles indels but is ~10× slower
  1546→  1546→    // and can fail to converge on long noisy reads. Progressive MSA is the
  1547→  1547→    // sweet spot: O(N²) in reads but each NW profile alignment is fast.
  1548→  1548→    // v59: Now supports conv inner code — runs conv Viterbi on the MSA
  1549→  1549→    // consensus before LDPC. This is critical for nanopore: per-read conv
  1550→  1550→    // Viterbi fails on noisy reads, but MSA consensus is clean enough for
  1551→  1551→    // conv Viterbi to succeed.
  1552→  1552→    // NOTE: For nanopore/pacbio, STRATEGY 2 already uses MSA as the primary
  1553→  1553→    // approach. This fallback is kept for the case where STRATEGY 2's MSA
  1554→  1554→    // fails CRC verification but a retry with different read selection might work.
  1555→  1555→    if (clusterReads.length >= 2 && useLDPC && innerLdpc && (channel === "nanopore" || channel === "pacbio")) {
  1556→  1556→      const msaResult = decodeWithProgressiveMSA(
  1557→  1557→        clusterReads, expectedDnaLen, innerLdpc,
  1558→  1558→        (dna: string) => dnaToBytes(dna), innerN,
  1559→  1559→        convInner ?? null,
  1560→  1560→        useConvInner ? {
  1561→  1561→          addressBytes: layout.addressBytes,
  1562→  1562→          convEncodedBytes: layout.convEncodedBytes,
  1563→  1563→          crcBytes: layout.crcBytes,
  1564→  1564→        } : null,
  1565→  1565→      );
  1566→  1566→      if (msaResult) {
  1567→  1567→        const reEncoded = innerLdpc.encode(msaResult.data);
  1568→  1568→        // v61: Verify CRC
  1569→  1569→        const firstRead = clusterReads[0];
  1570→  1570→        let firstReadDna = firstRead.sequence;
  1571→  1571→        if (firstReadDna.length > expectedDnaLen) firstReadDna = firstReadDna.slice(0, expectedDnaLen);
  1572→  1572→        else if (firstReadDna.length < expectedDnaLen) firstReadDna = firstReadDna + "A".repeat(expectedDnaLen - firstReadDna.length);
  1573→  1573→        let firstReadCrc: Uint8Array | null = null;
  1574→  1574→        try { firstReadCrc = dnaToBytes(firstReadDna).slice(innerN, innerN + 2); } catch {}
  1575→  1575→        const expectedCrc = crc16Bytes(reEncoded);
  1576→  1576→        const crcOk = firstReadCrc && expectedCrc[0] === firstReadCrc[0] && expectedCrc[1] === firstReadCrc[1];
  1577→  1577→        const whitenedAddr = reEncoded.slice(0, layout.addressBytes);
  1578→  1578→        const addr = unwhitenAddress(whitenedAddr);
  1579→  1579→        const decodedIndex = (addr[0] << 16) | (addr[1] << 8) | addr[2];
  1580→  1580→        if (crcOk && decodedIndex === oligoIdx) {
  1581→  1581→          let payload = reEncoded.slice(layout.addressBytes, layout.addressBytes + layout.payloadBytes);
  1582→  1582→          payloads.set(oligoIdx, payload);
  1583→  1583→          perOligo.push({
  1584→  1584→            index: oligoIdx, readCount: clusterReads.length, consensusLength: expectedDnaLen,
  1585→  1585→            crcPassed: true, innerRS: { corrected: msaResult.corrected, success: true },
  1586→  1586→            seed: 0, payloadBytes: payload, isParity: oligoIdx >= metadata.outerRS.k,
  1587→  1587→            strategy: 'msa',
  1588→  1588→          });
  1589→  1589→          continue;
  1590→  1590→        }
  1591→  1591→      }
  1592→  1592→    }
  1593→  1593→
  1594→  1594→    // STRATEGY 3: HMM-based soft-consensus (slower, handles indels).
  1595→  1595→    // Only try if fast consensus failed and we have enough reads.
  1596→  1596→    // v61: Skip for illumina (low-IDS) — too slow. Only run for nanopore/pacbio.
  1597→  1597→    if (clusterReads.length >= 3 && useLDPC && innerLdpc && (channel === "nanopore" || channel === "pacbio")) {
  1598→  1598→      const refRead = clusterReads[0];
  1599→  1599→      let refDna = refRead.sequence;
  1600→  1600→      if (refDna.length < expectedDnaLen) {
  1601→  1601→        refDna = refDna + "A".repeat(expectedDnaLen - refDna.length);
  1602→  1602→      } else if (refDna.length > expectedDnaLen) {
  1603→  1603→        refDna = refDna.slice(0, expectedDnaLen);
  1604→  1604→      }
  1605→  1605→
  1606→  1606→      try {
  1607→  1607→        const result = decodeWithSoftConsensus(clusterReads, refDna, innerLdpc, useGoldman);
  1608→  1608→        if (result) {
  1609→  1609→          const reEncoded = innerLdpc.encode(result.data);
  1610→  1610→          // v61: Verify CRC
  1611→  1611→          const firstRead = clusterReads[0];
  1612→  1612→          let firstReadDna = firstRead.sequence;
  1613→  1613→          if (firstReadDna.length > expectedDnaLen) firstReadDna = firstReadDna.slice(0, expectedDnaLen);
  1614→  1614→          else if (firstReadDna.length < expectedDnaLen) firstReadDna = firstReadDna + "A".repeat(expectedDnaLen - firstReadDna.length);
  1615→  1615→          let firstReadCrc: Uint8Array | null = null;
  1616→  1616→          try { firstReadCrc = dnaToBytes(firstReadDna).slice(innerN, innerN + 2); } catch {}
  1617→  1617→          const expectedCrc = crc16Bytes(reEncoded);
  1618→  1618→          const crcOk = firstReadCrc && expectedCrc[0] === firstReadCrc[0] && expectedCrc[1] === firstReadCrc[1];
  1619→  1619→          const whitenedAddr = reEncoded.slice(0, layout.addressBytes);
  1620→  1620→          const addr = unwhitenAddress(whitenedAddr);
  1621→  1621→          const decodedIndex = (addr[0] << 16) | (addr[1] << 8) | addr[2];
  1622→  1622→          if (crcOk && decodedIndex === oligoIdx) {
  1623→  1623→            let payload = reEncoded.slice(layout.addressBytes, layout.addressBytes + layout.payloadBytes);
  1624→  1624→            payloads.set(oligoIdx, payload);
  1625→  1625→            perOligo.push({
  1626→  1626→              index: oligoIdx, readCount: clusterReads.length, consensusLength: refDna.length,
  1627→  1627→              crcPassed: true, innerRS: { corrected: result.corrected, success: true },
  1628→  1628→              seed: 0, payloadBytes: payload, isParity: oligoIdx >= metadata.outerRS.k,
  1629→  1629→              strategy: 'soft_consensus',
  1630→  1630→            });
  1631→  1631→            continue;
  1632→  1632→          }
  1633→  1633→        }
  1634→  1634→      } catch {
  1635→  1635→        // HMM soft-consensus failed — fall through to OSD post-pass
  1636→  1636→      }
  1637→  1637→    }
  1638→  1638→
  1639→  1639→    // OSD-2 post-pass: try ordered statistics decoding before giving up.
  1640→  1640→    // This recovers many LDPC failures at the cost of ~2ms per oligo.
  1641→  1641→    // Only attempt for nanopore/pacbio channels where inner decode often fails
  1642→  1642→    // due to indels that the Viterbi preprocessor couldn't fully correct.
  1643→  1643→    if ((channel === "nanopore" || channel === "pacbio") && useLDPC && innerLdpc && clusterReads.length >= 1) {
  1644→  1644→      try {
  1645→  1645→        const { osdDecode } = await import('./osd-full');
  1646→  1646→        const { GF2Matrix } = await import('./osd');
  1647→  1647→
  1648→  1648→        // Build a consensus from cluster reads to use as the received word
  1649→  1649→        const refRead = clusterReads[0];
  1650→  1650→        let refDna = refRead.sequence;
  1651→  1651→        if (refDna.length < expectedDnaLen) {
  1652→  1652→          refDna = refDna + "A".repeat(expectedDnaLen - refDna.length);
  1653→  1653→        } else if (refDna.length > expectedDnaLen) {
  1654→  1654→          refDna = refDna.slice(0, expectedDnaLen);
  1655→  1655→        }
  1656→  1656→
  1657→  1657→        // Convert consensus DNA to bytes
  1658→  1658→        let rsCodewordForOsd: Uint8Array;
  1659→  1659→        try {
  1660→  1660→          rsCodewordForOsd = dnaToBytes(refDna).slice(0, innerN);
  1661→  1661→        } catch {
  1662→  1662→          // dnaToBytes failed — skip OSD
  1663→  1663→          throw new Error("dnaToBytes failed for OSD input");
  1664→  1664→        }
  1665→  1665→
  1666→  1666→        // Build LLRs from Q-scores if available, otherwise use uniform LLRs
  1667→  1667→        const nBits = innerN * 8;
  1668→  1668→        const llr = new Float32Array(nBits);
  1669→  1669→        if (refRead.quality && refRead.quality.length > 0) {
  1670→  1670→          // Use Q-scores to derive per-bit LLRs
  1671→  1671→          const qScores = refRead.quality.slice(0, Math.min(expectedDnaLen, refRead.quality.length));
  1672→  1672→          for (let baseIdx = 0; baseIdx < qScores.length && baseIdx * 2 < nBits; baseIdx++) {
  1673→  1673→            // Q-score to LLR: LLR = (Q - 33) * ln(10) / 10 * 2 (per-bit)
  1674→  1674→            const qVal = qScores[baseIdx];
  1675→  1675→            const llrMag = Math.max(0.1, ((qVal - 33) * 0.2303)); // 0.2303 ≈ ln(10)/10
  1676→  1676→            // Two bits per base (direct mapping)
  1677→  1677→            const byteIdx0 = Math.floor(baseIdx / 4);
  1678→  1678→            const bitIdx0 = (baseIdx % 4) * 2;
  1679→  1679→            if (byteIdx0 < innerN) {
  1680→  1680→              const byteVal = rsCodewordForOsd[byteIdx0];
  1681→  1681→              // bit 0
  1682→  1682→              const bit0 = (byteVal >> (7 - bitIdx0)) & 1;
  1683→  1683→              llr[baseIdx * 2] = bit0 === 0 ? llrMag : -llrMag;
  1684→  1684→              // bit 1
  1685→  1685→              const bit1 = (byteVal >> (7 - bitIdx0 - 1)) & 1;
  1686→  1686→              llr[baseIdx * 2 + 1] = bit1 === 0 ? llrMag : -llrMag;
  1687→  1687→            }
  1688→  1688→          }
  1689→  1689→        } else {
  1690→  1690→          // Uniform LLRs (hard decision) — magnitude 1.0
  1691→  1691→          for (let i = 0; i < nBits; i++) {
  1692→  1692→            const byteIdx = Math.floor(i / 8);
  1693→  1693→            const bitPos = 7 - (i % 8);
  1694→  1694→            const bit = (rsCodewordForOsd[byteIdx] >> bitPos) & 1;
  1695→  1695→            llr[i] = bit === 0 ? 1.0 : -1.0;
  1696→  1696→          }
  1697→  1697→        }
  1698→  1698→
  1699→  1699→        // Get parity-check matrix from LDPC code
  1700→  1700→        const H = innerLdpc.parityCheckMatrix;
  1701→  1701→
  1702→  1702→        // CRC + address check function
  1703→  1703→        const crcAndAddrCheck = (codewordBits: Uint8Array): boolean => {
  1704→  1704→          // Convert bit array to byte array
  1705→  1705→          const codewordBytes = new Uint8Array(innerN);
  1706→  1706→          for (let i = 0; i < innerN; i++) {
  1707→  1707→            let byteVal = 0;
  1708→  1708→            for (let b = 0; b < 8; b++) {
  1709→  1709→              byteVal |= (codewordBits[i * 8 + b] << (7 - b));
  1710→  1710→            }
  1711→  1711→            codewordBytes[i] = byteVal;
  1712→  1712→          }
  1713→  1713→          // Re-encode to get canonical form
  1714→  1714→          const reEncoded = innerLdpc!.encode(
  1715→  1715→            // Extract info bits
  1716→  1716→            codewordBytes.slice(0, innerLdpc!.k),
  1717→  1717→          );
  1718→  1718→          // Verify CRC
  1719→  1719→          const expectedCrc = crc16Bytes(reEncoded);
  1720→  1720→          // Get CRC from the first read
  1721→  1721→          const firstRead = clusterReads[0];
  1722→  1722→          let firstReadDna2 = firstRead.sequence;
  1723→  1723→          if (firstReadDna2.length > expectedDnaLen) firstReadDna2 = firstReadDna2.slice(0, expectedDnaLen);
  1724→  1724→          else if (firstReadDna2.length < expectedDnaLen) firstReadDna2 = firstReadDna2 + "A".repeat(expectedDnaLen - firstReadDna2.length);
  1725→  1725→          let firstReadCrc2: Uint8Array | null = null;
  1726→  1726→          try { firstReadCrc2 = dnaToBytes(firstReadDna2).slice(innerN, innerN + 2); } catch {}
  1727→  1727→          if (!firstReadCrc2 || expectedCrc[0] !== firstReadCrc2[0] || expectedCrc[1] !== firstReadCrc2[1]) return false;
  1728→  1728→          // Verify address
  1729→  1729→          const whitenedAddr = reEncoded.slice(0, layout.addressBytes);
  1730→  1730→          const addr = unwhitenAddress(whitenedAddr);
  1731→  1731→          const decodedIndex = (addr[0] << 16) | (addr[1] << 8) | addr[2];
  1732→  1732→          return decodedIndex === oligoIdx;
  1733→  1733→        };
  1734→  1734→
  1735→  1735→        // Run OSD-2 (maxOrder=2 is a good tradeoff: O(k²) candidates, ~2ms)
  1736→  1736→        const osdResult = osdDecode(llr, H, crcAndAddrCheck, {
  1737→  1737→          maxOrder: 2,
  1738→  1738→          k: innerK * 8,
  1739→  1739→        });
  1740→  1740→
  1741→  1741→        if (osdResult.codeword) {
  1742→  1742→          // OSD succeeded — extract payload from the decoded codeword
  1743→  1743→          const codewordBits = osdResult.codeword;
  1744→  1744→          const codewordBytes = new Uint8Array(innerN);
  1745→  1745→          for (let i = 0; i < innerN; i++) {
  1746→  1746→            let byteVal = 0;
  1747→  1747→            for (let b = 0; b < 8; b++) {
  1748→  1748→              byteVal |= (codewordBits[i * 8 + b] << (7 - b));
  1749→  1749→            }
  1750→  1750→            codewordBytes[i] = byteVal;
  1751→  1751→          }
  1752→  1752→          // Re-encode to get canonical form
  1753→  1753→          const reEncoded = innerLdpc.encode(codewordBytes.slice(0, innerK));
  1754→  1754→          const seed = 0; // OSD doesn't recover seed
  1755→  1755→          let payload = reEncoded.slice(layout.addressBytes, layout.addressBytes + layout.payloadBytes);
  1756→  1756→          payloads.set(oligoIdx, payload);
  1757→  1757→          perOligo.push({
  1758→  1758→            index: oligoIdx, readCount: clusterReads.length, consensusLength: refDna.length,
  1759→  1759→            crcPassed: true, innerRS: { corrected: osdResult.candidatesTried, success: true },
  1760→  1760→            seed, payloadBytes: payload, isParity: oligoIdx >= metadata.outerRS.k,
  1761→  1761→            strategy: 'osd_post_pass',
  1762→  1762→          });
  1763→  1763→          continue;
  1764→  1764→        }
  1765→  1765→      } catch {
  1766→  1766→        // OSD not available or failed — fall through to erasure
  1767→  1767→      }
  1768→  1768→    }
  1769→  1769→
  1770→  1770→    // STRATEGY 4: Mark as erasure for outer RS to recover.
  1771→  1771→    erasedIndices.push(oligoIdx);
  1772→  1772→    oligosFailedInnerRS++;
  1773→  1773→    perOligo.push({
  1774→  1774→      index: oligoIdx, readCount: clusterReads.length, consensusLength: 0,
  1775→  1775→      crcPassed: false, innerRS: { corrected: 0, success: false }, seed: 0,
  1776→  1776→      payloadBytes: new Uint8Array(layout.payloadBytes),
  1777→  1777→      isParity: oligoIdx >= metadata.outerRS.k,
  1778→  1778→      strategy: 'erasure',
  1779→  1779→    });
  1780→  1780→  }
  1781→  1781→
  1782→  1782→  // 3) Outer RS: recover missing oligos via erasure decoding.
  1783→  1783→  // With per-read decode, there are NO silent consensus errors — every
  1784→  1784→  // payload in the map was individually validated by RS+CRC. So the outer
  1785→  1785→  // RS only needs to recover genuinely missing oligos (erasures).
  1786→  1786→  let oligosFailedOuterRS = 0;
  1787→  1787→  const recoveredPayloads = new Map<number, Uint8Array>();
  1788→  1788→
  1789→  1789→  if (!useOuterRS) {
  1790→  1790→    for (const [idx, payload] of payloads) {
  1791→  1791→      recoveredPayloads.set(idx, payload);
  1792→  1792→    }
  1793→  1793→  } else if (useGF216 && outerRs216) {
  1794→  1794→    const numPairs = Math.floor(layout.payloadBytes / 2);
  1795→  1795→    let outerSuccessCount = 0, outerFailCount = 0, outerSkipCount = 0;
  1796→  1796→    for (let pairIdx = 0; pairIdx < numPairs; pairIdx++) {
  1797→  1797→      const j0 = pairIdx * 2;
  1798→  1798→      const j1 = pairIdx * 2 + 1;
  1799→  1799→      const codeword = new Uint16Array(metadata.outerRS.n);
  1800→  1800→      const erased: number[] = [];
  1801→  1801→      for (let i = 0; i < metadata.outerRS.n; i++) {
  1802→  1802→        const p = payloads.get(i);
  1803→  1803→        if (p) {
  1804→  1804→          codeword[i] = (p[j0] << 8) | p[j1];
  1805→  1805→        } else {
  1806→  1806→          codeword[i] = 0;
  1807→  1807→          if (erasedIndices.includes(i)) erased.push(i);
  1808→  1808→        }
  1809→  1809→      }
  1810→  1810→      if (erased.length === 0) {
  1811→  1811→        outerSkipCount++;
  1812→  1812→        for (let i = 0; i < metadata.outerRS.k; i++) {
  1813→  1813→          if (!recoveredPayloads.has(i)) {
  1814→  1814→            recoveredPayloads.set(i, new Uint8Array(layout.payloadBytes));
  1815→  1815→          }
  1816→  1816→          recoveredPayloads.get(i)![j0] = (codeword[i] >> 8) & 0xff;
  1817→  1817→          recoveredPayloads.get(i)![j1] = codeword[i] & 0xff;
  1818→  1818→        }
  1819→  1819→        continue;
  1820→  1820→      }
  1821→  1821→      if (erased.length > metadata.outerRS.n - metadata.outerRS.k) {
  1822→  1822→        oligosFailedOuterRS++;
  1823→  1823→        outerFailCount++;
  1824→  1824→        continue;
  1825→  1825→      }
  1826→  1826→      try {
  1827→  1827→        const r = outerRs216.decodeWithErasures(codeword, erased);
  1828→  1828→        outerSuccessCount++;
  1829→  1829→        for (let i = 0; i < metadata.outerRS.k; i++) {
  1830→  1830→          if (!recoveredPayloads.has(i)) {
  1831→  1831→            recoveredPayloads.set(i, new Uint8Array(layout.payloadBytes));
  1832→  1832→          }
  1833→  1833→          recoveredPayloads.get(i)![j0] = (r.data[i] >> 8) & 0xff;
  1834→  1834→          recoveredPayloads.get(i)![j1] = r.data[i] & 0xff;
  1835→  1835→        }
  1836→  1836→      } catch {
  1837→  1837→        oligosFailedOuterRS++;
  1838→  1838→        outerFailCount++;
  1839→  1839→      }
  1840→  1840→    }
  1841→  1841→    if (process.env.HELIX_DEBUG) {
  1842→  1842→      console.error(`[LDPC debug] outer RS: ${outerSuccessCount} success, ${outerFailCount} fail, ${outerSkipCount} skip (erasedIndices=${erasedIndices.length}, erasedCount=${erasedIndices.length})`);
  1843→  1843→    }
  1844→  1844→    // Handle odd byte
  1845→  1845→    if (layout.payloadBytes % 2 === 1) {
  1846→  1846→      const j = layout.payloadBytes - 1;
  1847→  1847→      // Build codeword for the odd byte using GF(2^16) (same as even bytes)
  1848→  1848→      const codewordOdd = new Uint16Array(metadata.outerRS.n);
  1849→  1849→      const erasedOdd: number[] = [];
  1850→  1850→      for (let i = 0; i < metadata.outerRS.n; i++) {
  1851→  1851→        const p = payloads.get(i);
  1852→  1852→        if (p) {
  1853→  1853→          codewordOdd[i] = p[j] & 0xff; // high byte = 0
  1854→  1854→        } else {
  1855→  1855→          codewordOdd[i] = 0;
  1856→  1856→          if (erasedIndices.includes(i)) erasedOdd.push(i);
  1857→  1857→        }
  1858→  1858→      }
  1859→  1859→      if (process.env.HELIX_DEBUG) {
  1860→  1860→        console.error(`[LDPC debug] odd byte (j=${j}): ${erasedOdd.length} erasures, codeword len=${codewordOdd.length}`);
  1861→  1861→      }
  1862→  1862→      if (erasedOdd.length === 0) {
  1863→  1863→        for (let i = 0; i < metadata.outerRS.k; i++) {
  1864→  1864→          if (!recoveredPayloads.has(i)) {
  1865→  1865→            recoveredPayloads.set(i, new Uint8Array(layout.payloadBytes));
  1866→  1866→          }
  1867→  1867→          recoveredPayloads.get(i)![j] = codewordOdd[i] & 0xff;
  1868→  1868→        }
  1869→  1869→      } else if (erasedOdd.length <= metadata.outerRS.n - metadata.outerRS.k && outerRs216) {
  1870→  1870→        try {
  1871→  1871→          const r = outerRs216.decodeWithErasures(codewordOdd, erasedOdd);
  1872→  1872→          if (process.env.HELIX_DEBUG) {
  1873→  1873→            console.error(`[LDPC debug] odd byte recovered: ${r.data.length} symbols`);
  1874→  1874→          }
  1875→  1875→          for (let i = 0; i < metadata.outerRS.k; i++) {
  1876→  1876→            if (!recoveredPayloads.has(i)) {
  1877→  1877→              recoveredPayloads.set(i, new Uint8Array(layout.payloadBytes));
  1878→  1878→            }
  1879→  1879→            recoveredPayloads.get(i)![j] = r.data[i] & 0xff;
