   103→   103→/** Configuration for the codec. */
   104→   104→export interface CodecConfig {
   105→   105→  /** Total oligo length in nucleotides (including primers). Can be overridden per-channel (e.g., shorter for nanopore to keep indel count within Viterbi maxDrift). */
   106→   106→  oligoLength: number;
   107→   107→  /** Length of each primer in nucleotides. */
   108→   108→  primerLength: number;
   109→   109→  /** Forward primer sequence (auto-generated if not set). */
   110→   110→  forwardPrimer?: string;
   111→   111→  /** Reverse primer sequence (auto-generated if not set). */
   112→   112→  reversePrimer?: string;
   113→   113→  /**
   114→   114→   * Inner code type:
   115→   115→   *   - "rs"   — Reed-Solomon over GF(2^8) (legacy, guaranteed minimum distance)
   116→   116→   *   - "ldpc" — PEG-constructed LDPC over GF(2) (higher density, CRC-gated)
   117→   117→   *
   118→   118→   * Default: "ldpc" (since v21.0). LDPC gives ~15% higher density (1.50 vs 1.30
   119→   119→   * bits/nt payload-only) at the cost of a small per-read failure rate (~3%),
   120→   120→   * which is recovered by the outer RS at sufficient coverage.
   121→   121→   */
   122→   122→  innerCode?: "rs" | "ldpc";
   123→   123→  /**
   124→   124→   * LDPC decoder mode:
   125→   125→   *   - "hard"       — Syndrome lookup + bit-flipping (fast, lower correction)
   126→   126→   *   - "osd"        — OSD-2 soft-decision (medium, better correction)
   127→   127→   *   - "bp"         — Belief propagation / sum-product (slow, best correction)
   128→   128→   *   - "auto"       — Try hard → bp fallback (default since v23.0)
   129→   129→   *
   130→   130→   * BP achieves 60x better correction than hard-decision at 3-bit errors.
   131→   131→   * Auto mode tries hard-decision first (fast path), falls back to BP on failure.
   132→   132→   */
   133→   133→  ldpcDecoder?: "hard" | "osd" | "bp" | "auto";
   134→   134→  /**
   135→   135→   * DNA mapping mode:
   136→   136→   *   - "direct"     — 2-bit per base (00=A, 01=C, 10=G, 11=T). Density 2.0 bits/nt.
   137→   137→   *                    Requires constraint screening (GC + homopolymer) with seed retries.
   138→   138→   *   - "goldman"    — Goldman rotational mapping (3-base codebook per prev base).
   139→   139→   *                    GUARANTEES no homopolymers (max run = 1). No screening needed.
   140→   140→   *   - "constrained" — 2-bit mapping with sliding-window homopolymer avoidance.
   141→   141→   *                     Density 2.0 bits/nt (same as direct). Homopolymer ≤ 3 guaranteed.
   142→   142→   *                     No screening needed. GC ~50% on average.
   143→   143→   *                     (Inspired by Yi Ding 2024 SRT approach.)
   144→   144→   *
   145→   145→   * Default: "constrained" (since v23.0). Best of both worlds: full 2.0 bits/nt
   146→   146→   * density AND homopolymer-free without screening.
   147→   147→   */
   148→   148→  mappingMode?: "direct" | "goldman" | "constrained" | "srt" | "arithmetic";
   149→   149→  /**
   150→   150→   * Goldman trit packing mode (only used when mappingMode === "goldman"):
   151→   151→   *   - "fast"  — 1 byte → 6 trits (3^6=729>256). Density 1.333 bits/nt. Simple.
   152→   152→   *   - "dense" — 5 bytes → 26 trits (3^26>2^40). Density 1.538 bits/nt. 15% higher density.
   153→   153→   *
   154→   154→   * Default: "dense" (since v23.0). The 15% density improvement is significant:
   155→   155→   *   - fast:  0.674 bits/nt total oligo (200nt, 18B payload)
   156→   156→   *   - dense: 0.776 bits/nt total oligo (200nt, 21B payload)
   157→   157→   *
   158→   158→   * Dense mode uses BigInt for the 40-bit base-3 expansion but is still fast
   159→   159→   * (~1M bytes/sec in benchmarks).
   160→   160→   */
   161→   161→  goldmanMode?: "fast" | "dense";
   162→   162→  /** Inner RS/LDPC parity bytes per oligo (e.g., 8 means RS(k+8, k)). */
   163→   163→  innerParityBytes: number;
   164→   164→  /** Outer RS parity oligos (e.g., 0.2 * dataOligos for 20% outer parity).
   165→   165→   *  Can be overridden per-channel via presets (e.g., nanopore uses 0.4, real-2024 uses 0.5). */
   166→   166→  outerParityRatio: number;
   167→   167→  /** Constraint settings. */
   168→   168→  constraints: {
   169→   169→    gcMin: number;
   170→   170→    gcMax: number;
   171→   171→    maxHomopolymer: number;
   172→   172→  };
   173→   173→  /** Whether to apply DEFLATE compression before encoding. */
   174→   174→  compress: boolean;
   175→   175→  /** Maximum number of seed-retries per oligo during constraint screening. */
   176→   176→  maxRetries: number;
   177→   177→  /**
   178→   178→   * Interleave depth for burst error spreading (Kim 2024).
   179→   179→   * 0 = no interleaving (default).
   180→   180→   * N = group N oligos, interleave their LDPC codewords so a burst error
   181→   181→   * in one oligo is spread across N codewords (1 error each, easily corrected).
   182→   182→   * Recommended: 4 for typical Illumina, 8 for nanopore (higher indel rates).
   183→   183→   */
   184→   184→  interleaveDepth?: number;
   185→   185→  /**
   186→   186→   * Optional encryption password. If set, data is encrypted with
   187→   187→   * XChaCha20-Poly1305 before compression and encoding.
   188→   188→   * Uses @noble/ciphers for authenticated encryption.
   189→   189→   */
   190→   190→  encryptPassword?: string;
   191→   191→
   192→   192→  /**
   193→   193→   * Sequencing channel profile (Helix v51+, Ultimate Phase 3).
   194→   194→   *
   195→   195→   * Controls which preprocessor runs before the LDPC inner decoder:
   196→   196→   *   - "illumina" (default) — Substitution-dominant. No Viterbi preprocess.
   197→   197→   *                            LDPC alone corrects substitutions efficiently.
   198→   198→   *   - "nanopore"           — Indel-dominant (5-15% total IDS). Viterbi
   199→   199→   *                            convolutional preprocessor runs BEFORE LDPC to
   200→   200→   *                            correct indels (HEDGES-style). Then LDPC cleans
   201→   201→   *                            up residual substitutions. Then outer RS.
   202→   202→   *
   203→   203→   * Pipeline per channel:
   204→   204→   *   illumina: Reads → cluster → LDPC → outer RS
   205→   205→   *   nanopore: Reads → cluster → Viterbi(indel-correct) → LDPC → outer RS
   206→   206→   *
   207→   207→   * Default: "illumina" (matches prior behavior; no perf regression).
   208→   208→   */
   209→   209→  channel?: "illumina" | "nanopore" | "pacbio";
   210→   210→
   211→   211→  /**
   212→   212→   * Low-coverage trigger threshold (Helix v51+, Ultimate Phase 2).
   213→   213→   *
   214→   214→   * If the average cluster size (reads per oligo) is below this threshold,
   215→   215→   * decode switches to the Profile-HMM + log-product fusion path
   216→   216→   * (profileHmm3.ts) instead of majority-vote consensus. This dramatically
   217→   217→   * improves recovery at 2-3× coverage where naive consensus fails.
   218→   218→   *
   219→   219→   * Default: 5 (matches Mahoraga / Yi Ding 2024 trigger).
   220→   220→   * Set to 0 to disable the low-coverage path entirely (always use consensus).
   221→   221→   */
   222→   222→  lowCoverageTrigger?: number;
   223→   223→
   224→   224→  /**
   225→   225→   * HEDGES-style convolutional inner code (Helix v52+, Nanopore full support).
   226→   226→   *
   227→   227→   * When enabled, a rate-1/2 convolutional code (memory=2, K=3, generators
   228→   228→   * [7,5]) is applied AFTER the LDPC inner code and BEFORE DNA mapping.
   229→   229→   * The convolutional code is decoded first (Viterbi MLSE) at decode time,
   230→   230→   * BEFORE the LDPC decoder. This is the HEDGES arrangement:
   231→   231→   *
   232→   232→   *   ENCODE:  data → LDPC outer → conv inner → DNA mapping
   233→   233→   *   DECODE:  DNA → conv Viterbi → LDPC outer → data
   234→   234→   *
   235→   235→   * The convolutional code provides indel tolerance via trellis-based
   236→   236→   * maximum-likelihood sequence estimation. Combined with the Profile-HMM
   237→   237→   * preprocessor (viterbi-preprocess.ts), this enables true 9% IDS recovery
   238→   238→   * on Nanopore channels (HEDGES Press 2020 / Völkel 2025 PMC11755093).
   239→   239→   *
   240→   240→   * Density cost: rate-1/2 halves the LDPC codeword capacity. Effective
   241→   241→   * payload bytes per oligo is roughly halved (e.g., 54 → 22 at 300nt/20nt
   242→   242→   * primer). Net density drops from ~1.76 bits/nt to ~0.88 bits/nt —
   243→   243→   * a deliberate trade for true Nanopore support.
   244→   244→   *
   245→   245→   * Default: false (preserves v51 behavior; Illumina does not need it).
   246→   246→   * Set to true automatically by the ULTIMATE_NANOPORE_V52 preset.
   247→   247→   */
   248→   248→  useConvolutionalInner?: boolean;
   249→   249→
   250→   250→  /**
   251→   251→   * Block size (in nucleotides) for arithmetic DNA mapping mode.
   252→   252→   * Each block encodes independently with its own CRC-8 sync marker.
   253→   253→   * Larger blocks give higher density (less CRC overhead per block) but
   254→   254→   * larger error confinement (one corrupted block loses more data).
   255→   255→   *
   256→   256→   * Default: 80 (good balance: 19 bytes/block, 5.3% CRC overhead).
   257→   257→   */
   258→   258→  arithmeticBlockSize?: number;
   259→   259→
   260→   260→  /**
   261→   261→   * v3.0: Address derivation mode.
   262→   262→   *   - 'sequential'    — legacy 3-byte index + 1-byte seed (default, backward-compatible)
   263→   263→   *   - 'content-derived' — BLAKE3(payload + archiveSalt) → 4-byte address (Babel-USB pattern)
   264→   264→   *   - 'hierarchical'  — content-derived + split into pool/well/oligo for physical synthesis layout
   265→   265→   *
   266→   266→   * Content-derived addressing provides:
   267→   267→   *   1. Automatic deduplication (same payload → same address)
   268→   268→   *   2. Self-verification (address is a commitment to payload)
   269→   269→   *   3. Physical layout mapping (hierarchical mode maps to Twist pools/wells)
   270→   270→   *
   271→   271→   * Default: 'sequential' (preserves backward compatibility).
   272→   272→   */
   273→   273→  addressMode?: 'sequential' | 'content-derived' | 'hierarchical';
   274→   274→
   275→   275→  /**
   276→   276→   * v3.0: Archive salt for content-derived addressing (32 bytes).
   277→   277→   * Used as BLAKE3 key for domain separation (prevents cross-archive collisions).
   278→   278→   * Only used when addressMode !== 'sequential'.
   279→   279→   * If not provided, a random salt is generated at encode time and stored in metadata.
   280→   280→   */
   281→   281→  archiveSalt?: Uint8Array;
   282→   282→
   283→   283→  /**
   284→   284→   * v3.0: Enable recipe-based generation for structured/redundant data.
   285→   285→   * When true, the encoder detects patterns (all-zeros, repeating, low-entropy)
   286→   286→   * and stores OligoRecipe objects instead of raw bytes.
   287→   287→   * Can reduce storage by 10-1000× for sparse/repetitive files.
   288→   288→   *
   289→   289→   * Default: false (preserves backward compatibility).
   290→   290→   */
   291→   291→  useRecipeGeneration?: boolean;
   292→   292→}

   399→   399→/**
   400→   400→ * v3.0: Resolve a partial config to a full CodecConfig by applying channel presets.
   401→   401→ * If cfg.channel is set but outerParityRatio is default (0.1), auto-upgrade
   402→   402→ * the parity for nanopore/pacbio channels.
   403→   403→ */
   404→   404→export function resolveConfig(cfg: Partial<CodecConfig>): CodecConfig {
   405→   405→  const channel = cfg.channel ?? 'illumina';
   406→   406→  let base: CodecConfig;
   407→   407→  switch (channel) {
   408→   408→    case 'nanopore':
   409→   409→      base = NANOPORE_CONFIG;
   410→   410→      break;
   411→   411→    case 'pacbio':
   412→   412→      base = PACBIO_CONFIG;
   413→   413→      break;
   414→   414→    default:
   415→   415→      base = DEFAULT_CONFIG;
   416→   416→  }
   417→   417→  return {
   418→   418→    ...base,
   419→   419→    ...cfg,
   420→   420→    constraints: {
   421→   421→      ...base.constraints,
   422→   422→      ...(cfg.constraints ?? {}),
   423→   423→    },
   424→   424→  };
   425→   425→}
   426→   426→
   427→   427→/**
   428→   428→ * Compute the byte layout of an oligo given the config.
   429→   429→ *
   430→   430→ * Total oligo = primerLen + primerLen + (inner nt)
   431→   431→ * inner nt = innerBytes * (mappingMode == "goldman" ? 6 : 4)
   432→   432→ * inner bytes = addressBytes + payloadBytes + innerParityBytes + crcBytes
   433→   433→ *
   434→   434→ * We use:
   435→   435→ *   - addressBytes = 4 (3 bytes index + 1 byte seed; supports up to 16M oligos)
   436→   436→ *   - crcBytes = 2 (CRC-16)
   437→   437→ *   - innerParityBytes = config.innerParityBytes (default 4 for LDPC, 8 for RS)
   438→   438→ *   - payloadBytes = (innerNt / ntPerByte) - address - parity - crc
   439→   439→ *
   440→   440→ * IMPORTANT: payloadBytes is forced to be EVEN for GF(2^16) outer RS compatibility.
   441→   441→ */
   442→   442→export function computeLayout(cfg: CodecConfig): OligoLayout {
   443→   443→  const innerNt = cfg.oligoLength - 2 * cfg.primerLength;
   444→   444→  if (innerNt <= 0) {
   445→   445→    throw new Error(`oligoLength ${cfg.oligoLength} too small for primers`);
   446→   446→  }
   447→   447→  // Determine nt per byte based on mapping mode
   448→   448→  //   direct/constrained: 4 nt/byte (2-bit mapping)
   449→   449→  //   goldman fast: 6 nt/byte (1 byte → 6 trits → 6 nt)
   450→   450→  //   goldman dense: 5.2 nt/byte (5 bytes → 26 trits → 26 nt) — must be divisible by 26
   451→   451→  let ntPerByte: number;
   452→   452→  let modUnit: number; // innerNt must be divisible by this
   453→   453→  if (cfg.mappingMode === "goldman") {
   454→   454→    if (cfg.goldmanMode === "dense") {
   455→   455→      ntPerByte = 26 / 5; // 5.2
   456→   456→      modUnit = 26;
   457→   457→    } else {
   458→   458→      ntPerByte = 6;
   459→   459→      modUnit = 6;
   460→   460→    }
   461→   461→  } else {
   462→   462→    // direct and constrained: both use 2-bit mapping (4 nt/byte)
   463→   463→    ntPerByte = 4;
   464→   464→    modUnit = 4;
   465→   465→  }
   466→   466→
   467→   467→  if (innerNt % modUnit !== 0) {
   468→   468→    throw new Error(
   469→   469→      `inner nt length ${innerNt} must be divisible by ${modUnit} (got oligoLength=${cfg.oligoLength}, primerLen=${cfg.primerLength}, mappingMode=${cfg.mappingMode ?? "direct"}, goldmanMode=${cfg.goldmanMode ?? "fast"})`,
   470→   470→    );
   471→   471→  }
   472→   472→
   473→   473→  // totalInnerBytes = innerNt / ntPerByte
   474→   474→  // For dense mode: innerNt / (26/5) = innerNt * 5 / 26
   475→   475→  const totalInnerBytes = cfg.mappingMode === "goldman" && cfg.goldmanMode === "dense"
   476→   476→    ? (innerNt * 5) / 26  // innerNt is divisible by 26, so this is integer
   477→   477→    : innerNt / ntPerByte;
   478→   478→
   479→   479→  const addressBytes = 4;
   480→   480→  const crcBytes = 2;
   481→   481→  const innerParityBytes = cfg.innerParityBytes;
   482→   482→  let payloadBytes = totalInnerBytes - addressBytes - innerParityBytes - crcBytes;
   483→   483→  if (payloadBytes <= 0) {
   484→   484→    throw new Error(
   485→   485→      `oligoLength ${cfg.oligoLength} too small: payload would be ${payloadBytes} bytes`,
   486→   486→    );
   487→   487→  }
   488→   488→  // Force payloadBytes to be EVEN for GF(2^16) outer RS compatibility.
   489→   489→  if (payloadBytes % 2 === 1) {
   490→   490→    payloadBytes -= 1;
   491→   491→  }
   492→   492→  // For dense Goldman, also ensure payloadBytes is a multiple of 5
   493→   493→  // (so the trit stream aligns with byte boundaries)
   494→   494→  if (cfg.mappingMode === "goldman" && cfg.goldmanMode === "dense") {
   495→   495→    // We need the INNER BLOCK (address + payload + parity + crc) to be a multiple of 5
   496→   496→    // for dense mode alignment. The CRC is outside the LDPC codeword, so we need
   497→   497→    // address + payload + parity = totalInnerBytes - crcBytes to be a multiple of 5.
   498→   498→    // Since totalInnerBytes = innerNt * 5 / 26, and innerNt is divisible by 26,
   499→   499→    // totalInnerBytes is always a multiple of 5. So address + payload + parity + crc
   500→   500→    // is a multiple of 5. But crc=2, so address + payload + parity = 5k - 2.
   501→   501→    // The LDPC codeword = address + payload + parity = 5k - 2 bytes.
   502→   502→    // For dense mode, we need this to be a multiple of 5.
   503→   503→    // 5k - 2 ≡ 0 (mod 5) → -2 ≡ 0 (mod 5) → never true!
   504→   504→    // So dense mode doesn't align perfectly with the CRC layout.
   505→   505→    // Solution: don't force alignment — the trit stream just pads the last partial chunk.
   506→   506→    // The decoder reads back the exact number of bytes and ignores extra trits.
   507→   507→  }
   508→   508→  // v58: Arithmetic mode capacity check — MUST match the encoder's actual
   509→   509→  // capacity formula (markov-arithmetic.ts bytesToArithmeticDnaCrc):
   510→   510→  //   bytesPerBlockTotal = max(2, floor(blockSize * 1.95 / 8))   [1.95 bits/nt]
   511→   511→  //   bytesPerBlockData  = bytesPerBlockTotal - 1                 [1 byte CRC-8]
   512→   512→  //   numBlocks           = floor(innerDnaLen / blockSize)
   513→   513→  //   dataCapacity        = numBlocks * bytesPerBlockData
   514→   514→  //
   515→   515→  // v57 bug: computeLayout used 2.0 bits/nt (bs/4) and 4 bytes overhead per
   516→   516→  // block. This UNDERESTIMATED capacity for small blocks (where the 1.95 vs
   517→   517→  // 2.0 difference matters less than the 1 vs 4 byte overhead). The encoder
   518→   518→  // would then produce MORE bytes than computeLayout expected, causing the
   519→   519→  // decoder to read garbage → hash FAIL.
   520→   520→  if (cfg.mappingMode === "arithmetic") {
   521→   521→    const innerDnaLenArith = totalInnerBytes * 4;
   522→   522→    const ARITH_CAPACITY_RATE_LAYOUT = 1.95;
   523→   523→    const defaultBlockSize = cfg.arithmeticBlockSize ?? Math.floor(innerDnaLenArith / 2);
   524→   524→    // Try multiple block sizes to find the best fit (v58: 1 byte CRC overhead)
   525→   525→    let bestCap = 0;
   526→   526→    for (const bs of [
   527→   527→      defaultBlockSize,
   528→   528→      innerDnaLenArith, // v58: 1 big block — max capacity
   529→   529→      Math.floor(innerDnaLenArith / 3),
   530→   530→      Math.floor(innerDnaLenArith / 4),
   531→   531→      Math.floor(innerDnaLenArith / 5),
   532→   532→      Math.floor(innerDnaLenArith / 8),
   533→   533→      Math.floor(innerDnaLenArith / 10),
   534→   534→      Math.floor(innerDnaLenArith / 14),
   535→   535→      Math.floor(innerDnaLenArith / 20),
   536→   536→    ]) {
   537→   537→      if (bs < 16) continue;
   538→   538→      const bTotal = Math.max(2, Math.floor((bs * ARITH_CAPACITY_RATE_LAYOUT) / 8));
   539→   539→      const bData = bTotal - 1; // 1 byte for CRC-8, no padding (v58)
   540→   540→      if (bData <= 0) continue;
   541→   541→      const bNum = Math.floor(innerDnaLenArith / bs);
   542→   542→      const cap = bNum * bData;
   543→   543→      if (cap > bestCap) bestCap = cap;
   544→   544→    }
   545→   545→    const innerNCurrent = addressBytes + payloadBytes + innerParityBytes;
   546→   546→    if (bestCap > 0 && bestCap < innerNCurrent) {
   547→   547→      // Reduce payloadBytes so innerN = bestCap (preserve even alignment for GF(2^16))
   548→   548→      let newPayload = bestCap - addressBytes - innerParityBytes;
   549→   549→      if (newPayload < 8) {
   550→   550→        throw new Error(
   551→   551→          `arithmetic mode: capacity ${bestCap} too small for address(${addressBytes})+parity(${innerParityBytes})+payload(min 8)`,
   552→   552→        );
   553→   553→      }
   554→   554→      if (newPayload % 2 === 1) newPayload -= 1;
   555→   555→      payloadBytes = newPayload;
   556→   556→    }
   557→   557→  }
   558→   558→  return {
   559→   559→    primerLen: cfg.primerLength,
   560→   560→    addressBytes,
   561→   561→    payloadBytes,
   562→   562→    innerParityBytes,
   563→   563→    crcBytes,
   564→   564→    totalInnerBytes,
   565→   565→    convEncodedBytes: 0,
   566→   566→  };
   567→   567→}
   568→   568→
   569→   569→/**
   570→   570→ * v52: Compute the byte layout when useConvolutionalInner is enabled.
   571→   571→ *
   572→   572→ * Layout (between primers):
   573→   573→ *   [address(4) + conv_encoded(LDPC codeword) + CRC(2)]
   574→   574→ *
   575→   575→ * Where:
   576→   576→ *   - address is direct-mapped (for clustering)
   577→   577→ *   - LDPC codeword = address + payload + parity (innerN bytes)
   578→   578→ *   - conv_encoded(LDPC codeword) ≈ 2*innerN + 1 bytes (rate 1/2 + memory tail)
   579→   579→ *   - CRC is over the original LDPC codeword
   580→   580→ *
   581→   581→ * Capacity is roughly halved vs the non-conv layout (rate-1/2 conv code).
   582→   582→ */
   583→   583→export function computeLayoutConv(cfg: CodecConfig): OligoLayout {
   584→   584→  const innerNt = cfg.oligoLength - 2 * cfg.primerLength;
   585→   585→  if (innerNt <= 0) {
   586→   586→    throw new Error(`oligoLength ${cfg.oligoLength} too small for primers`);
   587→   587→  }
   588→   588→  // direct mapping only (conv inner is currently only supported with direct mapping)
   589→   589→  const ntPerByte = 4;
   590→   590→  if (innerNt % ntPerByte !== 0) {
   591→   591→    throw new Error(`inner nt length ${innerNt} must be divisible by ${ntPerByte}`);
   592→   592→  }
   593→   593→  const totalInnerBytes = innerNt / ntPerByte;
   594→   594→  const addressBytes = 4;
   595→   595→  const crcBytes = 2;
   596→   596→  const innerParityBytes = cfg.innerParityBytes;
   597→   597→
   598→   598→  // v61: Conv memory depends on channel. Nanopore uses K=9 (memory=8), others K=3.
   599→   599→  //   outputBits = (innerN*8 + memory) * rate, padded to byte boundary
   600→   600→  //   convEncodedBytes = ceil((innerN*8 + memory) * 2 / 8)
   601→   601→  //   memory=2: convEncodedBytes = ceil(2*innerN + 0.5) = 2*innerN + 1
   602→   602→  //   memory=8: convEncodedBytes = ceil(2*innerN + 2) = 2*innerN + 2
   603→   603→  const convMemory = (cfg.channel === "nanopore") ? 8 : 2;
   604→   604→  const convRate = 2;
   605→   605→  // Total: addressBytes + convEncodedBytes + crcBytes ≤ totalInnerBytes
   606→   606→  //   addressBytes + ceil((innerN*8 + memory) * rate / 8) + crcBytes ≤ totalInnerBytes
   607→   607→  // Solve for innerN:
   608→   608→  //   ceil((innerN*8 + memory) * rate / 8) ≤ totalInnerBytes - addressBytes - crcBytes
   609→   609→  //   (innerN*8 + memory) * rate / 8 ≤ totalInnerBytes - addressBytes - crcBytes  (conservatively)
   610→   610→  //   innerN ≤ (totalInnerBytes - addressBytes - crcBytes) * 8 / (rate * 8) - memory/rate
   611→   611→  //          = (totalInnerBytes - addressBytes - crcBytes) / rate - memory/rate
   612→   612→  //          = (totalInnerBytes - addressBytes - crcBytes - memory) / rate
   613→   613→  const overhead = addressBytes + crcBytes + Math.ceil(convMemory / convRate);
   614→   614→  const maxInnerN = Math.floor((totalInnerBytes - overhead) / convRate);
   615→   615→  if (maxInnerN < addressBytes + innerParityBytes + 1) {
   616→   616→    throw new Error(
   617→   617→      `oligoLength ${cfg.oligoLength} too small for conv inner code: maxInnerN=${maxInnerN}`,
   618→   618→    );
   619→   619→  }
   620→   620→  const innerN = maxInnerN;
   621→   621→  const innerK = innerN - innerParityBytes;
   622→   622→  let payloadBytes = innerK - addressBytes;
   623→   623→  if (payloadBytes <= 0) {
   624→   624→    throw new Error(
   625→   625→      `oligoLength ${cfg.oligoLength} too small for conv inner payload: ${payloadBytes} bytes`,
   626→   626→    );
   627→   627→  }
   628→   628→  // Force even for GF(2^16) outer RS compatibility
   629→   629→  if (payloadBytes % 2 === 1) {
   630→   630→    payloadBytes -= 1;
   631→   631→  }
   632→   632→  const convEncodedBytes = Math.ceil((innerN * 8 + convMemory) * convRate / 8);
   633→   633→
   634→   634→  return {
   635→   635→    primerLen: cfg.primerLength,
   636→   636→    addressBytes,
   637→   637→    payloadBytes,
   638→   638→    innerParityBytes,
   639→   639→    crcBytes,
   640→   640→    totalInnerBytes,
   641→   641→    convEncodedBytes,
   642→   642→  };
   643→   643→}
   644→   644→
   645→   645→/**
   646→   646→ * v62: Compute the byte layout for arithmetic-v2 mode (address OUTSIDE arithmetic stream).
   647→   647→ *
   648→   648→ * Layout (between primers):
   649→   649→ *   [Address (16 nt direct DNA)] [Arithmetic stream (arithmeticNt nt)]
   650→   650→ *
   651→   651→ * Where:
   652→   652→ *   - Address: 4 bytes = 16 nt of direct 2-bit mapping (NOT arithmetic-coded)
   653→   653→ *   - Arithmetic stream: encodes [payload + LDPC parity] at 1.95 b/nt with
   654→   654→ *     per-block CRC-8 sync markers. NO address in the arithmetic stream.
   655→   655→ *     NO CRC-16 (per-block CRC-8 replaces it).
   656→   656→ *
   657→   657→ * LDPC codeword = payload + parity (k = payloadBytes, NO address).
   658→   658→ *
   659→   659→ * This fixes the v57-v61 "arithmetic mode still broken" issue where the
   660→   660→ * address was inside the arithmetic stream, causing:
   661→   661→ *   1. Address corruption from arithmetic termination (last bytes unreliable)
   662→   662→ *   2. Address corruption from IDS (indels shift arithmetic interval state)
   663→   663→ *
   664→   664→ * v62: address is direct DNA → robust to indels via k-mer clustering.
   665→   665→ *      Arithmetic stream is independent → termination corruption only
   666→   666→ *      affects the last byte of each block, corrected by LDPC erasure decoder.
   667→   667→ */
   668→   668→export function computeLayoutArithmeticV2(cfg: CodecConfig): OligoLayout {
   669→   669→  const innerNt = cfg.oligoLength - 2 * cfg.primerLength;
   670→   670→  if (innerNt <= 0) {
   671→   671→    throw new Error(`oligoLength ${cfg.oligoLength} too small for primers`);
   672→   672→  }
   673→   673→
   674→   674→  const addressNt = 16; // 4 bytes × 4 nt/byte (direct 2-bit mapping)
   675→   675→  const arithmeticNtRaw = innerNt - addressNt;
   676→   676→
   677→   677→  // Use blockSize=80 (good balance of density and error confinement)
   678→   678→  //   blockSize=80: 19 bytes/block, 1 CRC = 5.3% overhead → density 1.55 b/nt
   679→   679→  //   blockSize=160: 38 bytes/block, 1 CRC = 2.6% overhead → density 1.65 b/nt
   680→   680→  const blockSize = cfg.arithmeticBlockSize ?? 80;
   681→   681→  const arithmeticNt = Math.floor(arithmeticNtRaw / blockSize) * blockSize;
   682→   682→
   683→   683→  if (arithmeticNt < blockSize * 2) {
   684→   684→    throw new Error(
   685→   685→      `arithmetic-v2: arithmeticNt=${arithmeticNt} too small (need ≥${blockSize * 2}). ` +
   686→   686→      `Use longer oligo or shorter primer.`,
   687→   687→    );
   688→   688→  }
   689→   689→
   690→   690→  const ARITH_CAPACITY_RATE = 1.95;
   691→   691→  const bytesPerBlockTotal = Math.max(2, Math.floor((blockSize * ARITH_CAPACITY_RATE) / 8));
   692→   692→  const bytesPerBlockData = bytesPerBlockTotal - 1; // 1 byte for CRC-8
   693→   693→  const numBlocks = arithmeticNt / blockSize;
   694→   694→  const arithmeticDataBytes = numBlocks * bytesPerBlockData;
   695→   695→
   696→   696→  // LDPC codeword = payload + parity (NO address, NO CRC-16)
   697→   697→  const innerN = arithmeticDataBytes; // = payloadBytes + innerParityBytes
   698→   698→  const innerK = innerN - cfg.innerParityBytes;
   699→   699→  let payloadBytes = innerK;
   700→   700→
   701→   701→  if (payloadBytes <= 0) {
   702→   702→    throw new Error(
   703→   703→      `arithmetic-v2: payloadBytes=${payloadBytes} ≤ 0. Reduce innerParityBytes ` +
   704→   704→      `or increase oligoLength.`,
   705→   705→    );
   706→   706→  }
   707→   707→
   708→   708→  // Force even for GF(2^16) outer RS compatibility
   709→   709→  if (payloadBytes % 2 === 1) {
   710→   710→    payloadBytes -= 1;
   711→   711→  }
   712→   712→
   713→   713→  // totalInnerBytes: for arithmetic-v2, this is a VIRTUAL value used by the
   714→   714→  // outer pipeline. The actual DNA layout is:
   715→   715→  //   16 nt (address) + arithmeticNt (arithmetic stream) = innerNt
   716→   716→  // But the byte-level layout (for LDPC/CRC) is:
   717→   717→  //   addressBytes(4) + payloadBytes + innerParityBytes = innerN
   718→   718→  // There is NO CRC-16 (per-block CRC-8 replaces it).
   719→   719→  // We set totalInnerBytes = addressBytes + innerN so the outer pipeline
   720→   720→  // computes the right innerK/innerN for the LDPC instance.
   721→   721→  const addressBytes = 4;
   722→   722→  const totalInnerBytes = addressBytes + innerN;
   723→   723→
   724→   724→  return {
   725→   725→    primerLen: cfg.primerLength,
   726→   726→    addressBytes,
   727→   727→    payloadBytes,
   728→   728→    innerParityBytes: cfg.innerParityBytes,
   729→   729→    crcBytes: 0, // v62: NO CRC-16 in arithmetic-v2 (per-block CRC-8 replaces it)
   730→   730→    totalInnerBytes,
   731→   731→    convEncodedBytes: 0,
   732→   732→  };
   733→   733→}
   734→   734→
   735→   735→/**
   736→   736→ * v52: Compute the byte layout, dispatching to conv-aware variant if needed.
   737→   737→ * v62: Also dispatch to arithmetic-v2 layout when mappingMode === "arithmetic".
   738→   738→ */
   739→   739→export function computeLayoutAuto(cfg: CodecConfig): OligoLayout {
   740→   740→  if (cfg.useConvolutionalInner) {
   741→   741→    return computeLayoutConv(cfg);
   742→   742→  }
   743→   743→  // v62: arithmetic mode uses the v2 layout (address outside arithmetic stream)
   744→   744→  if ((cfg.mappingMode ?? "direct") === "arithmetic") {
   745→   745→    return computeLayoutArithmeticV2(cfg);
   746→   746→  }
   747→   747→  return computeLayout(cfg);
   748→   748→}
