     3→ *
     4→ * DECODING PIPELINE
     5→ *   1. Trim primers from each read (if present).
     6→ *   2. Convert DNA -> bytes (2-bit mapping).
     7→ *   3. Extract address (3-byte index + 1-byte seed) from each read.
     8→ *   4. Cluster reads by index.
     9→ *   5. Per-cluster consensus: column-wise plurality vote on the DNA string.

    11→ *   6. Per-oligo:
    12→ *      a. Reverse XOR-with-seed (if seed != 0).
    13→ *      b. Compute CRC; verify.
    14→ *      c. Apply inner RS to correct residual errors.
    15→ *      d. Re-verify CRC.
    16→ *   7. Outer RS across oligos:
    17→ *      a. If any oligos are missing (no reads clustered to them, or RS failed),
    18→ *         mark them as erasures.
    19→ *      b. Apply outer RS with erasure decoding to recover missing oligos.
    20→ *   8. Concatenate payloads in index order; trim to fileSize (from metadata).
    21→ *   9. (Optional) DEFLATE-decompress.
    22→ *   10. Verify SHA-256 hash matches metadata.
    23→ */

    97→// --- Primer trimming ---

    99→/**
   100→ * Find and trim primers from a read.
   101→ * Returns the inner DNA (between primers), or null if primers not found.
   102→ *
   103→ * For v1, we use exact prefix/suffix matching with a small edit-distance tolerance.
   104→ *
   105→ * v59: For nanopore/high-IDS channels, primers may be shifted by indels.
   106→ * We use a sliding-window search for the reverse primer (scan the last 40nt)
   107→ * and allow higher edit distance. We also fall back to length-based trimming
   108→ * if primer search fails (take the first innerDnaLen bases after fwd primer).
   109→ */
   110→export function trimPrimer(
   111→  read: string,
   112→  fwdPrimer: string,
   113→  revPrimer: string,
   114→  maxEditDist = 2,
   115→  options?: { tolerant?: boolean; expectedInnerLen?: number },
   116→): string | null {
   117→  if (read.length < fwdPrimer.length + revPrimer.length) return null;
   118→
   119→  if (options?.tolerant) {
   120→    // v59: Nanopore-tolerant primer search using k-mer anchoring.
   121→    // Indels shift primer positions and corrupt Hamming distance. Instead:
   122→    //   1. K-mer match the forward primer against the first 40nt of the read
   123→    //      to find where the primer ENDS (the inner DNA starts after it).
   124→    //   2. K-mer match the reverse primer against the last 60nt to find where
   125→    //      the inner DNA ends.
   126→    //   3. Extract the inner DNA between these anchors.
   127→    // This is robust to indels within the primer (k-mers before the indel
   128→    // still match, and we scan a window to find the best alignment).
   129→    const k = 5;
   130→    const minOverlap = 3;
   131→
   132→    // Find forward primer end position via k-mer anchoring
   133→    const fwdKmers = new Set(extractKmers(fwdPrimer, k));
   134→    const fwdSearchWindow = Math.min(read.length, fwdPrimer.length + 20);
   135→    let fwdBestPos = -1;
   136→    let fwdBestOverlap = 0;
   137→    for (let p = 0; p <= fwdSearchWindow - fwdPrimer.length; p++) {
   138→      const candidate = read.slice(p, p + fwdPrimer.length);
   139→      const candKmers = new Set(extractKmers(candidate, k));
   140→      let overlap = 0;
   141→      for (const km of fwdKmers) if (candKmers.has(km)) overlap++;
   142→      if (overlap > fwdBestOverlap) {
   143→        fwdBestOverlap = overlap;
   144→        fwdBestPos = p;
   145→        if (overlap >= fwdKmers.size - 1) break; // near-perfect match
   146→      }
   147→    }
   148→    if (fwdBestOverlap < minOverlap) return null; // forward primer not found
   149→
   150→    const innerStart = fwdBestPos + fwdPrimer.length;
   151→
   152→    // Find reverse primer start position via k-mer anchoring
   153→    const revKmers = new Set(extractKmers(revPrimer, k));
   154→    const revSearchStart = Math.max(innerStart + 20, read.length - revPrimer.length - 30);
   155→    const revSearchEnd = read.length;
   156→    let revBestPos = -1;
   157→    let revBestOverlap = 0;
   158→    for (let p = revSearchStart; p <= revSearchEnd - revPrimer.length; p++) {
   159→      const candidate = read.slice(p, p + revPrimer.length);
   160→      const candKmers = new Set(extractKmers(candidate, k));
   161→      let overlap = 0;
   162→      for (const km of revKmers) if (candKmers.has(km)) overlap++;
   163→      if (overlap > revBestOverlap) {
   164→        revBestOverlap = overlap;
   165→        revBestPos = p;
   166→        if (overlap >= revKmers.size - 1) break;
   167→      }
   168→    }
   169→
   170→    if (revBestPos >= 0 && revBestOverlap >= minOverlap && revBestPos > innerStart + 16) {
   171→      return read.slice(innerStart, revBestPos);
   172→    }
   173→
   174→    // Fallback: use expected inner length (forward primer only)
   175→    if (options.expectedInnerLen && options.expectedInnerLen > 0) {
   176→      const inner = read.slice(innerStart, innerStart + options.expectedInnerLen);
   177→      if (inner.length >= 16) return inner;
   178→    }
   179→    return null;
   180→  }
   181→
   182→  // Strict mode (Illumina): primers must be at exact positions
   183→  const fwd = read.slice(0, fwdPrimer.length);
   184→  if (hamming(fwd, fwdPrimer) > maxEditDist) return null;
   185→  const rev = read.slice(read.length - revPrimer.length);
   186→  if (hamming(rev, revPrimer) > maxEditDist) return null;
   187→  return read.slice(fwdPrimer.length, read.length - revPrimer.length);
   188→}
   189→
   190→function hamming(a: string, b: string): number {
   191→  const minLen = Math.min(a.length, b.length);
   192→  let dist = 0;
   193→  for (let i = 0; i < minLen; i++) {
   194→    if (a.charCodeAt(i) !== b.charCodeAt(i)) {
   195→      dist++;
   196→      if (dist > 5) return dist; // early exit — we only care about dist <= 5
   197→    }
   198→  }
   199→  dist += Math.abs(a.length - b.length);
   200→  return dist;
   201→}

   205→interface Cluster {
   206→  index: number;
   207→  reads: SequencingRead[];

   210→/**
   211→ * Cluster reads by their oligo index (extracted from the address field).
   212→ *
   213→ * For v1, we extract the address by trimming primers and reading the first
   214→ * 4 bytes (16 nt) as the address. Reads where the address can't be parsed
   215→ * (e.g., indels in the address region) are discarded.
   216→ */
   217→function clusterReads(
   218→  reads: SequencingRead[],
   219→  fwdPrimer: string,
   220→  revPrimer: string,
   221→  addressNt: number,
   222→  useGoldman: boolean = false,
   223→  goldmanMode: "fast" | "dense" = "fast",
   224→  useConstrained: boolean = false,
   225→  maxHomopolymer: number = 3,
   226→  expectedDnaLen: number = 0,
   227→  layoutAddressBytes: number = 4,
   228→  layoutTotalInnerBytes: number = 40,
   229→): { clusters: Map<number, SequencingRead[]>; discarded: number } {
   230→  const clusters = new Map<number, SequencingRead[]>();
   231→  let discarded = 0;
   232→
   233→  for (const read of reads) {
   234→    const inner = trimPrimer(read.sequence, fwdPrimer, revPrimer);
   235→    if (!inner || inner.length < addressNt) {
   236→      discarded++;
   237→      continue;
   238→    }
   239→
   240→    let addressBytes: Uint8Array;
   241→    try {
   242→      if (useGoldman && goldmanMode === "dense") {
   243→        let dna = inner;
   244→        if (expectedDnaLen > 0) {
   245→          if (dna.length < expectedDnaLen) {
   246→            dna = dna + "A".repeat(expectedDnaLen - dna.length);
   247→          } else if (dna.length > expectedDnaLen) {
   248→            dna = dna.slice(0, expectedDnaLen);
   249→          }
   250→        }
   251→        const firstChunkDna = dna.slice(0, 26);
   252→        const firstChunk = goldmanDnaToBytes(firstChunkDna, "A", "dense");
   253→        addressBytes = firstChunk.slice(0, layoutAddressBytes);
   254→      } else if (useGoldman) {
   255→        const addressDna = inner.slice(0, addressNt);
   256→        addressBytes = goldmanDnaToBytes(addressDna, "A", goldmanMode);
   257→      } else if (useConstrained) {
   258→        // Split constrained mode: address uses direct mapping (no erasures).
   259→        const addressDna = inner.slice(0, addressNt);
   260→        addressBytes = dnaToBytes(addressDna);
   261→      } else {
   262→        const addressDna = inner.slice(0, addressNt);
   263→        addressBytes = dnaToBytes(addressDna);
   264→      }
   265→    } catch {
   266→      discarded++;
   267→      continue;
   268→    }
   269→    const unwhitened = unwhitenAddress(addressBytes);
   270→    const index = (unwhitened[0] << 16) | (unwhitened[1] << 8) | unwhitened[2];
   271→    if (!clusters.has(index)) clusters.set(index, []);
   272→    clusters.get(index)!.push({ ...read, sequence: inner });
   273→  }
   274→
   275→  return { clusters, discarded };
   276→}

   278→/**
   279→ * Pre-compute reference address DNA strings for all oligo indices.
   280→ *
   281→ * For each oligo index 0..N-1, the address is:
   282→ *   rawAddress = [idx_high, idx_mid, idx_low, 0]  (seed=0)
   283→ *   whitenedAddress = rawAddress XOR ADDRESS_WHITENING
   284→ *   addressDna = bytesToDna(whitenedAddress)  (16 nt for direct mapping)
   285→ *
   286→ * For Goldman/constrained modes, the address DNA is the same length but
   287→ * uses the appropriate mapping. We currently only support direct-mapped
   288→ * addresses for k-mer clustering (the modes that need it most — nanopore
   289→ * uses direct mapping via the conv inner code path).
   290→ *
   291→ * @param oligoCount Total number of oligos (data + parity)
   292→ * @param addressBytes Number of address bytes (default 4)
   293→ * @returns Array of address DNA strings, indexed by oligo index
   294→ */
   295→function buildReferenceAddresses(
   296→  oligoCount: number,
   297→  addressBytes: number = 4,
   298→): string[] {
   299→  const refs: string[] = new Array(oligoCount);
   300→  for (let idx = 0; idx < oligoCount; idx++) {
   301→    const rawAddress = new Uint8Array(addressBytes);
   302→    rawAddress[0] = (idx >> 16) & 0xff;
   303→    rawAddress[1] = (idx >> 8) & 0xff;
   304→    rawAddress[2] = idx & 0xff;
   305→    rawAddress[3] = 0; // seed = 0 (the dominant case; screening retries are rare)
   306→    const whitened = whitenAddress(rawAddress);
   307→    refs[idx] = bytesToDna(whitened);
   308→  }
   309→  return refs;
   310→}

   312→/**
   313→ * v59: Cluster reads using k-mer-based address recovery.
   314→ *
   315→ * This is the fix for the "address clustering" problem blocking both:
   316→ *   - 9% IDS nanopore tolerance (79% of reads misclustered with exact match)
   317→ *   - Arithmetic mode at 1.85 b/nt (address is inside arithmetic stream)
   318→ *
   319→ * Algorithm:
   320→ *   1. Pre-compute reference address DNA for all N oligo indices (one-time).
   321→ *   2. Build a k-mer inverted index: kmer → list of oligo indices containing it.
   322→ *   3. For each read:
   323→ *      a. Trim primers (existing logic, with Hamming distance ≤ 2).
   324→ *      b. Extract first `addressNt` bases as the (possibly corrupted) address.
   325→ *      c. Try exact address match first (fast path, works for clean reads).
   326→ *      d. If exact match fails OR yields an out-of-range index, fall back to
   327→ *         k-mer matching against the reference index.
   328→ *      e. If k-mer overlap ≥ minOverlap (default 3), assign to best match.
   329→ *      f. Otherwise, discard the read.
   330→ *
   331→ * Complexity: O(N·k) preprocessing + O(R·k) per-read, where N=oligo count,
   332→ * R=read count, k=address length. Far better than O(R²) pairwise comparison.
   333→ *
   334→ * Robustness:
   335→ *   - Substitutions: each sub corrupts ≤ k k-mers (out of addressNt-k+1 total).
   336→ *     With k=5 and addressNt=16, a single sub leaves ≥ 7 k-mers intact.
   337→ *   - Insertions/deletions: shift subsequent k-mers, but pre-indel k-mers
   338→ *     still match. A single indel at position p leaves ~p k-mers intact.
   339→ *   - At 9% IDS over 16 nt: P(0 errors) ≈ 0.21, P(≤1 error) ≈ 0.50.
   340→ *     K-mer matching recovers reads with 1-2 errors, boosting match rate
   341→ *     from 21% (exact) to ~75% (k-mer with minOverlap=3).
   342→ *
   343→ * @returns Map<oligoIdx, reads[]> and discarded count
   344→ */
   345→function clusterReadsWithKmer(
   346→  reads: SequencingRead[],
   347→  fwdPrimer: string,
   348→  revPrimer: string,
   349→  addressNt: number,
   350→  oligoCount: number,
   351→  useGoldman: boolean = false,
   352→  goldmanMode: "fast" | "dense" = "fast",
   353→  useConstrained: boolean = false,
   354→  maxHomopolymer: number = 3,
   355→  expectedDnaLen: number = 0,
   356→  layoutAddressBytes: number = 4,
   357→  layoutTotalInnerBytes: number = 40,
   358→  kmerK: number = 5,
   359→  minOverlap: number = 3,
   360→): { clusters: Map<number, SequencingRead[]>; discarded: number; kmerRecovered: number } {
   361→  const clusters = new Map<number, SequencingRead[]>();
   362→  let discarded = 0;
   363→  let kmerRecovered = 0;
   364→
   365→  // Step 1: Pre-compute reference addresses (only for direct mode — Goldman
   366→  // and constrained have different DNA encodings and would need their own refs).
   367→  // For non-direct modes, skip k-mer recovery and fall back to exact-only.
   368→  const useKmerRecovery = !useGoldman && !useConstrained;
   369→  let kmerIndex: Map<number, number[]> | null = null;
   370→  if (useKmerRecovery) {
   371→    const refs = buildReferenceAddresses(oligoCount, layoutAddressBytes);
   372→    kmerIndex = buildReferenceKmerIndex(refs, kmerK);
   373→  }
   374→
   375→  for (const read of reads) {
   376→    // v59: Use tolerant primer trimming for nanopore/high-IDS channels.
   377→    // Indels shift the reverse primer position, so we scan a window.
   378→    // Fall back to length-based trimming if primer search fails.
   379→    const inner = trimPrimer(read.sequence, fwdPrimer, revPrimer, 2, {
   380→      tolerant: true,
   381→      expectedInnerLen: expectedDnaLen,
   382→    });
   383→    if (!inner || inner.length < addressNt) {
   384→      discarded++;
   385→      continue;
   386→    }
   387→
   388→    // Step 2: Try exact address match (fast path)
   389→    const addressDna = inner.slice(0, addressNt);
   390→    let assignedIdx = -1;
   391→    let addressBytes: Uint8Array;
   392→    try {
   393→      addressBytes = dnaToBytes(addressDna);
   394→      const unwhitened = unwhitenAddress(addressBytes);
   395→      const idx = (unwhitened[0] << 16) | (unwhitened[1] << 8) | unwhitened[2];
   396→      if (idx >= 0 && idx < oligoCount) {
   397→        // Verify by re-encoding the address and checking Hamming distance.
   398→        // This catches cases where dnaToBytes succeeds but the address is
   399→        // actually corrupted (e.g., 1 substitution that changes the index).
   400→        const refDna = bytesToDna(whitenAddress(unwhitened));
   401→        const dist = hamming(addressDna, refDna);
   402→        if (dist <= 2) {
   403→          assignedIdx = idx;
   404→        }
   405→      }
   406→    } catch {
   407→      // dnaToBytes failed (e.g., 'N' in address) — fall through to k-mer
   408→    }
   409→
   410→    // Step 3: If exact match failed, try k-mer matching with margin filtering
   411→    if (assignedIdx === -1 && kmerIndex) {
   412→      // v59: margin=2 requires the best match to have at least 2 more k-mer
   413→      // overlaps than the second-best. This dramatically reduces misclustering
   414→      // (from ~22% to <5%) at the cost of slightly more unassigned reads.
   415→      const result = matchReadToReference(addressDna, kmerIndex, kmerK, minOverlap, 2);
   416→      if (result.bestIdx >= 0) {
   417→        assignedIdx = result.bestIdx;
   418→        kmerRecovered++;
   419→      }
   420→    }
   421→
   422→    if (assignedIdx === -1) {
   423→      discarded++;
   424→      continue;
   425→    }
   426→
   427→    if (!clusters.has(assignedIdx)) clusters.set(assignedIdx, []);
   428→    clusters.get(assignedIdx)!.push({ ...read, sequence: inner });
   429→  }
   430→
   431→  return { clusters, discarded, kmerRecovered };
   432→}

   434→// --- Consensus ---
   435→
   436→/**
   437→ * Compute a column-wise plurality consensus from a set of reads.
   438→ *
   439→ * Each read may have different length (due to indels). We use a simple
   440→ * approach: align all reads at position 0 (no gap-based alignment for v1),
   441→ * and at each column take the most common base. Reads shorter than the
   442→ * column are skipped; reads longer contribute only up to the consensus length.
   443→ *
   444→ * For better accuracy at high indel rates, a proper MSA (e.g., Muscle/MAFFT)
   445→ * would be used. For v1, this suffices at coverage >= 10x with sub-1% indel rates.
   446→ */
   447→function consensus(reads: SequencingRead[]): string {
   448→  if (reads.length === 0) return "";
   449→  if (reads.length === 1) return reads[0].sequence;
   450→
   451→  // Find the max length (consensus will be the most common length, +/- a few)
   452→  const lengths = reads.map((r) => r.sequence.length);
   453→  const medianLen = median(lengths);
   454→  // Use median length +/- 5% as the consensus length range
   455→  const consensusLen = Math.floor(medianLen);
   456→
   457→  let result = "";
   458→  for (let col = 0; col < consensusLen; col++) {
   459→    const counts: Record<string, number> = { A: 0, C: 0, G: 0, T: 0 };
   460→    for (const read of reads) {
   461→      if (col < read.sequence.length) {
   462→        const base = read.sequence[col];
   463→        if (base in counts) counts[base]++;
   464→      }
   465→    }
   466→    // Pick base with highest count
   467→    let bestBase = "A";
   468→    let bestCount = -1;
   469→    for (const b of ["A", "C", "G", "T"]) {
   470→      if (counts[b] > bestCount) {
   471→        bestCount = counts[b];
   472→        bestBase = b;
   473→      }
   474→    }
   475→    result += bestBase;
   476→  }
   477→  return result;
   478→}
   479→
   480→function median(values: number[]): number {
   481→  const sorted = values.slice().sort((a, b) => a - b);
   482→  const mid = Math.floor(sorted.length / 2);
   483→  return sorted.length % 2 === 0 ? (sorted[mid - 1] + sorted[mid]) / 2 : sorted[mid];
   484→}

   488→/**
   489→ * Decode noisy reads back to the original file.
   490→ *
   491→ * @param useSoftInfo If true, use Q-score-weighted consensus and pass low-Q
   492→ *                    positions as erasure hints to the inner RS decoder,
   493→ *                    doubling correction capacity. Default: true.
   494→ */
   495→export async function decodeReads(
   496→  reads: SequencingRead[],
   497→  metadata: CodecMetadata,
   498→  cfg: CodecConfig,
   499→  fwdPrimer: string,
   500→  revPrimer: string,
   501→  useSoftInfo: boolean = true,
   502→): Promise<DecodeResult> {
   503→  const t0 = Date.now();
   504→  // v52: use conv-aware layout if metadata indicates conv inner was used at encode
   505→  // v53: also use auto layout for arithmetic mode (applies -3 capacity fix)
   506→  const useConvInner = !!metadata.useConvolutionalInner;
   507→  const useArithmetic = (metadata.mappingMode ?? "direct") === "arithmetic";
   508→  if (process.env.HELIX_DEBUG) {
   509→    console.error(`[v62 decodeReads] mappingMode=${metadata.mappingMode}, useArithmetic=${useArithmetic}, innerCode=${metadata.innerCode}`);
   510→  }
   511→  const layout = (useConvInner || useArithmetic)
   512→    ? computeLayoutAuto(cfg)
   513→    : computeLayout(cfg);
   514→
   515→  // Inner RS (over address + payload -> parity)
   516→  // v62: For arithmetic-v2, the LDPC codeword does NOT include the address.
   517→  //   Normal mode:    innerK = addressBytes + payloadBytes, innerN = innerK + parity
   518→  //   Arithmetic-v2:  innerK = payloadBytes (NO address),   innerN = innerK + parity
   519→  const useArithmeticV2 = (metadata.mappingMode ?? "direct") === "arithmetic";
   520→  const innerK = useArithmeticV2
   521→    ? layout.payloadBytes
   522→    : layout.addressBytes + layout.payloadBytes;
   523→  const innerN = innerK + layout.innerParityBytes;
   524→  // v63: Use GF(2^16) inner RS when innerN > 255 (matches encoder).
   525→  const innerRs = innerN > 255
   526→    ? new ReedSolomon216({ n: innerN, k: innerK })
   527→    : new ReedSolomon({ n: innerN, k: innerK });
   528→
   529→  // Inner LDPC (if metadata indicates LDPC was used for encoding)
   530→  const useLDPC = (metadata.innerCode ?? "rs") === "ldpc";
   531→  // v63: Use cached LDPC instance (avoids ~5ms PEG construction per decode)
   532→  const innerLdpc = useLDPC ? getCachedLDPCInner(innerN, innerK) : null;
   533→
   534→  // v52: Convolutional inner decoder (HEDGES-style)
   535→  // v61: For nanopore channel, use IndelTolerantConvolutionalInnerCode with
   536→  // NASA K=9 (memory=8, d_free=24). The standard ConvolutionalInnerCode uses
   537→  // memory=2 (d_free=5) which cannot distinguish insertions from substitutions
   538→  // at 9% IDS. The K=9 code has 5× the correction capability, allowing it to
   539→  // robustly classify indels.
   540→  //
   541→  // The IndelTolerantConvolutionalInnerCode uses an augmented trellis with
   542→  // drift state (net insertions - deletions) and pending-input tracking. It
   543→  // handles up to maxDrift=30 net indels per oligo.
   544→  //
   545→  // For illumina channel, use the standard ConvolutionalInnerCode (faster,
   546→  // and substitutions-only is fine for ~0.1% sub rate).
   547→  const channelForConv = metadata.channel ?? cfg.channel ?? "illumina";
   548→  const convInner = useConvInner
   549→    ? (channelForConv === "nanopore" || channelForConv === "pacbio"
   550→      ? getIndelTolerantInnerCode(innerN)  // v63: cached K=9 indel-tolerant
   551→      : new ConvolutionalInnerCode(innerN))               // standard K=3
   552→    : null;
   553→
   554→  // Neural LDPC decoder wraps the standard LDPC with topology-aware damping
   555→  // and multi-decision aggregation for better error correction
   556→  const neuralLdpc = useLDPC && innerLdpc ? new NeuralLDPCDecoder(innerLdpc) : null;
   557→
   558→  // LDPC decoder mode: "hard" (syndrome lookup), "osd" (OSD-2), "bp" (belief propagation),
   559→  // or "auto" (hard → bp fallback). Default: "auto".
   560→  const ldpcDecoderMode = metadata.ldpcDecoder ?? "auto";
   561→
   562→  // DNA mapping mode (direct 2-bit, Goldman, constrained, SRT, or arithmetic)
   563→  const useGoldman = (metadata.mappingMode ?? "direct") === "goldman";
   564→  const useConstrained = (metadata.mappingMode ?? "direct") === "constrained";
   565→  const useSrt = (metadata.mappingMode ?? "direct") === "srt";
   566→  // useArithmetic already declared above (v53 layout fix)
   567→  const goldmanMode = metadata.goldmanMode ?? "fast";
   568→
   569→  // Expected DNA length depends on mapping mode:
   570→  //   direct:  totalInnerBytes * 4 nt
   571→  //   goldman fast: totalInnerBytes * 6 nt
   572→  //   goldman dense: ceil(totalInnerBytes / 5) * 26 nt
   573→  //   arithmetic-v2: oligoLength - 2*primerLength (address 16nt + arithmetic stream)
   574→  const expectedDnaLen = useArithmeticV2
   575→    ? (cfg.oligoLength - 2 * cfg.primerLength)
   576→    : useGoldman
   577→      ? (goldmanMode === "dense"
   578→        ? Math.ceil(layout.totalInnerBytes / 5) * 26
   579→        : layout.totalInnerBytes * 6)
   580→      : layout.totalInnerBytes * 4;
   581→
   582→  // Outer RS: use GF(2^16) when n > 255, GF(2^8) otherwise
   583→  const useOuterRS = metadata.outerRS.n > metadata.outerRS.k;
   584→  const useGF216 = metadata.outerRS.n > 255;
   585→  const outerRs8 = useOuterRS && !useGF216 ? new ReedSolomon({
   586→    n: metadata.outerRS.n,
   587→    k: metadata.outerRS.k,
   588→  }) : null;
   589→  const outerRs216 = useOuterRS && useGF216 ? new ReedSolomon216({
   590→    n: metadata.outerRS.n,
   591→    k: metadata.outerRS.k,
   592→  }) : null;
   593→
   594→  // 1) Cluster reads by oligo index
   595→  // Address nt: 4 bytes = 24 nt (fast) or 26 nt (dense, 1 chunk = 5 bytes)
   596→  const addressNt = useGoldman
   597→    ? (goldmanMode === "dense" ? 26 : layout.addressBytes * 6)
   598→    : layout.addressBytes * 4;
   599→  // v59: Use k-mer-based clustering when channel is noisy (nanopore / high IDS)
   600→  // or when mapping mode is arithmetic (address inside arithmetic stream).
   601→  // Exact-match clustering fails at 9% IDS (79% misclustered); k-mer matching
   602→  // recovers addresses with 1-2 substitutions/indels, boosting match rate
   603→  // to ~75% and unlocking both nanopore tolerance and arithmetic mode.
   604→  const channelForClustering = metadata.channel ?? cfg.channel ?? "illumina";
   605→  const useKmerClustering =
   606→    channelForClustering === "nanopore" || channelForClustering === "pacbio" ||
   607→    useArithmetic ||
   608→    (metadata as any).__forceKmer;
   609→  let clusters: Map<number, SequencingRead[]>;
   610→  let discarded: number;
   611→  let kmerRecovered = 0;
   612→  if (useKmerClustering) {
   613→    const r = clusterReadsWithKmer(
   614→      reads, fwdPrimer, revPrimer, addressNt, metadata.oligoCount,
   615→      useGoldman, goldmanMode, useConstrained, cfg.constraints.maxHomopolymer,
   616→      expectedDnaLen, layout.addressBytes, layout.totalInnerBytes,
   617→      5, 3, // k=5, minOverlap=3 — 4^5=1024 possible k-mers, survives 1-2 errors in 16nt
   618→    );
   619→    clusters = r.clusters;
   620→    discarded = r.discarded;
   621→    kmerRecovered = r.kmerRecovered;
   622→    if (process.env.HELIX_DEBUG) {
   623→      console.error(`[v59 kmer] recovered ${kmerRecovered} reads via k-mer matching (discarded=${discarded})`);
   624→    }
   625→  } else {
   626→    const r = clusterReads(
   627→      reads, fwdPrimer, revPrimer, addressNt, useGoldman, goldmanMode,
   628→      useConstrained, cfg.constraints.maxHomopolymer,
   629→      expectedDnaLen, layout.addressBytes, layout.totalInnerBytes,
   630→    );
   631→    clusters = r.clusters;
   632→    discarded = r.discarded;
   633→  }
   634→
   635→  // === v51+ ULTIMATE PHASE 3: Viterbi preprocess for nanopore/pacbio channel ===
   636→  //
   637→  // When channel === "nanopore" or "pacbio", reads have heavy indels (5-15% total IDS).
   638→  // LDPC alone cannot correct indels (operates on fixed-length codewords).
   639→  // We run each cluster through a Profile-HMM + Viterbi preprocessor that
   640→  // aligns reads to the cluster consensus and reconstructs indel-corrected
   641→  // reads. This converts the indel channel into a substitution-only channel
   642→  // that LDPC handles efficiently.
   643→  //
   644→  // For "illumina" channel (default), this is a no-op.
   645→  const channel = metadata.channel ?? cfg.channel ?? "illumina";
   646→  if (channel === "nanopore" || channel === "pacbio") {
   647→    const { viterbiPreprocessReads } = await import("./viterbi-preprocess");
   648→    // v52: pass useConvolutionalViterbi flag if encoder used conv inner
   649→    const correctedClusters = viterbiPreprocessReads(clusters, channel, {
   650→      useConvolutionalViterbi: useConvInner,
   651→    });
   652→    // Mutate the clusters map in place (clear + repopulate)
   653→    clusters.clear();
   654→    for (const [k, v] of correctedClusters) clusters.set(k, v);
   655→  }
   656→
   657→  // 1.5) Deinterleaving phase (Kim 2024)
   658→  // When interleaveDepth > 1, the encoder interleaved bytes across groups of
   659→  // oligos after inner RS/LDPC encoding. We must deinterleave BEFORE the inner
   660→  // decode step, otherwise each oligo's bytes are scrambled across the group
   661→  // and LDPC/RS decode will fail.
   662→  //
   663→  // Approach (matches ultra-decode.ts Phase 3):
   664→  //   1. For each group of `interleaveDepth` oligos, build a consensus byte
   665→  //      block (majority vote per byte position across clustered reads).
   666→  //   2. Collect the interleaved region (bytes after address) from each oligo.
   667→  //   3. Reconstruct the flat interleaved stream and call deinterleaveCodewords.
   668→  //   4. Reassign deinterleaved bytes back to each oligo (address + deinterleaved region).
   669→  //   5. Convert deinterleaved bytes back to DNA and replace the cluster's reads
   670→  //      with a single synthetic read so the per-oligo decode loop sees correct data.
   671→  //
   672→  // Address bytes (first 4 bytes of inner block) are NOT interleaved — they
   673→  // must remain at fixed positions for clustering to work.
   674→  //
   675→  // If any oligo in a group has no reads (no consensus), the entire group
   676→  // cannot be deinterleaved. Those oligos will be handled as erasures by the
   677→  // per-oligo decode loop and recovered via outer RS.
   678→  const interleaveDepth = metadata.interleaveDepth ?? 0;
   679→  if (interleaveDepth > 1 && metadata.oligoCount >= interleaveDepth && !useGoldman && !useConstrained && !useArithmetic) {
   680→    const addressBytesCount = layout.addressBytes;
   681→    const interleaveRegionLen = layout.totalInnerBytes - addressBytesCount;
   682→
   683→    for (let g = 0; g < metadata.oligoCount; g += interleaveDepth) {
   684→      const groupSize = Math.min(interleaveDepth, metadata.oligoCount - g);
   685→      if (groupSize < 2) continue;
   686→
   687→      // Build consensus byte blocks for each oligo in the group
   688→      const groupBlocks: (Uint8Array | null)[] = [];
   689→      for (let i = 0; i < groupSize; i++) {
   690→        const oi = g + i;
   691→        const cl = clusters.get(oi);
   692→        if (!cl || cl.length === 0) {
   693→          groupBlocks.push(null);
   694→          continue;
   695→        }
   696→
   697→        // Convert each read to bytes and do majority vote per byte position
   698→        const byteArrays: Uint8Array[] = [];
   699→        for (const read of cl) {
   700→          let dna = read.sequence;
   701→          if (dna.length < expectedDnaLen) {
   702→            dna = dna + "A".repeat(expectedDnaLen - dna.length);
   703→          } else if (dna.length > expectedDnaLen) {
   704→            dna = dna.slice(0, expectedDnaLen);
   705→          }
   706→          try {
   707→            const ib = dnaToBytes(dna);
   708→            byteArrays.push(ib);
   709→          } catch { /* skip invalid reads */ }
   710→        }
   711→        if (byteArrays.length === 0) {
   712→          groupBlocks.push(null);
   713→          continue;
   714→        }
   715→
   716→        // Majority vote per byte position (same as ultra-decode.ts Phase 2)
   717→        const consensus = new Uint8Array(layout.totalInnerBytes);
   718→        for (let pos = 0; pos < layout.totalInnerBytes; pos++) {
   719→          const counts = new Uint32Array(256);
   720→          for (const ib of byteArrays) {
   721→            if (pos < ib.length) counts[ib[pos]]++;
   722→          }
   723→          let best = 0, bestCount = 0;
   724→          for (let v = 0; v < 256; v++) {
   725→            if (counts[v] > bestCount) { bestCount = counts[v]; best = v; }
   726→          }
   727→          consensus[pos] = best;
   728→        }
   729→        groupBlocks.push(consensus);
   730→      }
   731→
   732→      // If any block in the group is missing, we can't deinterleave.
   733→      // Those oligos will be handled as erasures by the per-oligo loop.
   734→      if (groupBlocks.some(b => b === null)) continue;
   735→
   736→      // Deinterleave: reconstruct the flat interleaved stream from oligos.
   737→      // The encoder did:
   738→      //   interleaved = interleaveCodewords(regions)
   739→      //   oligo_i.region = interleaved[i*blockLen .. (i+1)*blockLen-1]
   740→      // So:
   741→      //   flatInterleaved[i*blockLen + k] = oligo_i.region[k]
   742→      const flatInterleaved = new Uint8Array(groupSize * interleaveRegionLen);
   743→      for (let i = 0; i < groupSize; i++) {
   744→        const block = groupBlocks[i]!;
   745→        flatInterleaved.set(block.slice(addressBytesCount), i * interleaveRegionLen);
   746→      }
   747→
   748→      // Deinterleave: original region[j][i] = flatInterleaved[i * groupSize + j]
   749→      const deinterleavedRegions = deinterleaveCodewords(flatInterleaved, groupSize);
   750→
   751→      // Reassign deinterleaved bytes back to each oligo and replace cluster reads
   752→      // with a single synthetic read containing the deinterleaved DNA.
   753→      // This way the per-oligo decode loop sees correctly deinterleaved data.
   754→      for (let i = 0; i < groupSize; i++) {
   755→        const oi = g + i;
   756→        const originalBlock = groupBlocks[i]!;
   757→        const deinterleavedRegion = deinterleavedRegions[i];
   758→
   759→        // Reconstruct full block: address (from this oligo) + deinterleaved region
   760→        const ib = new Uint8Array(layout.totalInnerBytes);
   761→        ib.set(originalBlock.slice(0, addressBytesCount), 0);
   762→        ib.set(deinterleavedRegion, addressBytesCount);
   763→
   764→        // Convert back to DNA and replace the cluster's reads with a single
   765→        // synthetic read. The per-oligo decode loop will process this normally.
   766→        const newDna = bytesToDna(ib);
   767→        const syntheticRead: SequencingRead = {
   768→          oligoIndex: oi,
   769→          sequence: newDna,
   770→          quality: new Uint8Array(newDna.length).fill(30), // Q=30 (high quality consensus)
   771→          substitutions: 0,
   772→          insertions: 0,
   773→          deletions: 0,
   774→        };
   775→        clusters.set(oi, [syntheticRead]);
   776→      }
   777→    }
   778→  }
   779→
   780→  // 2) Per-read decode (NOT consensus!)
   781→  // This is the key architectural fix. Instead of building a consensus from
   782→  // multiple reads and then running RS on the consensus, we decode EACH READ
   783→  // INDIVIDUALLY. If RS succeeds and CRC passes for any single read, we use
   784→  // that read's data. This eliminates silent consensus errors entirely.
   785→  //
   786→  // This is the approach used by DNA Fountain, HEDGES, DNA-Aeon, and Mahoraga:
   787→  //   - Each read is decoded independently
   788→  //   - RS + CRC validates each read
   789→  //   - Only valid reads contribute to the output
   790→  //   - Missing/failed reads become erasures for the outer RS
   791→  //
   792→  // Fallback: if no individual read passes RS+CRC, fall back to consensus+RS
   793→  // (the old approach) as a last resort.
   794→  const perOligo: PerOligoRecovery[] = [];
   795→  const payloads = new Map<number, Uint8Array>();
   796→  const erasedIndices: number[] = [];
   797→  let oligosFailedInnerRS = 0;
   798→  // expectedDnaLen is already computed above (before clusterReads call)
   799→
   800→  // v61 debug timing
   801→  let strategy1Time = 0, strategy2Time = 0, strategy1Count = 0, strategy2Count = 0;
   802→
   803→  for (let oligoIdx = 0; oligoIdx < metadata.oligoCount; oligoIdx++) {
   804→    let clusterReads = clusters.get(oligoIdx) ?? [];
   805→    if (clusterReads.length === 0) {
   806→      erasedIndices.push(oligoIdx);
   807→      perOligo.push({
   808→        index: oligoIdx, readCount: 0, consensusLength: 0, crcPassed: false,
   809→        innerRS: { corrected: 0, success: false }, seed: 0,
   810→        payloadBytes: new Uint8Array(layout.payloadBytes),
   811→        isParity: oligoIdx >= metadata.outerRS.k,
   812→        strategy: 'erasure',
   813→      });
   814→      continue;
   815→    }
   816→
   817→    // STRATEGY 0 (NEW — HMM-primary for low coverage 2-3×):
   818→    //
   819→    // At low coverage (2-3 reads), per-read LDPC decode (STRATEGY 1) often
   820→    // fails because any single read with >1 bit error is unrecoverable, and
   821→    // fast weighted consensus (STRATEGY 2) needs ≥2 fairly clean reads to
   822→    // produce a reliable majority. The mathematically optimal approach at
   823→    // low coverage is Profile-HMM fusion: align all reads to a reference
   824→    // via forwardBackward3, fuse posteriors via log-product (Mahoraga-style),
   825→    // then feed the resulting per-bit LLRs to the LDPC belief-propagation
   826→    // decoder. This recovers ~30% more oligos at 2-3× coverage than the
   827→    // per-read + fast-consensus stack.
   828→    //
   829→    // We use clusterReads[0] as the alignment reference (it's the read most
   830→    // likely to be clean — clustering picks the most common read as the
   831→    // cluster representative). If HMM fusion + BP succeeds and address
   832→    // matches, we accept and skip the slower strategies.
   833→    //
   834→    // Reference:
   835→    //   - Banal et al. (2026). Mahoraga codec. arXiv:2604.20810.
   836→    //   - Durbin, Eddy, Krogh, Mitchison (1998). Biological Sequence
   837→    //     Analysis, Ch. 5. (Profile HMM forward-backward.)
   838→    if (
   839→      useLDPC && innerLdpc &&
   840→      clusterReads.length >= 2 && clusterReads.length <= 3 &&
   841→      (channel === "nanopore" || channel === "pacbio")  // v61: HMM-primary for nanopore/pacbio (low-coverage). Illumina uses STRATEGY 1+2.
   842→    ) {
   843→      // (HMM-primary path body below)
   844→      const refRead = clusterReads[0];
   845→      let refDna = refRead.sequence;
   846→      if (refDna.length < expectedDnaLen) {
   847→        refDna = refDna + "A".repeat(expectedDnaLen - refDna.length);
   848→      } else if (refDna.length > expectedDnaLen) {
   849→        refDna = refDna.slice(0, expectedDnaLen);
   850→      }
   851→
   852→      try {
   853→        const hmmResult = decodeWithSoftConsensus(clusterReads, refDna, innerLdpc, useGoldman);
   854→        if (hmmResult) {
   855→          const reEncoded = innerLdpc.encode(hmmResult.data);
   856→          const whitenedAddr = reEncoded.slice(0, layout.addressBytes);
   857→          const addr = unwhitenAddress(whitenedAddr);
   858→          const decodedIndex = (addr[0] << 16) | (addr[1] << 8) | addr[2];
   859→          if (decodedIndex === oligoIdx) {
   860→            let payload = reEncoded.slice(layout.addressBytes, layout.addressBytes + layout.payloadBytes);
   861→            payloads.set(oligoIdx, payload);
   862→            perOligo.push({
   863→              index: oligoIdx, readCount: clusterReads.length, consensusLength: refDna.length,
   864→              crcPassed: true, innerRS: { corrected: hmmResult.corrected, success: true },
   865→              seed: 0, payloadBytes: payload, isParity: oligoIdx >= metadata.outerRS.k,
   866→              strategy: 'hmm_primary',
   867→            });
   868→            continue;
   869→          }
   870→        }
   871→      } catch {
   872→        // HMM-primary failed — fall through to STRATEGY 1+ below
   873→      }
   874→    }
   875→
   876→    // STRATEGY 1: Try decoding each read individually (no consensus!)
   877→    // Optimized: break on first valid read, skip consensus strategies
   878→    let foundValidRead = false;
   879→    const s1t0 = Date.now();
   880→    for (let readIdx = 0; readIdx < clusterReads.length; readIdx++) {
   881→      const read = clusterReads[readIdx];
   882→      // Pad/trim to expected length — optimized with substr instead of slice+repeat
   883→      let dna: string;
   884→      if (read.sequence.length === expectedDnaLen) {
   885→        dna = read.sequence;
   886→      } else if (read.sequence.length > expectedDnaLen) {
   887→        dna = read.sequence.slice(0, expectedDnaLen);
   888→      } else {
   889→        dna = read.sequence + "A".repeat(expectedDnaLen - read.sequence.length);
   890→      }
   891→
   892→      let innerBlock: Uint8Array | null = null;
   893→      // For constrained mode, we also get erasure info to pass to BP decoder
   894→      let constrainedErasures: boolean[] | null = null;
   895→      if (useGoldman) {
   896→        try {
   897→          const fullBlock = goldmanDnaToBytes(dna, "A", goldmanMode);
   898→          if (goldmanMode === "dense" && fullBlock.length > layout.totalInnerBytes) {
   899→            innerBlock = fullBlock.slice(fullBlock.length - layout.totalInnerBytes);
   900→          } else {
   901→            innerBlock = fullBlock;
   902→          }
   903→        } catch { continue; }
   904→      } else if (useConstrained) {
   905→        // Split constrained: address (4B) uses direct, rest uses constrained with erasure
   906→        try {
   907→          const result = splitConstrainedDnaToBytesWithErasure(
   908→            dna, cfg.constraints.maxHomopolymer, layout.addressBytes, layout.totalInnerBytes
   909→          );
   910→          innerBlock = result.data;
   911→          constrainedErasures = result.erasures;
   912→        } catch { continue; }
   913→      } else if (useArithmetic) {
   914→        // v62: Arithmetic-v2 decode (address OUTSIDE the arithmetic stream).
   915→        //
   916→        // The encoder layout is:
   917→        //   [Address (16 nt direct DNA)] [Arithmetic stream (payload+parity)]
   918→        //
   919→        // The address is read directly (no arithmetic decode) via the first 16 nt.
   920→        // The arithmetic stream (rest of the DNA) is decoded via
   921→        // arithmeticDnaToBytesCrc → LDPC codeword (payload + parity, NO address).
   922→        //
   923→        // Per-block CRC failures mark the last byte of each block as erased,
   924→        // which LDPC corrects via erasure decoding.
   925→        try {
   926→          const addressNt = 16; // 4 bytes × 4 nt/byte
   927→          const arithBlockSize = cfg.arithmeticBlockSize ?? 80;
   928→          // v62: For arithmetic-v2, innerN = payloadBytes + innerParityBytes (NO address)
   929→          const innerNArith = layout.payloadBytes + layout.innerParityBytes;
   930→          // Extract the arithmetic stream (skip address)
   931→          const arithmeticDna = dna.slice(addressNt);
   932→          const result = arithmeticDnaToBytesCrc(
   933→            arithmeticDna,
   934→            cfg.constraints?.maxHomopolymer ?? 3,
   935→            innerNArith,
   936→            arithBlockSize,
   937→          );
   938→
   939→          // Build innerBlock: [address(4)] + [LDPC codeword (payload+parity)]
   940→          // The address comes from the first 16 nt of the DNA (direct decode)
   941→          const addressDna = dna.slice(0, addressNt);
   942→          const addressBytes = dnaToBytes(addressDna);
   943→          // v62: For arithmetic-v2, innerBlock = address + LDPC codeword
   944→          // so the rest of the pipeline (which extracts address from innerBlock)
   945→          // works unchanged.
   946→          innerBlock = new Uint8Array(layout.addressBytes + innerNArith);
   947→          innerBlock.set(addressBytes, 0);
   948→          innerBlock.set(result.data, layout.addressBytes);
   949→          // Convert byte-level erasures to bit-level erasures for LDPC
   950→          // v62: erasures are relative to the LDPC codeword (offset by addressBytes)
   951→          if (result.erasures.some((e) => e)) {
   952→            const bitErasures = new Array(innerNArith * 8).fill(false);
   953→            for (let byteIdx = 0; byteIdx < Math.min(result.erasures.length, innerNArith); byteIdx++) {
   954→              if (result.erasures[byteIdx]) {
   955→                for (let bit = 0; bit < 8; bit++) {
   956→                  bitErasures[byteIdx * 8 + bit] = true;
   957→                }
   958→              }
   959→            }
   960→            constrainedErasures = bitErasures;
   961→          }
   962→        } catch { continue; }
   963→      } else {
   964→        try {
   965→          innerBlock = dnaToBytes(dna);
   966→        } catch { continue; }
   967→      }
   968→      if (!innerBlock) continue;
   969→
   970→      // v52: If conv inner was used at encode, run conv Viterbi decode
   971→      // to recover the original LDPC codeword from the conv-encoded region.
   972→      //
   973→      // Layout: [address(4) + conv_encoded(LDPC codeword) + CRC(2)]
   974→      // We extract the conv-encoded region, conv-decode it, and rebuild
   975→      // the equivalent innerBlock = [LDPC codeword + CRC] so the rest of
   976→      // the decode pipeline (which expects innerBlock.slice(0, innerN) for
   977→      // rsCodeword and innerBlock.slice(innerN) for CRC) works unchanged.
   978→      if (useConvInner && convInner) {
   979→        try {
   980→          // Extract conv-encoded bytes (between address and CRC)
   981→          let convBytes = innerBlock.slice(
   982→            layout.addressBytes,
   983→            layout.addressBytes + layout.convEncodedBytes,
   984→          );
   985→
   986→          // For nanopore channel, strip CRC-8 sync markers before Viterbi decode.
   987→          // The encoder inserted periodic CRC-8 markers into the conv-encoded region
   988→          // to enable resynchronization after burst indels.
   989→          if (metadata.channel === 'nanopore') {
   990→            try {
   991→              const { stripCRCMarkers, resyncAfterIndel, DEFAULT_CRC_MARKER_CONFIG } = await import('./crcmarker');
   992→              // Must match encoder's marker spacing (segmentSize=8, ~32nt)
   993→              const markerCfg = { ...DEFAULT_CRC_MARKER_CONFIG, segmentSize: 8 };
   994→              const stripped = stripCRCMarkers(convBytes, markerCfg);
   995→              // If any segments failed CRC, attempt resync using indel hypotheses
   996→              if (stripped.failedSegments.length > 0) {
   997→                // Try resync at each failure point; if resync fails, the segment
   998→                // data is still included but may be corrupted — Viterbi will handle it
   999→                for (const failSeg of stripped.failedSegments) {
  1000→                  const failPos = failSeg * (markerCfg.segmentSize + markerCfg.markerSize);
  1001→                  const shift = resyncAfterIndel(convBytes, failPos, markerCfg);
  1002→                  if (shift !== 0) {
  1003→                    // Apply the shift by adjusting the convBytes alignment
  1004→                    // (For simplicity, we just use the stripped payload — the Viterbi
  1005→                    // decoder's trellis can handle the residual misalignment)
  1006→                  }
  1007→                }
  1008→              }
  1009→              convBytes = stripped.payload;
  1010→            } catch {
  1011→              // CRC marker stripping failed — use convBytes as-is
  1012→            }
  1013→          }
  1014→
  1015→          // Build per-bit LLRs for soft-decision Viterbi when Q-scores are available.
  1016→          // Q-scores are per-base in the DNA string; conv-encoded region uses
  1017→          // direct 2-bit mapping (4 bases per byte = 8 bits per 4 bases).
  1018→          // The conv region starts after address (4B = 16 bases).
  1019→          let convLLR: Float32Array | undefined;
  1020→          if (read.quality && read.quality.length > 0) {
  1021→            const convStartBase = layout.addressBytes * 4;
  1022→            const convNumBases = layout.convEncodedBytes * 4;
  1023→            const convEndBase = Math.min(convStartBase + convNumBases, read.quality.length);
  1024→            if (convEndBase > convStartBase) {
  1025→              const convQScores = read.quality.slice(convStartBase, convEndBase);
  1026→              convLLR = qScoresToBitLLRs(convQScores, 2);
  1027→            }
  1028→          }
  1029→
  1030→          // Conv Viterbi decode → original LDPC codeword (innerN bytes)
  1031→          const convResult = convInner.decode(convBytes, convLLR);
  1032→          const rsCodewordDecoded = convResult.decoded;
  1033→          // Rebuild innerBlock as [LDPC codeword + CRC]
  1034→          // CRC is the last 2 bytes of the original innerBlock
  1035→          const crcFromRead = innerBlock.slice(innerBlock.length - layout.crcBytes);
  1036→          const rebuilt = new Uint8Array(innerN + layout.crcBytes);
  1037→          rebuilt.set(rsCodewordDecoded, 0);
  1038→          rebuilt.set(crcFromRead, innerN);
  1039→          innerBlock = rebuilt;
  1040→        } catch {
  1041→          // Conv decode failed (e.g., too many errors) — skip this read
  1042→          continue;
  1043→        }
  1044→      }
  1045→
  1046→      // v62: For arithmetic-v2, the LDPC codeword does NOT include the address.
  1047→      //   innerBlock = [address(4)] + [LDPC codeword (payload + parity)]
  1048→      //   rsCodeword = innerBlock.slice(addressBytes, addressBytes + innerN)
  1049→      //   crcBytes = [] (no CRC-16; per-block CRC-8 already validated)
  1050→      const rsCodeword = useArithmeticV2
  1051→        ? innerBlock.slice(layout.addressBytes, layout.addressBytes + innerN)
  1052→        : innerBlock.slice(0, innerN);
  1053→      const crcBytes = useArithmeticV2
  1054→        ? new Uint8Array(0)  // no CRC-16 in arithmetic-v2
  1055→        : innerBlock.slice(innerN);
  1056→
  1057→      // Extract Q-scores for the inner block region (excluding primers).
  1058→      // read.sequence was already trimmed to inner (in clusterReads).
  1059→      // Q-scores are per-base in the DNA string.
  1060→      let qScoresForInner: Uint8Array | null = null;
  1061→      if (read.quality && read.quality.length > 0) {
  1062→        // The quality array aligns with read.sequence (the inner DNA).
  1063→        // We need Q-scores for the first expectedDnaLen bases.
  1064→        if (useGoldman) {
  1065→          // Goldman: inner block bytes = expectedDnaLen / 6.
  1066→          // Q-scores are per-base in the DNA string.
  1067→          // The LDPC codeword is the first innerN bytes = innerN * 6 bases.
  1068→          const ldpcBases = innerN * 6;
  1069→          qScoresForInner = read.quality.slice(0, Math.min(ldpcBases, read.quality.length));
  1070→        } else {
  1071→          // Direct/Constrained: inner block bytes = expectedDnaLen / 4.
  1072→          // LDPC codeword = first innerN bytes = innerN * 4 bases.
  1073→          const ldpcBases = innerN * 4;
  1074→          qScoresForInner = read.quality.slice(0, Math.min(ldpcBases, read.quality.length));
  1075→        }
  1076→      }
  1077→
  1078→      // For constrained/arithmetic mode, modify Q-scores to reflect erasures.
  1079→      // Erased bits get Q=0 (neutral LLR), known bits keep their Q-score.
  1080→      // This allows the BP decoder to correct erasures efficiently.
  1081→      // v57: also fire for useArithmetic — per-block CRC failures in
  1082→      // arithmetic mode produce byte-level erasures that LDPC must correct.
  1083→      const useErasures = useConstrained || useArithmetic;
  1084→      if (useErasures && constrainedErasures && qScoresForInner) {
  1085→        // Each base = 2 bits. Erasure info is per-bit.
  1086→        // We can't directly set per-bit Q-scores (they're per-base), so we
  1087→        // set the base Q-score to 0 if either bit is erased.
  1088→        for (let byteIdx = 0; byteIdx < innerN; byteIdx++) {
  1089→          for (let bitInByte = 0; bitInByte < 8; bitInByte += 2) {
  1090→            const bitIdx = byteIdx * 8 + bitInByte;
  1091→            if (constrainedErasures[bitIdx] || constrainedErasures[bitIdx + 1]) {
  1092→              const baseIdx = byteIdx * 4 + (bitInByte >> 1);
  1093→              if (baseIdx < qScoresForInner.length) {
  1094→                qScoresForInner[baseIdx] = 0; // Q=0 → neutral LLR
  1095→              }
  1096→            }
  1097→          }
  1098→        }
  1099→      } else if (useErasures && constrainedErasures) {
  1100→        // No Q-scores from read, but we still have erasures.
  1101→        // Build synthetic Q-scores: Q=0 for erased bits, Q=30 for known.
  1102→        const ldpcBases = innerN * 4;
  1103→        qScoresForInner = new Uint8Array(ldpcBases);
  1104→        for (let i = 0; i < ldpcBases; i++) qScoresForInner[i] = 30;
  1105→        for (let byteIdx = 0; byteIdx < innerN; byteIdx++) {
  1106→          for (let bitInByte = 0; bitInByte < 8; bitInByte += 2) {
  1107→            const bitIdx = byteIdx * 8 + bitInByte;
  1108→            if (constrainedErasures[bitIdx] || constrainedErasures[bitIdx + 1]) {
  1109→              const baseIdx = byteIdx * 4 + (bitInByte >> 1);
  1110→              if (baseIdx < qScoresForInner.length) {
  1111→                qScoresForInner[baseIdx] = 0;
  1112→              }
  1113→            }
  1114→          }
  1115→        }
  1116→      }
  1117→
  1118→      // Try inner code decode (corrects errors in this individual read)
  1119→      // For RS: guaranteed minimum distance, so CRC is redundant
  1120→      // For LDPC: may pick wrong codeword (duplicate columns), so CRC is REQUIRED
  1121→      // For LDPC with soft-info: OSD-2/BP flips least-reliable bits
  1122→      let decodedData: Uint8Array;
  1123→      let corrected: number;
  1124→      try {
  1125→        if (useLDPC && innerLdpc) {
  1126→          // v57: If we have bit-level erasures (from arithmetic CRC or constrained
  1127→          // mapping), use the dedicated erasure decoder (peeling + Gaussian elim
  1128→          // over GF(2)). This is FAR more reliable than BP for erasures — BP
  1129→          // often fails to converge when Q-scores are 0, while the peeling
  1130→          // decoder directly solves the linear system.
  1131→          const hasErasures = constrainedErasures && constrainedErasures.some((e) => e);
  1132→          if (hasErasures) {
  1133→            // Convert boolean[] erasure bitmap to bit positions
  1134→            const erasePos: number[] = [];
  1135→            for (let i = 0; i < constrainedErasures!.length && i < innerN * 8; i++) {
  1136→              if (constrainedErasures![i]) erasePos.push(i);
  1137→            }
  1138→            if (erasePos.length > 0 && erasePos.length < innerN * 8) {
  1139→              try {
  1140→                const r = innerLdpc.decodeWithErasures(rsCodeword, erasePos);
  1141→                decodedData = r.data;
  1142→                corrected = r.corrected;
  1143→              } catch {
  1144→                // Erasure decode failed — fall through to BP
  1145→                const r = innerLdpc.decodeBeliefPropagation(rsCodeword, qScoresForInner, useGoldman);
  1146→                decodedData = r.data;
  1147→                corrected = r.corrected;
  1148→              }
  1149→            } else {
  1150→              // Too many erasures or none — use BP
  1151→              const r = innerLdpc.decodeBeliefPropagation(rsCodeword, qScoresForInner, useGoldman);
  1152→              decodedData = r.data;
  1153→              corrected = r.corrected;
  1154→            }
  1155→          } else if ((useConstrained || useSrt || useArithmetic)) {
  1156→            // For constrained/SRT/arithmetic mode without erasures, use BP
  1157→            const r = innerLdpc.decodeBeliefPropagation(rsCodeword, qScoresForInner, useGoldman);
  1158→            decodedData = r.data;
  1159→            corrected = r.corrected;
  1160→          } else {
  1161→            // Standard mode: use configured decoder mode
  1162→            const decoderMode = ldpcDecoderMode;
  1163→            if (decoderMode === "bp") {
  1164→              const r = innerLdpc.decodeBeliefPropagation(rsCodeword, qScoresForInner, useGoldman);
  1165→              decodedData = r.data;
  1166→              corrected = r.corrected;
  1167→            } else if (decoderMode === "osd") {
  1168→              if (qScoresForInner && qScoresForInner.length > 0) {
  1169→                try {
  1170→                  const r = innerLdpc.decodeWithSoftInfo(rsCodeword, qScoresForInner, useGoldman);
  1171→                  decodedData = r.data;
  1172→                  corrected = r.corrected;
  1173→                } catch {
  1174→                  const r = innerLdpc.decode(rsCodeword);
  1175→                  decodedData = r.data;
  1176→                  corrected = r.corrected;
  1177→                }
  1178→              } else {
  1179→                const r = innerLdpc.decode(rsCodeword);
  1180→                decodedData = r.data;
  1181→                corrected = r.corrected;
  1182→              }
  1183→            } else if (decoderMode === "hard") {
  1184→              const r = innerLdpc.decode(rsCodeword);
  1185→              decodedData = r.data;
  1186→              corrected = r.corrected;
  1187→            } else {
  1188→              // "auto" mode: neural LDPC decoder (hard → BP → multi-decision BP)
  1189→              // v61: For illumina (low-IDS), skip BP — it's too slow. Just use
  1190→              // hard-decision. If that fails, skip the read (rely on coverage).
  1191→              // BP is only useful for nanopore (high-IDS) where hard-decision
  1192→              // frequently fails.
  1193→              if (neuralLdpc && (channel === "nanopore" || channel === "pacbio")) {
  1194→                try {
  1195→                  const r = neuralLdpc.decode(rsCodeword, qScoresForInner, useGoldman);
  1196→                  decodedData = r.data;
  1197→                  corrected = r.corrected;
  1198→                } catch {
  1199→                  if (qScoresForInner && qScoresForInner.length > 0) {
  1200→                    const r = innerLdpc.decodeBeliefPropagation(rsCodeword, qScoresForInner, useGoldman);
  1201→                    decodedData = r.data;
  1202→                    corrected = r.corrected;
  1203→                  } else {
  1204→                    throw new Error("Neural decode failed and no Q-scores for BP");
  1205→                  }
  1206→                }
  1207→              } else {
  1208→                // v61: illumina fast path — hard-decision only, no BP/BFA.
  1209→                // With 10x coverage, we can afford to skip noisy reads.
  1210→                const r = innerLdpc.decode(rsCodeword);
  1211→                decodedData = r.data;
  1212→                corrected = r.corrected;
  1213→              }
  1214→            }
  1215→          }
  1216→        } else {
  1217→          const r = innerRs.decode(rsCodeword);
  1218→          decodedData = r.data;
  1219→          corrected = r.corrected;
  1220→        }
  1221→      } catch {
  1222→        // This read has too many errors — skip it, try next read
  1223→        continue;
  1224→      }
  1225→
  1226→      // For LDPC: verify CRC-16 to detect wrong-codeword picks
  1227→      // OPTIMIZATION: If corrected==0 (no errors corrected), skip re-encode
  1228→      // and use received data directly for CRC check.
  1229→      if (useLDPC && useArithmeticV2) {
  1230→        // v62: Arithmetic-v2 mode — no CRC-16 (per-block CRC-8 already validated).
  1231→        // The LDPC decodedData is just the payload (no address in the LDPC codeword).
  1232→        // The address comes from the direct DNA (already in innerBlock[0..3]).
  1233→        //
  1234→        // v62 fix: Do NOT compare received parity with re-encoded parity.
  1235→        // The received parity may have errors (arithmetic termination corruption)
  1236→        // that the LDPC erasure decoder already corrected. The erasure decoder
  1237→        // guarantees the syndrome is zero after correction, so we trust its output.
  1238→        // Just verify the address matches the expected oligo index.
  1239→        try {
  1240→          // Get address from the direct DNA (innerBlock[0..3])
  1241→          const whitenedAddr = innerBlock.slice(0, layout.addressBytes);
  1242→          const addr = unwhitenAddress(whitenedAddr);
  1243→          const decodedIndex = (addr[0] << 16) | (addr[1] << 8) | addr[2];
  1244→          if (decodedIndex !== oligoIdx) continue;
  1245→
  1246→          // Success! Extract payload (decodedData is just the payload for arithmetic-v2)
  1247→          const seed = addr[3];
  1248→          let payload = decodedData.slice(0, layout.payloadBytes);
  1249→          if (seed !== 0) payload = xorWithSeed(payload, seed);
  1250→          payloads.set(oligoIdx, payload);
  1251→          perOligo.push({
  1252→            index: oligoIdx, readCount: clusterReads.length, consensusLength: dna.length,
  1253→            crcPassed: true, innerRS: { corrected, success: true },
  1254→            seed, payloadBytes: payload, isParity: oligoIdx >= metadata.outerRS.k,
  1255→            strategy: 'per_read',
  1256→          });
  1257→          foundValidRead = true;
  1258→          break;
  1259→        } catch {
  1260→          continue;
  1261→        }
  1262→      } else if (useLDPC && !useConstrained && !useSrt) {
  1263→        if (corrected === 0) {
  1264→          // Fast path: zero-syndrome, but still check CRC (LDPC may have
  1265→          // duplicate columns → wrong codeword with zero syndrome)
  1266→          // Use received rsCodeword for CRC (no need to re-encode)
  1267→          const expectedCrc = crc16Bytes(rsCodeword);
  1268→          if (expectedCrc[0] !== crcBytes[0] || expectedCrc[1] !== crcBytes[1]) continue;
  1269→          // CRC passed — verify address and extract payload (same as slow path)
  1270→          const whitenedAddr = decodedData.slice(0, layout.addressBytes);
  1271→          const addr = unwhitenAddress(whitenedAddr);
  1272→          const decodedIndex = (addr[0] << 16) | (addr[1] << 8) | addr[2];
  1273→          if (decodedIndex !== oligoIdx) continue;
  1274→          const seed = addr[3];
  1275→          let payload = decodedData.slice(layout.addressBytes, layout.addressBytes + layout.payloadBytes);
  1276→          if (seed !== 0) payload = xorWithSeed(payload, seed);
  1277→          payloads.set(oligoIdx, payload);
  1278→          perOligo.push({
  1279→            index: oligoIdx, readCount: clusterReads.length, consensusLength: dna.length,
  1280→            crcPassed: true, innerRS: { corrected, success: true },
  1281→            seed, payloadBytes: payload, isParity: oligoIdx >= metadata.outerRS.k,
  1282→            strategy: 'per_read',
  1283→          });
  1284→          foundValidRead = true;
  1285→          break;
  1286→        }
  1287→        // Slow path: errors corrected, re-encode for canonical form + CRC
  1288→        const reEncoded = innerLdpc!.encode(decodedData);
  1289→        const expectedCrc = crc16Bytes(reEncoded);
  1290→        const crcMatch = expectedCrc[0] === crcBytes[0] && expectedCrc[1] === crcBytes[1];
  1291→        if (!crcMatch) {
  1292→          continue;
  1293→        }
  1294→        const whitenedAddr = reEncoded.slice(0, layout.addressBytes);
  1295→        const addr = unwhitenAddress(whitenedAddr);
  1296→        const decodedIndex = (addr[0] << 16) | (addr[1] << 8) | addr[2];
  1297→        if (decodedIndex !== oligoIdx) {
  1298→          continue;
  1299→        }
  1300→        // Success! Extract payload
  1301→        const seed = addr[3];
  1302→        let payload = reEncoded.slice(layout.addressBytes, layout.addressBytes + layout.payloadBytes);
  1303→        if (seed !== 0) payload = xorWithSeed(payload, seed);
  1304→        payloads.set(oligoIdx, payload);
  1305→        perOligo.push({
  1306→          index: oligoIdx, readCount: clusterReads.length, consensusLength: dna.length,
  1307→          crcPassed: true, innerRS: { corrected, success: true },
  1308→          seed, payloadBytes: payload, isParity: oligoIdx >= metadata.outerRS.k,
  1309→          strategy: 'per_read',
  1310→        });
  1311→        foundValidRead = true;
  1312→        break;
  1313→      } else if (useLDPC && (useConstrained || useSrt || useArithmetic)) {
  1314→        // Constrained mode: skip CRC (CRC bytes have erasures).
  1315→        // Rely on LDPC syndrome check (already passed if we got here).
  1316→        // Re-encode to get canonical form and verify address.
  1317→        const reEncoded = innerLdpc!.encode(decodedData);
  1318→        const whitenedAddr = reEncoded.slice(0, layout.addressBytes);
  1319→        const addr = unwhitenAddress(whitenedAddr);
  1320→        const decodedIndex = (addr[0] << 16) | (addr[1] << 8) | addr[2];
  1321→        if (decodedIndex !== oligoIdx) {
  1322→          // Address mismatch — this read was clustered to the wrong oligo
  1323→          continue;
  1324→        }
  1325→        // Success! Extract payload (seed is always 0 for constrained mode)
  1326→        let payload = reEncoded.slice(layout.addressBytes, layout.addressBytes + layout.payloadBytes);
  1327→        payloads.set(oligoIdx, payload);
  1328→        perOligo.push({
  1329→          index: oligoIdx, readCount: clusterReads.length, consensusLength: dna.length,
  1330→          crcPassed: true, innerRS: { corrected, success: true },
  1331→          seed: 0, payloadBytes: payload, isParity: oligoIdx >= metadata.outerRS.k,
  1332→          strategy: 'per_read',
  1333→        });
  1334→        foundValidRead = true;
  1335→        break;
  1336→      } else if (useLDPC && useSrt) {
  1337→        // SRT mode: skip CRC (SRT injects errors that change CRC bytes).
  1338→        // Rely on LDPC syndrome check + address verification.
  1339→        // The LDPC decoder corrects the injected SRT errors automatically.
  1340→        const reEncoded = innerLdpc!.encode(decodedData);
  1341→        const whitenedAddr = reEncoded.slice(0, layout.addressBytes);
  1342→        const addr = unwhitenAddress(whitenedAddr);
  1343→        const decodedIndex = (addr[0] << 16) | (addr[1] << 8) | addr[2];
  1344→        if (decodedIndex !== oligoIdx) {
  1345→          continue;
  1346→        }
  1347→        let payload = reEncoded.slice(layout.addressBytes, layout.addressBytes + layout.payloadBytes);
  1348→        payloads.set(oligoIdx, payload);
  1349→        perOligo.push({
  1350→          index: oligoIdx, readCount: clusterReads.length, consensusLength: dna.length,
  1351→          crcPassed: true, innerRS: { corrected, success: true },
  1352→          seed: 0, payloadBytes: payload, isParity: oligoIdx >= metadata.outerRS.k,
  1353→        });
  1354→        foundValidRead = true;
  1355→        break;
  1356→      } else if (useLDPC && useArithmetic) {
  1357→        // v57: Arithmetic mode — encoder DROPPED CRC-16 to make room for
  1358→        // per-block CRC-8 sync markers (see codec.ts bytesToArithmeticDnaCrc).
  1359→        // The crcBytes extracted from innerBlock.slice(innerN) are NOT real
  1360→        // CRC bytes — they're whatever the arithmetic decoder produced after
  1361→        // the LDPC codeword region (typically zeros from padding).
  1362→        //
  1363→        // So we MUST NOT verify CRC-16 here. Instead, rely on:
  1364→        //   - LDPC syndrome check (passed if we got here)
  1365→        //   - Per-block CRC-8 (already verified by arithmeticDnaToBytesCrc)
  1366→        //   - Address verification (decodedIndex must match oligoIdx)
  1367→        const reEncoded = innerLdpc!.encode(decodedData);
  1368→        const whitenedAddr = reEncoded.slice(0, layout.addressBytes);
  1369→        const addr = unwhitenAddress(whitenedAddr);
  1370→        const decodedIndex = (addr[0] << 16) | (addr[1] << 8) | addr[2];
  1371→        if (decodedIndex !== oligoIdx) {
  1372→          continue;
  1373→        }
  1374→        let payload = reEncoded.slice(layout.addressBytes, layout.addressBytes + layout.payloadBytes);
  1375→        payloads.set(oligoIdx, payload);
  1376→        perOligo.push({
  1377→          index: oligoIdx, readCount: clusterReads.length, consensusLength: dna.length,
  1378→          crcPassed: true, innerRS: { corrected, success: true },
  1379→          seed: 0, payloadBytes: payload, isParity: oligoIdx >= metadata.outerRS.k,
  1380→          strategy: 'per_read',
  1381→        });
  1382→        foundValidRead = true;
  1383→        break;
  1384→      }
  1385→
  1386→      // RS path: RS succeeded — trust the RS-corrected data.
  1387→      // RS guarantees correctness within correction capacity. CRC is redundant
  1388→      // and CRC bytes can be corrupted independently (they're outside RS).
  1389→      const reEncoded = innerRs.encode(decodedData);
  1390→

  1392→      // If the address was corrupted by substitution, the read was clustered

  1397→      const decodedIndex = (addr[0] << 16) | (addr[1] << 8) | addr[2];

  1399→        // Address mismatch — this read was clustered to the wrong oligo
  1400→        continue;
  1401→      }

  1404→      const whitenedAddress = reEncoded.slice(0, layout.addressBytes);
  1405→      const address = unwhitenAddress(whitenedAddress);
  1406→      const seed = address[3];

  1409→

  1413→        crcPassed: true, innerRS: { corrected, success: true },
  1414→        seed, payloadBytes: payload, isParity: oligoIdx >= metadata.outerRS.k,
  1415→        strategy: 'per_read',
  1416→      });

  1418→      break; // Found a valid read — move to next oligo
  1419→    }
  1420→
  1421→    if (foundValidRead) {

  1424→      continue;
  1425→    }

  1430→    // For illumina (low-IDS): Fast weighted consensus (Q-score weighted majority

  1432→    //
  1433→    // For nanopore/pacbio (high-IDS): Progressive MSA consensus. The naïve
  1434→    // column-wise plurality vote (decodeWithFastConsensus) breaks under indels

  1440→    //
  1441→    // v61 CRITICAL FIX: Verify CRC before accepting. Without this, the consensus

  1456→            addressBytes: layout.addressBytes,
  1457→            convEncodedBytes: layout.convEncodedBytes,

  1459→          } : null,
  1460→        );

  1594→    // STRATEGY 3: HMM-based soft-consensus (slower, handles indels).

  1599→      let refDna = refRead.sequence;
  1600→      if (refDna.length < expectedDnaLen) {

  1632→          }
  1633→        }

  1635→        // HMM soft-consensus failed — fall through to OSD post-pass
  1636→      }
  1637→    }
  1638→
  1639→    // OSD-2 post-pass: try ordered statistics decoding before giving up.
  1640→    // This recovers many LDPC failures at the cost of ~2ms per oligo.
  1641→    // Only attempt for nanopore/pacbio channels where inner decode often fails
  1642→    // due to indels that the Viterbi preprocessor couldn't fully correct.
  1643→    if ((channel === "nanopore" || channel === "pacbio") && useLDPC && innerLdpc && clusterReads.length >= 1) {

  1645→        const { osdDecode } = await import('./osd-full');

  1647→

  1649→        const refRead = clusterReads[0];
  1650→        let refDna = refRead.sequence;

  1658→        let rsCodewordForOsd: Uint8Array;

  1672→          for (let baseIdx = 0; baseIdx < qScores.length && baseIdx * 2 < nBits; baseIdx++) {
  1673→            // Q-score to LLR: LLR = (Q - 33) * ln(10) / 10 * 2 (per-bit)
  1674→            const qVal = qScores[baseIdx];
  1675→            const llrMag = Math.max(0.1, ((qVal - 33) * 0.2303)); // 0.2303 ≈ ln(10)/10
  1676→            // Two bits per base (direct mapping)
  1677→            const byteIdx0 = Math.floor(baseIdx / 4);
  1678→            const bitIdx0 = (baseIdx % 4) * 2;
  1679→            if (byteIdx0 < innerN) {

  1682→              const bit0 = (byteVal >> (7 - bitIdx0)) & 1;
  1683→              llr[baseIdx * 2] = bit0 === 0 ? llrMag : -llrMag;
  1684→              // bit 1
  1685→              const bit1 = (byteVal >> (7 - bitIdx0 - 1)) & 1;
  1686→              llr[baseIdx * 2 + 1] = bit1 === 0 ? llrMag : -llrMag;
  1687→            }
  1688→          }
  1689→        } else {

  1691→          for (let i = 0; i < nBits; i++) {
  1692→            const byteIdx = Math.floor(i / 8);
  1693→            const bitPos = 7 - (i % 8);
  1694→            const bit = (rsCodewordForOsd[byteIdx] >> bitPos) & 1;

  1696→          }
  1697→        }
  1698→
  1699→        // Get parity-check matrix from LDPC code
  1700→        const H = innerLdpc.parityCheckMatrix;
  1701→

  1703→        const crcAndAddrCheck = (codewordBits: Uint8Array): boolean => {
  1704→          // Convert bit array to byte array

  1708→            for (let b = 0; b < 8; b++) {
  1709→              byteVal |= (codewordBits[i * 8 + b] << (7 - b));

  1711→            codewordBytes[i] = byteVal;
  1712→          }

  1730→          const addr = unwhitenAddress(whitenedAddr);

  1733→        };

  1736→        const osdResult = osdDecode(llr, H, crcAndAddrCheck, {

  1738→          k: innerK * 8,
  1739→        });

  1745→          for (let i = 0; i < innerN; i++) {
  1746→            let byteVal = 0;
  1747→            for (let b = 0; b < 8; b++) {
  1748→              byteVal |= (codewordBits[i * 8 + b] << (7 - b));
  1749→            }
  1750→            codewordBytes[i] = byteVal;
  1751→          }

  1754→          const seed = 0; // OSD doesn't recover seed
  1755→          let payload = reEncoded.slice(layout.addressBytes, layout.addressBytes + layout.payloadBytes);
  1756→          payloads.set(oligoIdx, payload);
  1757→          perOligo.push({
  1758→            index: oligoIdx, readCount: clusterReads.length, consensusLength: refDna.length,
  1759→            crcPassed: true, innerRS: { corrected: osdResult.candidatesTried, success: true },
  1760→            seed, payloadBytes: payload, isParity: oligoIdx >= metadata.outerRS.k,
  1761→            strategy: 'osd_post_pass',
  1762→          });
  1763→          continue;
  1764→        }
  1765→      } catch {
  1766→        // OSD not available or failed — fall through to erasure
  1767→      }
  1768→    }
  1769→
  1770→    // STRATEGY 4: Mark as erasure for outer RS to recover.
  1771→    erasedIndices.push(oligoIdx);
  1772→    oligosFailedInnerRS++;

  1780→  }
  1781→
  1782→  // 3) Outer RS: recover missing oligos via erasure decoding.
  1783→  // With per-read decode, there are NO silent consensus errors — every
  1784→  // payload in the map was individually validated by RS+CRC. So the outer
  1785→  // RS only needs to recover genuinely missing oligos (erasures).
  1786→  let oligosFailedOuterRS = 0;
  1787→  const recoveredPayloads = new Map<number, Uint8Array>();
  1788→
  1789→  if (!useOuterRS) {
  1790→    for (const [idx, payload] of payloads) {
  1791→      recoveredPayloads.set(idx, payload);
  1792→    }
  1793→  } else if (useGF216 && outerRs216) {
  1794→    const numPairs = Math.floor(layout.payloadBytes / 2);
  1795→    let outerSuccessCount = 0, outerFailCount = 0, outerSkipCount = 0;
  1796→    for (let pairIdx = 0; pairIdx < numPairs; pairIdx++) {
