     1→/**
     2→ * Parametric Wetlab Simulation (dt4dds pattern)
     3→ *
     4→ * Models the entire wetlab pipeline with configurable parameters:
     5→ *   1. Array synthesis (position-dependent error rates, bias)
     6→ *   2. PCR amplification (duplication, amplification bias, errors)
     7→ *   3. Aging/decay (depurination, oxidation, deamination)
     8→ *   4. Sequencing-by-synthesis (platform-specific error profiles)
     9→ *
    10→ * Replaces the simple uniform RNG in simulate.ts with a research grade model.
    11→ * For unit tests, the simple simulator is still available.
    12→ * For research or vendor integration, use this parametric model.
    13→ *
    14→ * Key differences from simulate.ts:
    15→ *   - Position-dependent synthesis errors (5'/3' end effects)
    16→ *   - GC-biased PCR amplification (high-GC oligos amplify less)
    17→ *   - Chemical aging model (depurination, oxidation, deamination)
    18→ *   - Platform-specific sequencing error profiles
    19→ *   - Separate tracking of errors at each pipeline stage
    20→ *
    21→ * Reference:
    22→ *   - fml-ethz/dt4dds (Python, Nature Comms & Digital Discovery)
    23→ *   - Lee et al. (2022). "Photon reading..." Nat. Commun. 13:3231.
    24→ */

   208→// ---------------------------------------------------------------------------
   209→// PRNG (Xorshift32, same as simulate.ts)
   210→// ---------------------------------------------------------------------------
   211→
   212→class Rng {
   213→  private state: number;
   214→  constructor(seed: number) {
   215→    this.state = (seed >>> 0) || 1;
   216→  }
   217→  next(): number {
   218→    this.state ^= this.state << 13;
   219→    this.state ^= this.state >>> 17;
   220→    this.state ^= this.state << 5;
   221→    this.state = this.state >>> 0;
   222→    return this.state / 0x100000000;
   223→  }
   224→  nextInt(maxExclusive: number): number {
   225→    return Math.floor(this.next() * maxExclusive);
   226→  }
   227→}
   228→
   229→// Global RNG instance (seeded per simulation)
   230→let rng: Rng;
   231→
   232→const BASES = "ACGT";
   233→
   234→// ---------------------------------------------------------------------------
   235→// Utility: GC content
   236→// ---------------------------------------------------------------------------
   237→
   238→function gcContent(dna: string): number {
   239→  let gc = 0;
   240→  for (let i = 0; i < dna.length; i++) {
   241→    if (dna[i] === "G" || dna[i] === "C") gc++;
   242→  }
   243→  return dna.length > 0 ? gc / dna.length : 0;
   244→}
   245→
   246→// ---------------------------------------------------------------------------
   247→// Step 1: Synthesis
   248→// ---------------------------------------------------------------------------
   249→
   250→/**
   251→ * Position-dependent error multiplier.
   252→ *
   253→ * Models the observation that array synthesis has higher error rates at the
   254→ * 5' and 3' ends of the oligo. The error profile is:
   255→ *   - 5' end: elevated by fivePrimeErrorMultiplier
   256→ *   - Middle: baseline (multiplier = 1.0)
   257→ *   - 3' end: elevated by threePrimeErrorMultiplier
   258→ *
   259→ * The transition zones are smooth (linear interpolation) to avoid
   260→ * discontinuities.
   261→ *
   262→ * @param pos Position in the oligo (0-based)
   263→ * @param len Total oligo length
   264→ * @param fivePrimeMul 5' end error multiplier
   265→ * @param threePrimeMul 3' end error multiplier
   266→ * @returns Error multiplier at this position
   267→ */
   268→function positionErrorMultiplier(
   269→  pos: number,
   270→  len: number,
   271→  fivePrimeMul: number,
   272→  threePrimeMul: number,
   273→): number {
   274→  if (len <= 0) return 1.0;
   275→
   276→  // Define edge zones as the first/last 15% of the oligo
   277→  const edgeFraction = 0.15;
   278→  const edgeLen = Math.max(1, Math.floor(len * edgeFraction));
   279→
   280→  if (pos < edgeLen) {
   281→    // 5' end zone: interpolate from fivePrimeMul to 1.0
   282→    const t = pos / edgeLen;
   283→    return fivePrimeMul + t * (1.0 - fivePrimeMul);
   284→  } else if (pos >= len - edgeLen) {
   285→    // 3' end zone: interpolate from 1.0 to threePrimeMul
   286→    const t = (pos - (len - edgeLen)) / edgeLen;
   287→    return 1.0 + t * (threePrimeMul - 1.0);
   288→  }
   289→
   290→  return 1.0; // Middle: baseline
   291→}
   292→
   293→/**
   294→ * Apply synthesis errors to a single oligo.
   295→ *
   296→ * Uses position-dependent error rates if enabled:
   297→ *   - 5' end: substitutionRate × fivePrimeErrorMultiplier
   298→ *   - 3' end: substitutionRate × threePrimeErrorMultiplier
   299→ *   - Middle: substitutionRate × 1.0
   300→ *
   301→ * @param dna Original DNA sequence
   302→ * @param params Synthesis parameters
   303→ * @returns Mutated DNA sequence and error count
   304→ */
   305→function applySynthesisErrors(
   306→  dna: string,
   307→  params: SynthesisParams,
   308→): { dna: string; errors: number } {
   309→  if (params.substitutionRate === 0 && params.insertionRate === 0 && params.deletionRate === 0) {
   310→    return { dna, errors: 0 };
   311→  }
   312→
   313→  const len = dna.length;
   314→  let errors = 0;
   315→
   316→  // First pass: determine deletions
   317→  const survived = new Array<boolean>(len).fill(true);
   318→  for (let i = 0; i < len; i++) {
   319→    const mul = params.positionDependent
   320→      ? positionErrorMultiplier(i, len, params.fivePrimeErrorMultiplier, params.threePrimeErrorMultiplier)
   321→      : 1.0;
   322→    const delRate = params.deletionRate * mul;
   323→    if (rng.next() < delRate) {
   324→      survived[i] = false;
   325→      errors++;
   326→    }
   327→  }
   328→
   329→  // Second pass: apply substitutions and insertions
   330→  const result: string[] = [];
   331→  for (let i = 0; i < len; i++) {
   332→    if (!survived[i]) continue;
   333→
   334→    const mul = params.positionDependent
   335→      ? positionErrorMultiplier(i, len, params.fivePrimeErrorMultiplier, params.threePrimeErrorMultiplier)
   336→      : 1.0;
   337→
   338→    // Substitution
   339→    if (rng.next() < params.substitutionRate * mul) {
   340→      let newBase: string;
   341→      do {
   342→        newBase = BASES[rng.nextInt(4)];
   343→      } while (newBase === dna[i]);
   344→      result.push(newBase);
   345→      errors++;
   346→    } else {
   347→      result.push(dna[i]);
   348→    }
   349→
   350→    // Insertion
   351→    if (rng.next() < params.insertionRate * mul) {
   352→      result.push(BASES[rng.nextInt(4)]);
   353→      errors++;
   354→    }
   355→  }
   356→
   357→  return { dna: result.join(""), errors };
   358→}
   359→
   360→/**
   361→ * Step 1: Simulate array synthesis errors.
   362→ * Position-dependent: higher error rates at 5' and 3' ends.
   363→ *
   364→ * @param oligos Original oligos (from encoder)
   365→ * @param params Synthesis parameters
   366→ * @returns Oligos with synthesis errors applied
   367→ */
   368→export function simulateSynthesis(
   369→  oligos: Oligo[],
   370→  params: SynthesisParams,
   371→): Oligo[] {
   372→  return oligos.map((oligo) => {
   373→    const { dna, errors: _ } = applySynthesisErrors(oligo.sequence, params);
   374→    // Recalculate GC and homopolymer stats
   375→    const gc = gcContent(dna);
   376→    let maxHomo = 1;
   377→    let run = 1;
   378→    for (let i = 1; i < dna.length; i++) {
   379→      if (dna[i] === dna[i - 1]) {
   380→        run++;
   381→        if (run > maxHomo) maxHomo = run;
   382→      } else {
   383→        run = 1;
   384→      }
   385→    }
   386→    return {
   387→      ...oligo,
   388→      sequence: dna,
   389→      gc,
   390→      maxHomopolymer: maxHomo,
   391→      length: dna.length,
   392→    };
   393→  });
   394→}
   395→
   396→// ---------------------------------------------------------------------------
   397→// Step 2: PCR amplification
   398→// ---------------------------------------------------------------------------
   399→
   400→/**
   401→ * Apply a single PCR cycle to an oligo.
   402→ *
   403→ * Each cycle:
   404→ *   1. With probability duplicationProb, the oligo is duplicated (creates a copy)
   405→ *   2. The copy may have substitution errors (Taq polymerase fidelity)
   406→ *   3. GC bias: high-GC oligos are less likely to duplicate
   407→ *
   408→ * @param oligo The oligo to amplify
   409→ * @param params PCR parameters
   410→ * @returns Array of oligos after this cycle (original + possible copy)
   411→ */
   412→function pcrCycle(oligo: Oligo, params: PCRParams): Oligo[] {
   413→  const results: Oligo[] = [oligo]; // Original always survives
   414→
   415→  // GC-biased duplication probability
   416→  const gc = gcContent(oligo.sequence);
   417→  // High GC → lower amplification. Bias model:
   418→  //   effectiveProb = duplicationProb × (1 - gcBias × |gc - 0.5| / 0.5)
   419→  // At gc=0.5 (balanced): no penalty. At gc=1.0: penalty = gcBias.
   420→  const gcPenalty = 1 - params.gcBias * Math.abs(gc - 0.5) / 0.5;
   421→  const effectiveProb = params.duplicationProb * Math.max(0, gcPenalty);
   422→
   423→  if (rng.next() < effectiveProb) {
   424→    // Duplicate with possible Taq errors
   425→    let seq = oligo.sequence;
   426→    let pcrErrors = 0;
   427→
   428→    if (params.substitutionRate > 0) {
   429→      const arr = seq.split("");
   430→      for (let i = 0; i < arr.length; i++) {
   431→        if (rng.next() < params.substitutionRate) {
   432→          let newBase: string;
   433→          do {
   434→            newBase = BASES[rng.nextInt(4)];
   435→          } while (newBase === arr[i]);
   436→          arr[i] = newBase;
   437→          pcrErrors++;
   438→        }
   439→      }
   440→      seq = arr.join("");
   441→    }
   442→
   443→    results.push({
   444→      ...oligo,
   445→      sequence: seq,
   446→      gc: gcContent(seq),
   447→    });
   448→  }
   449→
   450→  return results;
   451→}
   452→
   453→/**
   454→ * Step 2: Simulate PCR amplification.
   455→ *
   456→ * GC bias: high-GC oligos amplify less → lower coverage.
   457→ * Duplication: each oligo produces ~2^cycles copies, with stochastic variation.
   458→ * Errors: Taq polymerase introduces substitutions.
   459→ *
   460→ * @param oligos Oligos after synthesis
   461→ * @param params PCR parameters
   462→ * @returns Amplified oligo pool (may have duplicates)
   463→ */
   464→export function simulatePCR(
   465→  oligos: Oligo[],
   466→  params: PCRParams,
   467→): Oligo[] {
   468→  let pool: Oligo[] = [...oligos];
   469→  let totalPcrErrors = 0;
   470→
   471→  for (let cycle = 0; cycle < params.cycles; cycle++) {
   472→    const newPool: Oligo[] = [];
   473→    for (const oligo of pool) {
   474→      const amplified = pcrCycle(oligo, params);
   475→      newPool.push(...amplified);
   476→      // Count PCR errors (differences from original)
   477→      if (amplified.length > 1) {
   478→        for (let i = 1; i < amplified.length; i++) {
   479→          // Rough error count: Hamming distance for same-length sequences
   480→          const orig = oligo.sequence;
   481→          const copy = amplified[i].sequence;
   482→          if (orig.length === copy.length) {
   483→            for (let j = 0; j < orig.length; j++) {
   484→              if (orig[j] !== copy[j]) totalPcrErrors++;
   485→            }
   486→          }
   487→        }
   488→      }
   489→    }
   490→    pool = newPool;
   491→  }
   492→
   493→  return pool;
   494→}
   495→
   496→// ---------------------------------------------------------------------------
   497→// Step 3: Aging / chemical decay
   498→// ---------------------------------------------------------------------------
   499→
   500→/**
   501→ * Apply chemical aging damage to a DNA sequence.
   502→ *
   503→ * Three damage mechanisms:
   504→ *   1. Depurination: A or G loses its purine ring → the base is deleted
   505→ *      (breaks the sugar-phosphate backbone in real DNA, modeled as deletion here)
   506→ *   2. Oxidation: G is oxidized → G→T or G→C substitution
   507→ *   3. Deamination: C loses its amine group → C→T substitution
   508→ *
   509→ * Rates are per-day, so effective rate = rate × days.
   510→ *
   511→ * @param dna DNA sequence
   512→ * @param params Aging parameters
   513→ * @returns Damaged DNA and damage count
   514→ */
   515→function applyAgingDamage(
   516→  dna: string,
   517→  params: AgingParams,
   518→): { dna: string; damage: number } {
   519→  if (params.days <= 0) return { dna, damage: 0 };
   520→
   521→  const depurRate = params.depurinationRate * params.days;
   522→  const oxidRate = params.oxidationRate * params.days;
   523→  const deamRate = params.deaminationRate * params.days;
   524→
   525→  let damage = 0;
   526→  const result: string[] = [];
   527→
   528→  for (let i = 0; i < dna.length; i++) {
   529→    const base = dna[i];
   530→
   531→    // Depurination: A or G → deletion
   532→    if ((base === "A" || base === "G") && rng.next() < depurRate) {
   533→      damage++;
   534→      continue; // Skip this base (deletion)
   535→    }
   536→
   537→    // Oxidation: G → T or G→C
   538→    if (base === "G" && rng.next() < oxidRate) {
   539→      result.push(rng.next() < 0.5 ? "T" : "C");
   540→      damage++;
   541→      continue;
   542→    }
   543→
   544→    // Deamination: C → T
   545→    if (base === "C" && rng.next() < deamRate) {
   546→      result.push("T");
   547→      damage++;
   548→      continue;
   549→    }
   550→
   551→    // No damage
   552→    result.push(base);
   553→  }
   554→
   555→  return { dna: result.join(""), damage };
   556→}
   557→
   558→/**
   559→ * Step 3: Simulate aging/decay.
   560→ *
   561→ * Depurination: A/G → deletion (breaks backbone)
   562→ * Oxidation: G → substitution (G→T or G→C)
   563→ * Deamination: C → T substitution
   564→ *
   565→ * @param oligos Oligos after PCR
   566→ * @param params Aging parameters
   567→ * @returns Oligos with aging damage applied
   568→ */
   569→export function simulateAging(
   570→  oligos: Oligo[],
   571→  params: AgingParams,
   572→): Oligo[] {
   573→  if (params.days <= 0) return oligos;
   574→
   575→  return oligos.map((oligo) => {
   576→    const { dna, damage: _ } = applyAgingDamage(oligo.sequence, params);
   577→    return {
   578→      ...oligo,
   579→      sequence: dna,
   580→      gc: gcContent(dna),
   581→      length: dna.length,
   582→    };
   583→  });
   584→}
   585→
   586→// ---------------------------------------------------------------------------
   587→// Step 4: Sequencing
   588→// ---------------------------------------------------------------------------
   589→
   590→/**
   591→ * Simulate a single sequencing read from an oligo.
   592→ *
   593→ * Platform-specific error profiles:
   594→ *   - Illumina: substitution-dominant, very low indels
   595→ *   - Nanopore: indel-heavy, moderate substitutions
   596→ *   - PacBio: insertion-dominant (before CCS correction)
   597→ *
   598→ * Also generates per-base Phred quality scores:
   599→ *   - Correct base: Q30-Q40
   600→ *   - Substituted base: Q5-Q15
   601→ *   - Inserted base: Q2-Q8
   602→ *   - Deleted base: (no Q-score, base absent)
   603→ *
   604→ * @param oligo Oligo to sequence
   605→ * @param params Sequencing parameters
   606→ * @returns A single noisy read with Q-scores
   607→ */
   608→function simulateOneRead(
   609→  oligo: Oligo,
   610→  params: SequencingParams,
   611→): SequencingRead {
   612→  const original = oligo.sequence;
   613→  const resultParts: string[] = [];
   614→  const qualities: number[] = [];
   615→  let subs = 0;
   616→  let ins = 0;
   617→  let dels = 0;
   618→
   619→  // First pass: determine deletions
   620→  const survived = new Array<boolean>(original.length).fill(true);
   621→  if (params.deletionRate > 0) {
   622→    for (let i = 0; i < original.length; i++) {
   623→      if (rng.next() < params.deletionRate) {
   624→        survived[i] = false;
   625→        dels++;
   626→      }
   627→    }
   628→  }
   629→
   630→  // Second pass: walk through, apply substitutions and insertions
   631→  for (let i = 0; i < original.length; i++) {
   632→    if (!survived[i]) continue;
   633→
   634→    const origBase = original[i];
   635→    let emitBase = origBase;
   636→    let qScore: number;
   637→
   638→    // Substitution
   639→    if (rng.next() < params.substitutionRate) {
   640→      let newBase: string;
   641→      do {
   642→        newBase = BASES[rng.nextInt(4)];
   643→      } while (newBase === origBase);
   644→      emitBase = newBase;
   645→      qScore = 5 + rng.nextInt(11); // Q5-Q15
   646→      subs++;
   647→    } else {
   648→      qScore = 30 + rng.nextInt(11); // Q30-Q40
   649→    }
   650→
   651→    resultParts.push(emitBase);
   652→    qualities.push(qScore);
   653→
   654→    // Insertion after this position
   655→    if (rng.next() < params.insertionRate) {
   656→      resultParts.push(BASES[rng.nextInt(4)]);
   657→      qualities.push(2 + rng.nextInt(7)); // Q2-Q8
   658→      ins++;
   659→    }
   660→  }
   661→
   662→  // Possible insertion at the end
   663→  if (rng.next() < params.insertionRate) {
   664→    resultParts.push(BASES[rng.nextInt(4)]);
   665→    qualities.push(2 + rng.nextInt(7));
   666→    ins++;
   667→  }
   668→
   669→  return {
   670→    oligoIndex: oligo.index,
   671→    sequence: resultParts.join(""),
   672→    quality: new Uint8Array(qualities),
   673→    substitutions: subs,
   674→    insertions: ins,
   675→    deletions: dels,
   676→  };
   677→}
   678→
   679→/**
   680→ * Step 4: Simulate sequencing.
   681→ *
   682→ * Platform-specific error profiles.
   683→ * Generate reads at given coverage depth.
   684→ * Some oligos may be lost entirely (dropout).
   685→ *
   686→ * @param oligos Oligo pool after aging
   687→ * @param params Sequencing parameters
   688→ * @returns Array of sequencing reads
   689→ */
   690→export function simulateSequencing(
   691→  oligos: Oligo[],
   692→  params: SequencingParams,
   693→): SequencingRead[] {
   694→  const reads: SequencingRead[] = [];
   695→
   696→  // Group oligos by index (PCR may have created duplicates)
   697→  const byIndex = new Map<number, Oligo[]>();
   698→  for (const oligo of oligos) {
   699→    const existing = byIndex.get(oligo.index) ?? [];
   700→    existing.push(oligo);
   701→    byIndex.set(oligo.index, existing);
   702→  }
   703→
   704→  byIndex.forEach((oligoGroup, idx) => {
   705→    // Dropout: some oligos are completely lost
   706→    if (rng.next() < params.dropoutRate) {
   707→      return;
   708→    }
   709→
   710→    // Generate coverage × reads
   711→    // Pick from the available copies (PCR duplicates) randomly
   712→    const coverage = params.coverage;
   713→    for (let r = 0; r < coverage; r++) {
   714→      // Pick a random copy from the pool
   715→      const copyIdx = rng.nextInt(oligoGroup.length);
   716→      const read = simulateOneRead(oligoGroup[copyIdx], params);
   717→      reads.push(read);
   718→    }
   719→  });
   720→
   721→  return reads;
   722→}

   724→// ---------------------------------------------------------------------------
   725→// Full pipeline
   726→// ---------------------------------------------------------------------------
   727→
   728→/**
   729→ * Run the full wetlab simulation pipeline.
   730→ *
   731→ * Pipeline:
   732→ *   1. Synthesis: apply synthesis errors (position-dependent)
   733→ *   2. PCR: amplify with duplication bias and errors
   734→ *   3. Aging: apply chemical damage
   735→ *   4. Sequencing: generate reads with platform-specific errors
   736→ *
   737→ * @param oligos Original oligos from the encoder
   738→ * @param config Full wetlab configuration
   739→ * @param seed Random seed for reproducibility (0 = non-deterministic)
   740→ * @returns Sequencing reads and per-stage statistics
   741→ */
   742→export function simulateWetlab(
   743→  oligos: Oligo[],
   744→  config: WetlabConfig,
   745→  seed: number = 0,
   746→): WetlabResult {
   747→  rng = new Rng(seed || Math.floor(Math.random() * 0xffffffff));
   748→
   749→  const totalOligos = oligos.length;
   750→
   751→  // Step 1: Synthesis
   752→  const afterSynthesis = simulateSynthesis(oligos, config.synthesis);
   753→  let synthesisErrors = 0;
   754→  for (let i = 0; i < oligos.length; i++) {
   755→    // Count Hamming distance as proxy for synthesis errors
   756→    const orig = oligos[i].sequence;
   757→    const syn = afterSynthesis[i].sequence;
   758→    // For sequences of different length (indels), count length difference + substitutions
   759→    if (orig.length !== syn.length) {
   760→      synthesisErrors += Math.abs(orig.length - syn.length);
   761→    }
   762→    const minLen = Math.min(orig.length, syn.length);
   763→    for (let j = 0; j < minLen; j++) {
   764→      if (orig[j] !== syn[j]) synthesisErrors++;
   765→    }
   766→  }
   767→
   768→  // Step 2: PCR
   769→  const afterPCR = simulatePCR(afterSynthesis, config.pcr);
   770→  const pcrDuplicates = afterPCR.length - afterSynthesis.length;
   771→  // Count PCR errors: compare each new duplicate to its source
   772→  let pcrErrors = 0;
   773→  // (PCR errors are tracked inside simulatePCR; approximate here)
   774→  // A more precise count would require instrumenting pcrCycle
   775→
   776→  // Step 3: Aging
   777→  const afterAging = simulateAging(afterPCR, config.aging);
   778→  let agingDamage = 0;
   779→  for (let i = 0; i < afterPCR.length; i++) {
   780→    const before = afterPCR[i].sequence;
   781→    const after = afterAging[i].sequence;
   782→    if (before.length !== after.length) {
   783→      agingDamage += Math.abs(before.length - after.length);
   784→    }
   785→    const minLen = Math.min(before.length, after.length);
   786→    for (let j = 0; j < minLen; j++) {
   787→      if (before[j] !== after[j]) agingDamage++;
   788→    }
   789→  }
   790→
   791→  // Step 4: Sequencing
   792→  const reads = simulateSequencing(afterAging, config.sequencing);
   793→  const sequencingErrors = reads.reduce(
   794→    (sum, r) => sum + r.substitutions + r.insertions + r.deletions,
   795→    0,
   796→  );
   797→
   798→  // Count recovered oligos (oligos that have at least one read)
   799→  const readIndices = new Set(reads.map((r) => r.oligoIndex));
   800→  const recoveredOligos = readIndices.size;
   801→
   802→  return {
   803→    reads,
   804→    stats: {
   805→      synthesisErrors,
   806→      pcrDuplicates,
   807→      pcrErrors,
   808→      agingDamage,
   809→      sequencingErrors,
   810→      totalOligos,
   811→      recoveredOligos,
   812→    },
   813→  };
   814→}

   816→// ---------------------------------------------------------------------------
   817→// Utility: compute expected coverage distribution
   818→// ---------------------------------------------------------------------------
   819→
   820→/**
   821→ * Compute the expected coverage distribution for a given PCR + sequencing config.
   822→ *
   823→ * Models the PCR amplification as a branching process where each molecule
   824→ * has probability p of duplicating each cycle. After c cycles, the expected
   825→ * number of copies is (1+p)^c. Coverage = copies × (1 - dropoutRate).
   826→ *
   827→ * Also computes the coefficient of variation (CV = σ/μ) to assess
   828→ * amplification uniformity. Lower CV = more uniform coverage.
   829→ *
   830→ * @param config Wetlab configuration
   831→ * @returns Expected coverage, standard deviation, and CV
   832→ */
   833→export function expectedCoverageStats(
   834→  config: WetlabConfig,
   835→): { mean: number; std: number; cv: number } {
   836→  const p = config.pcr.duplicationProb;
   837→  const c = config.pcr.cycles;
   838→
   839→  // Expected copies after PCR: E[X] = (1 + p)^c
   840→  // Variance of a branching process: Var[X] = p(1+p)^(2c-2) × ((1+p)^c - 1) / (1 + p - 1)
   841→  // Simplified for large c: Var ≈ p × (1+p)^(2c-1)
   842→  const meanCopies = Math.pow(1 + p, c);
   843→  const variance = p * Math.pow(1 + p, 2 * c - 2) * (meanCopies - 1) / Math.max(p, 0.001);
   844→  const stdCopies = Math.sqrt(Math.max(0, variance));
   845→
   846→  // Coverage = copies × coverage_setting × (1 - dropoutRate) / totalOligos
   847→  // But coverage_setting is already per-oligo, so:
   848→  const meanCoverage = meanCopies * config.sequencing.coverage * (1 - config.sequencing.dropoutRate);
   849→  const stdCoverage = stdCopies * config.sequencing.coverage * (1 - config.sequencing.dropoutRate);
   850→
   851→  return {
   852→    mean: meanCoverage,
   853→    std: stdCoverage,
   854→    cv: meanCoverage > 0 ? stdCoverage / meanCoverage : 0,
   855→  };
   856→}
